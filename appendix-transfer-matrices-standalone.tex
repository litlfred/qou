\subsection*{How transfer matrices are used}

For a nucleus with $A = Z + N$ nucleons, the braid word has $A - 1$
generators. The transfer matrices at generator~$\sigma_i$ are composed as operators acting by left-multiplication, with crossings applied from right to left in the braid word:
\[
  M_i(q) = \prod_{j=A-1}^{i+1} T_{K_j, K_i}(q),
\]
where the product is taken in decreasing order of~$j$.
The Markov-traced volume at each generator is
\[
  \hat{V}_i(q) = z(q) \cdot (M_i)_{11} + (M_i)_{21}.
\]
The nuclear Skein polynomial is
\[
  P_{\mathcal{N}}(q) = \prod_{i=0}^{A-2} \hat{V}_i(q).
\]
This requires exactly $\binom{A}{2}$ matrix multiplications.

%% ============================================================
\section{The General Transfer Matrix}
%% ============================================================

The general transfer matrix has the form
\[
  T(c,d) = \begin{pmatrix} c(q - q^{-1}) + d & c \\ c & d \end{pmatrix}.
\]

\medskip

\noindent\textbf{Crossing coefficients.}
For knots meeting via connected sum, $J_{K_1 \# K_2} = J_{K_1} \cdot J_{K_2}$, giving $c = 1$ and $d = 0$.  For nucleon crossings (Hecke product), the coefficients are determined by the representation-theoretic structure of the crossing; the three nuclear types have specific $(c, d)$ values listed in Section~3.2.

\medskip

\noindent\textbf{Action on Hecke elements.}
\[
  \begin{pmatrix} a' \\ b' \end{pmatrix}
  = T(c,d) \cdot \begin{pmatrix} a \\ b \end{pmatrix},
\]
so that $a' = a\,c(q - q^{-1}) + a\,d + b\,c$ and $b' = a\,c + b\,d$.

\medskip

\noindent\textbf{Eigenvalues.}
\[
  \lambda_\pm
  = \frac{1}{2}\Bigl[c(q - q^{-1}) + 2d
    \pm \sqrt{\bigl[c(q - q^{-1})\bigr]^2 + 4c^2}\Bigr].
\]

\medskip

\noindent\textbf{Special cases.}
\begin{itemize}
  \item $c = 0$: scalar matrix $d\,\II$.
  \item $d = 0$: $c \cdot T_{pp}$.
  \item $c = 1,\; d = 0$: the fundamental matrix $T_{pp}$.
\end{itemize}

%% ============================================================
\section{Transfer Matrices of Knot Pairs}
%% ============================================================

\subsection{Trivial matrices}

\noindent\textbf{Unknot--unknot} ($T_{\nu\nu}$): $c = 1,\; d = 0$.
\[
  \boxed{T_{\nu\nu}(q)
  = \begin{pmatrix} q - q^{-1} & 1 \\ 1 & 0 \end{pmatrix}}
\]

\noindent\textbf{Identity (no crossing)}: $c = 0,\; d = 1$.
\[
  \boxed{T_\varnothing(q)
  = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = \II}
\]

\subsection{Nuclear transfer matrices}

\noindent\textbf{Proton--proton} ($T_{pp}$): $c = 1,\; d = 0$.
\[
  \boxed{T_{pp}(q)
  = \begin{pmatrix} q - q^{-1} & 1 \\ 1 & 0 \end{pmatrix}}
\]

\medskip

\begin{tabular}{@{} l c c c c c c @{}}
\toprule
\textbf{Entry} & $(1,1)$ & $(1,2)$ & $(2,1)$ & $(2,2)$ & $\det$ & $\operatorname{tr}$ \\
\midrule
$T_{pp}$ & $q - q^{-1}$ & $1$ & $1$ & $0$ & $-1$ & $q - q^{-1}$ \\
\bottomrule
\end{tabular}

\medskip

\noindent Eigenvalues:
$\lambda_\pm = \dfrac{(q - q^{-1}) \pm \sqrt{(q - q^{-1})^2 + 4}}{2}$,
\quad $\lambda_+\lambda_- = -1$.

\bigskip

\noindent\textbf{Neutron--neutron} ($T_{nn}$): $c = q^2,\; d = -qs$.
\[
  \boxed{T_{nn}(q)
  = \begin{pmatrix}
      q^2(q - q^{-1}) - qs & q^2 \\
      q^2 & -qs
    \end{pmatrix}}
\]

\medskip

\begin{tabular}{@{} l c c c c c @{}}
\toprule
\textbf{Entry} & $(1,1)$ & $(1,2)$ & $(2,1)$ & $(2,2)$ & $\det$ \\
\midrule
$T_{nn}$ & $q^3 - q - q^{3/2} + q^{1/2}$ & $q^2$ & $q^2$ & $-q^{3/2} + q^{1/2}$ & $cd(q\!-\!q^{-1})\!+\!d^2\!-\!c^2$ \\
\bottomrule
\end{tabular}

\medskip

\noindent Relation:
$T_{nn}(q) = q^2\, T_{pp}(q) - qs\,\II$.

\bigskip

\noindent\textbf{Mixed} ($T_{\mathrm{mix}}$):
$c = \dfrac{1 + q^2}{2},\; d = -\dfrac{qs}{2}$.
\[
  \boxed{T_{\mathrm{mix}}(q)
  = \begin{pmatrix}
      \dfrac{1+q^2}{2}(q - q^{-1}) - \dfrac{qs}{2}
        & \dfrac{1+q^2}{2} \\[6pt]
      \dfrac{1+q^2}{2}
        & -\dfrac{qs}{2}
    \end{pmatrix}}
\]

\noindent Midpoint relation:
$T_{\mathrm{mix}} = \dfrac{1}{2}\bigl(T_{pp} + T_{nn}\bigr)$.

\subsection{Electromagnetic matrices}

\begin{itemize}
  \item \textbf{Photon--electron}: $T_{\gamma e} = T_{pp}$
    \quad (unknot invisible).
  \item \textbf{Electron--electron}: $T_{ee} = T_{pp}$.
  \item \textbf{Photon--photon}: $T_{\gamma\gamma} = T_{pp}$.
  \item \textbf{Photon--positron}: $T_{\gamma e^+} = T_{pp}$.
  \item \textbf{Electron--positron}: $T_{e^-e^+} = T_{\mathrm{mix}}$.
\end{itemize}

\subsection{Higher-generation and universality}

For \emph{any} knot~$K$ paired with the unknot:
$T_{T_{1,0},K} = T_{pp}$ (the unknot is universally invisible).

\medskip

Same-chirality identical pairs all give~$T_{pp}$
(cinquefoil--cinquefoil, top--top, etc.), because the connected sum
satisfies $J_{K_1 \# K_2} = J_{K_1} \cdot J_{K_2}$, whence $c = 1$
and $d = 0$.

\subsection{Classification summary}

\begin{tabular}{@{} l c c c l @{}}
\toprule
\textbf{Class} & $c$ & $d$ & \textbf{Matrix} & \textbf{Occurs when} \\
\midrule
Type~I (fundamental)
  & $1$ & $0$ & $T_{pp}$ & Same-chirality or unknot \\[3pt]
Type~II (heavy)
  & $q^2$ & $-qs$ & $T_{nn}$ & Mirror-chirality (down--down) \\[3pt]
Type~III (mixed)
  & $\frac{1+q^2}{2}$ & $-\frac{qs}{2}$ & $T_{\mathrm{mix}}$ & Cross-chirality (up--down) \\[3pt]
\bottomrule
\end{tabular}

\bigskip

\noindent\textbf{Key identities.}
\begin{gather*}
  T_{\mathrm{mix}}
    = \tfrac{1}{2}\bigl(T_{pp} + T_{nn}\bigr), \qquad
  T_{nn} = q^2\, T_{pp} - qs\,\II, \\[4pt]
  T_{\gamma e} = T_{ee} = T_{\gamma\gamma}
    = T_{\nu\nu} = T_{\mathfrak{bb}} = T_{\mathfrak{tt}} = T_{pp}.
\end{gather*}

\section{Interaction and Decay Matrices}
%% ============================================================

The general change-of-crossing operator is
\[
  \Delta_{K_1 K_2 \to K_1' K_2'}
  = T_{K_1', K_2'} - T_{K_1, K_2}.
\]

\subsection*{Universal beta decay}

\[
  \boxed{\Delta_\beta(q)
  = \frac{1-q^2}{2}\,T_{pp}(q) + \frac{qs}{2}\,\II}
\]
Explicit entries:
\[
  \Delta_\beta(q)
  = \begin{pmatrix}
      \dfrac{1-q^2}{2}(q - q^{-1}) + \dfrac{qs}{2}
        & \dfrac{1-q^2}{2} \\[6pt]
      \dfrac{1-q^2}{2}
        & \dfrac{qs}{2}
    \end{pmatrix}.
\]
Properties: symmetric; vanishes at $q = 1$.

\subsection*{Reverse beta}

$\Delta_{\beta^+} = -\Delta_\beta$.

\subsection*{Photon absorption (R-I move)}

Appends a $T_{pp}$ factor:
$M_i^{(n+1)} = T_{pp} \cdot M_i^{(n)}$.
For hydrogen, energy $\propto \lambda_+^n$.

\subsection*{Pair creation}

$\Delta_{\mathrm{pair}} = T_{\mathrm{mix}} - T_{pp} = -\Delta_\beta$.

\subsection*{Double beta decay}

Quadratic term: $\Delta_\beta \cdot C \cdot \Delta_\beta$
(the nuclear matrix element), where $C$ depends on the intermediate
nucleus.

\subsection*{Toffoli gate}

Target-flip amplitude $P_{\mathrm{flip}} = c_{k_1 k_2} \cdot \Phi(q)$,
where
\[
  \boxed{\Phi(q)
  = \frac{(q-1)(q^{3/2} + q^{1/2} - 1)(q^{3/2} + q + 1)}{2(q+1)^2}}
\]
Toffoli matrix entries:
\begin{alignat*}{2}
  P_{6,6} &= \frac{q^4 + q^{7/2} + 2q^{3/2} + q^{1/2} - 1}{2(q+1)^2},
  &\qquad
  P_{7,6} &= \Phi(q), \\[4pt]
  P_{6,7} &= -\Phi(q),
  &\qquad
  P_{7,7} &= \frac{q + q^{1/2} - 1}{q + 1}.
\end{alignat*}

\subsection*{Interaction catalogue}

\begin{tabular}{@{} l c l @{}}
\toprule
\textbf{Process} & $\Delta$~\textbf{operator} & \textbf{Matrix} \\
\midrule
$\beta^-$
  & $\Delta_\beta$
  & $\frac{1-q^2}{2}\,T_{pp} + \frac{qs}{2}\,\II$ \\[4pt]
$\beta^+$
  & $-\Delta_\beta$
  & $\frac{q^2-1}{2}\,T_{pp} - \frac{qs}{2}\,\II$ \\[4pt]
Photon absorption
  & $T_{pp}$ (appended)
  & new factor \\[2pt]
Pair creation
  & $-\Delta_\beta$
  & same as $\beta^+$ \\[2pt]
Pair annihilation
  & $\Delta_\beta$
  & same as $\beta^-$ \\[2pt]
$2\beta$ (quadratic)
  & $\Delta_\beta\, C\, \Delta_\beta$
  & depends on $C$ \\[2pt]
\bottomrule
\end{tabular}

\section{Worked Examples}
%% ============================================================

\subsection*{Example 1: Hydrogen ($A=1,\; Z=1$)}

Zero generators, zero multiplications.
$P_{{}^1\mathrm{H}} = 1$.

\subsection*{Example 2: Deuterium ($A=2,\; Z=1,\; N=1$)}

Ordering: $[\mathfrak{p},\, \mathfrak{n}]$.
One generator~$\sigma_0$, one crossing (mixed).

\[
  M_0 = T_{\mathrm{mix}}.
\]
\[
  \hat{V}_0
  = z \cdot (T_{\mathrm{mix}})_{11} + (T_{\mathrm{mix}})_{21}.
\]
\[
  P_{{}^2\mathrm{H}} = \hat{V}_0.
\]
$\binom{2}{2} = 1$ multiplication.

\subsection*{Example 3: Helium-3 ($A=3,\; Z=2,\; N=1$)}

Ordering: $[\mathfrak{p}_0,\, \mathfrak{p}_1,\, \mathfrak{n}_2]$.
Two generators.

\medskip

\noindent\textbf{Generator $\sigma_0$}: 2 crossings ---
$(0,1)$ pp $\to T_{pp}$, \;$(0,2)$ mix $\to T_{\mathrm{mix}}$.
\[
  M_0 = T_{\mathrm{mix}} \cdot T_{pp}
  = \begin{pmatrix}
      \frac{1+q^2}{2}(q-q^{-1}) - \frac{qs}{2} & \frac{1+q^2}{2} \\[4pt]
      \frac{1+q^2}{2} & -\frac{qs}{2}
    \end{pmatrix}
  \begin{pmatrix}
      q-q^{-1} & 1 \\ 1 & 0
    \end{pmatrix}.
\]

\noindent\textbf{Generator $\sigma_1$}: 1 crossing ---
$(1,2)$ mix $\to T_{\mathrm{mix}}$.
\[
  M_1 = T_{\mathrm{mix}}.
\]
\[
  P_{{}^3\mathrm{He}} = \hat{V}_0 \cdot \hat{V}_1.
\]
3 multiplications.

\subsection*{Example 4: Helium-4 ($A=4,\; Z=2,\; N=2$)}

Ordering: $[\mathfrak{p}_0,\, \mathfrak{p}_1,\, \mathfrak{n}_2,\, \mathfrak{n}_3]$.
Three generators.

\medskip

\noindent\textbf{Generator $\sigma_0$}:
$(0,1)$ pp, $(0,2)$ mix, $(0,3)$ mix.
\[
  M_0 = T_{\mathrm{mix}} \cdot T_{\mathrm{mix}} \cdot T_{pp}.
\]

\noindent\textbf{Generator $\sigma_1$}:
$(1,2)$ mix, $(1,3)$ mix.
\[
  M_1 = T_{\mathrm{mix}}^{\,2}.
\]

\noindent\textbf{Generator $\sigma_2$}:
$(2,3)$ nn.
\[
  M_2 = T_{nn}.
\]
\[
  P_{{}^4\mathrm{He}}
  = \hat{V}_0 \cdot \hat{V}_1 \cdot \hat{V}_2.
\]
6 multiplications.

\subsection*{Example 5: Tritium beta decay}

\[
  {}^3\mathrm{H}\,(\mathfrak{p},\mathfrak{n},\mathfrak{n})
  \;\longrightarrow\;
  {}^3\mathrm{He}\,(\mathfrak{p},\mathfrak{p},\mathfrak{n}).
\]
Converting position~1: $\mathfrak{n} \to \mathfrak{p}$.

\medskip

\noindent\textbf{Parent} (${}^3\mathrm{H}$):
\[
  M_0^{\mathrm{par}} = T_{\mathrm{mix}}^{\,2}, \qquad
  M_1^{\mathrm{par}} = T_{nn}.
\]

\noindent\textbf{Daughter} (${}^3\mathrm{He}$):
\[
  M_0^{\mathrm{dau}} = T_{\mathrm{mix}} \cdot T_{pp}, \qquad
  M_1^{\mathrm{dau}} = T_{\mathrm{mix}}.
\]

\noindent\textbf{Crossing-type changes.}

\medskip

\begin{tabular}{@{} c c c c @{}}
\toprule
\textbf{Generator} & \textbf{Pair} & \textbf{Parent type} & \textbf{Daughter type} \\
\midrule
$\sigma_0$ & $(0,1)$ & mix & pp \\
$\sigma_0$ & $(0,2)$ & mix & mix \\
$\sigma_1$ & $(1,2)$ & nn & mix \\
\bottomrule
\end{tabular}

\medskip

\noindent Matrix differences:
\[
  \Delta M_0 = T_{\mathrm{mix}} \cdot \Delta_\beta, \qquad
  \Delta M_1 = \Delta_\beta.
\]
The log-ratio of Skein polynomials:
\[
  \ln\!\Bigl(\frac{P_{\mathrm{dau}}}{P_{\mathrm{par}}}\Bigr)
  = \sum_i \ln\!\Bigl(\frac{\hat{V}_i^{\,\mathrm{dau}}}
                            {\hat{V}_i^{\,\mathrm{par}}}\Bigr).
\]
