\def\AxesThreeD{\@ifnextchar[{\AxesThreeD@i}{\AxesThreeD@i[]}}
\def\AxesThreeD@i[#1](#2){{%
	\setkeys{psset}{#1}%
	\AxesThreeD@ii#2,@comma*\relax%
}}
\def\AxesThreeD@ii#1,#2*{%
	\def\@tempa{#2}\def\@tempb{@comma}%
	\ifx\@tempa\@tempb% only one value
		\AxesThreeD@iii#1,#2\else\AxesThreeD@iv#1,#2\relax%
	\fi%
}
\def\AxesThreeD@iii#1,#2\relax{%
	\AxesThreeD@iv#1,#1,#1,#1\relax%
}
\def\AxesThreeD@iv#1,#2,#3,#4\relax{{%
	\pNodeThreeD(0,0,0){O}%
	\pNodeThreeD(0,0,#3){Z}\pNodeThreeD(#1,0,0){X}\pNodeThreeD(0,#2,0){Y}%
	\psline{->}(O)(Z)\psline{->}(O)(X)\psline{->}(O)(Y)%
	\uput[0](Z){\small Z}\uput[90](X){\small X}\uput[0](Y){\small Y}%
}}

\def\showCoorThreeD{\@ifnextchar[{\showCoorThreeD@i}{\pshowCoorThreeD@i[]}}
\def\showCoorThreeD@i[#1](#2){{%
	\psset{linestyle=dashed}
	\setkeys{psset}{#1}
	\showCoorThreeD@ii#2,@comma*\relax
}}
\def\showCoorThreeD@ii#1,#2*{%
	\def\@tempa{#2}\def\@tempb{@comma}%
	\ifx\@tempa\@tempb%
		\showCoorThreeD@iii#1;#2\else\showCoorThreeD@iv#1,#2\relax%
	\fi%
}
\def\showCoorThreeD@iii#1;#2;#3;#4\relax{%
	\psline(!
		\variablesTroisD
		/Radius #1 def
		/longitude #2 def
		/latitude #3 def
		/Zcote Radius latitude sin mul def
		/Xabscisse Radius longitude cos mul latitude cos mul def
		/Yordonnee Radius longitude sin mul latitude cos mul def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)(!
		/Zcote 0 def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)
	\psline(!
		/Zcote 0 def
		/Xabscisse Radius longitude cos mul latitude cos mul def
		/Yordonnee Radius longitude sin mul latitude cos mul def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)(!
		/Zcote 0 def
		/Yordonnee 0 def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)
	\psline(!
		/Zcote 0 def
		/Xabscisse Radius longitude cos mul latitude cos mul def
		/Yordonnee Radius longitude sin mul latitude cos mul def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)(!
		/Zcote 0 def
		/Xabscisse 0 def
		tx@3DDict begin
			formulesTroisD
			Xi reduction mul Yi reduction mul
		end)
}
