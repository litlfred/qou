\newcommand{\drawS}[3]{%
	\filldraw[fill=white,thick] (#1,#2) ellipse (3mm and 3mm);
	\node at (#1,#2) {\Large $S$};
	\path(#1,#2) ++(#3:0.37) node {$\star$};
}
