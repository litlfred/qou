    noncomputable def QOU.Knot.clausen (θ : ℝ) :
    ℝ

    The Clausen function Cl₂(θ) = ∫₀^θ -log(2 sin(t/2)) dt (Lobachevsky primitive of -log(2 sin(·/2))). Not available in Mathlib.
