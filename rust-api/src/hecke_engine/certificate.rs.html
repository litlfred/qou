1//! Witness certificate: the canonical output format for Pauli witnesses.
2//!
3//! Each certificate records:
4//! - The isotope (Z, N, A)
5//! - The engine version and git commit that produced it
6//! - The 4 summary polynomials (tr_alt, net, observable, coral)
7//!   as sparse Laurent polynomial coefficients
8//! - Per-strand metadata (word counts, timing)
9//! - F_Pauli evaluated at q₀ (when computable)
10//!
11//! Format: JSON (human-readable) + optional binary (compact).
12//! The certificate is the input for:
13//! - Lean witness verification (strand-by-strand replay)
14//! - Δ-chain computation (difference between parent and daughter)
15//! - F_Pauli evaluation at arbitrary precision
16
17use serde::Serialize;
18use std::collections::BTreeMap;
19
20/// Sparse Laurent polynomial: list of (exponent, coefficient_string) pairs.
21/// Coefficients are stored as decimal strings to preserve BigInt precision.
22#[derive(Serialize, Clone)]
23pub struct SparsePoly {
24    /// Sorted (exponent, coefficient) pairs. Coefficient is decimal string.
25    pub terms: Vec<(i32, String)>,
26}
27
28impl SparsePoly {
29    pub fn new() -> Self { Self { terms: Vec::new() } }
30
31    pub fn from_btree(map: &BTreeMap<i32, impl ToString>) -> Self {
32        Self {
33            terms: map.iter()
34                .filter(|(_, v)| v.to_string() != "0")
35                .map(|(k, v)| (*k, v.to_string()))
36                .collect()
37        }
38    }
39
40    pub fn n_terms(&self) -> usize { self.terms.len() }
41
42    pub fn max_abs_coeff_log10(&self) -> f64 {
43        self.terms.iter()
44            .map(|(_, s)| {
45                let s = s.trim_start_matches('-');
46                if s.is_empty() || s == "0" { return 0.0; }
47                s.len() as f64 - 1.0
48            })
49            .fold(0.0f64, f64::max)
50    }
51}

53/// Per-strand metadata for verification.
54#[derive(Serialize, Clone)]
55pub struct StrandInfo {
56    pub strand: usize,
57    pub words_before_strip: usize,
58    pub words_after_strip: usize,
59    pub mid_strips: u64,
60    pub elapsed_ms: u128,
61    pub max_coeff_log10: f64,
62}

64/// The complete witness certificate.
65#[derive(Serialize)]
66pub struct WitnessCertificate {
67    // Identification
68    pub isotope: IsotopeInfo,
69    pub engine: EngineInfo,
70
71    // The 4 summary polynomials (exact in t = q^{1/2} over ℤ)
72    pub tr_alt: SparsePoly,
73    pub net: SparsePoly,
74    pub observable: SparsePoly,
75    pub coral: SparsePoly,
76
77    // Jet bundle: net_by_len[ℓ] = Σ c_w for words of length ℓ
78    pub jets: Vec<SparsePoly>,
79
80    // Denominator: true value = poly(t) / 2^pn_count
81    pub pn_count: u32,
82
83    // Evaluation at q₀ (if computable without overflow)
84    pub f_pauli_f64: Option<f64>,
85    pub tr_alt_f64: Option<f64>,
86    pub net_f64: Option<f64>,
87    pub tr_markov_f64: Option<f64>,        // Markov trace of full NF at q₀
88    pub tr_markov_cleared: SparsePoly,     // tr_M × (t+t⁻¹)^L — symbolic in t
89    pub tr_markov_denom_power: usize,      // L: divide by (t+t⁻¹)^L to get tr_M
90
91    // Per-strand build metadata
92    pub strands: Vec<StrandInfo>,
93
94    // Summary
95    pub total_words: usize,
96    pub max_coeff_log10: f64,
97    pub elapsed_seconds: f64,
98    pub computed_at: String,
99}

115    pub commit: String,

117}

119impl EngineInfo {
120    pub fn current(binary: &str) -> Self {
121        Self {
122            name: "hecke-engine".to_string(),
123            version: env!("CARGO_PKG_VERSION").to_string(),
124            commit: option_env!("GIT_SHA").unwrap_or("unknown").to_string(),
125            binary: binary.to_string(),
126        }
127    }
