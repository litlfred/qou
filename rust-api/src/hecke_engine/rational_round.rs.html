1//! §S9.1 — Local Stern–Brocot continued-fraction rationalisation.
2//!
3//! Given an `f64` and a denominator bound `q_max`, produce the
4//! best rational approximation `p/q` with `|q| ≤ q_max`.  Algorithm:
5//! the standard semi-convergent / Stern-Brocot continued-fraction
6//! recursion.
7//!
8//! Exists as a hecke-engine-local replacement for the upstream
9//! `clarabel::algebra::tighten_scalar` because that helper sits
10//! behind the `clarabel-bigrational` feature flag, which is
11//! mutually exclusive with `clarabel-sdp` (see
12//! [`docs/audits/exact-sdp-feasibility.md`](../../docs/audits/exact-sdp-feasibility.md)).
13//! §S9 needs to round f64 SDP solutions to rationals, so the
14//! tightening must live on the SDP side of the mutex.
15//!
16//! Per workplan
17//! [`CLARABEL_PRECISION_PLAN.md`](../CLARABEL_PRECISION_PLAN.md) §S9.1.
18
19use num_bigint::BigInt;
20use num_traits::{One, Zero};
21
22/// Round an `f64` to the closest rational `p/q` with `|q| ≤ q_max`,
23/// returned as a `(num, den)` `BigInt` pair with `den > 0`.
24///
25/// Algorithm: continued-fraction expansion of `|x|` truncated to
26/// `q_max`, then sign reattached.  Returns `(0, 1)` for `x = 0`,
27/// panics on non-finite input.
28pub fn tighten_f64(x: f64, q_max: u64) -> (BigInt, BigInt) {
29    if !x.is_finite() {
30        panic!("tighten_f64: non-finite input ({x})");
31    }
32    if x == 0.0 {
33        return (BigInt::zero(), BigInt::one());
34    }
35    let sign = if x < 0.0 { -1 } else { 1 };
36    let x_orig = x;
37    let mut x = x.abs();
38
39    // Convergents (h_{k-2}, h_{k-1}) → h_k = a_k h_{k-1} + h_{k-2}
40    // and similarly for k_k.  Stop when k_k > q_max; back off to
41    // the previous convergent or a semi-convergent that's still
42    // within bound.
43    let mut h_prev2: BigInt = BigInt::zero();
44    let mut h_prev: BigInt = BigInt::one();
45    let mut k_prev2: BigInt = BigInt::one();
46    let mut k_prev: BigInt = BigInt::zero();
47    let q_max_bi = BigInt::from(q_max);
48
49    // 30 iterations is enough for any f64 (the partial-quotient
50    // sequence of an f64 has length bounded by O(log denom)).
51    for _ in 0..40 {
52        let a = x.trunc();
53        if a < 0.0 {
54            break;
55        }
56        let a_bi = BigInt::from(a as u64);
57        let h_new: BigInt = &a_bi * &h_prev + &h_prev2;
58        let k_new: BigInt = &a_bi * &k_prev + &k_prev2;
59        if k_new > q_max_bi {
60            // Out-of-budget: candidate convergents are
61            //   (h_prev / k_prev)        — the previous full convergent
62            //   (h_semi / k_semi)        — best semi-convergent within q_max
63            // Pick whichever is closer to the original `x`.  This
64            // matches the standard "best rational approximation under
65            // a denominator bound" definition (see Khinchin 1964 §16).
66            let a_max_num: BigInt = &q_max_bi - &k_prev2;
67            let a_max = if k_prev > BigInt::zero() {
68                &a_max_num / &k_prev
69            } else {
70                BigInt::zero()
71            };
72            // Always-available candidate: the prior convergent.
73            let mut best_h = h_prev.clone();
74            let mut best_k = k_prev.clone();
75            // Compute its approximation error vs original |x|.
76            let orig = x_orig.abs();
77            let err_prev = approx_error(&best_h, &best_k, orig);
78            let mut best_err = err_prev;
79            if a_max > BigInt::zero() && a_max < a_bi {
80                let h_semi: BigInt = &a_max * &h_prev + &h_prev2;
81                let k_semi: BigInt = &a_max * &k_prev + &k_prev2;
82                if k_semi > BigInt::zero() && k_semi <= q_max_bi {
83                    let err_semi = approx_error(&h_semi, &k_semi, orig);
84                    if err_semi < best_err {
85                        best_err = err_semi;
86                        best_h = h_semi;
87                        best_k = k_semi;
88                    }
89                }
90            }
91            let sign_bi = if sign < 0 {
92                -BigInt::one()
93            } else {
94                BigInt::one()
95            };
96            return (sign_bi * best_h, best_k);
97        }
98        h_prev2 = h_prev;
99        h_prev = h_new;
100        k_prev2 = k_prev;
101        k_prev = k_new;
102        let frac = x - a;
103        if frac.abs() < f64::EPSILON {
104            break;
105        }
106        x = 1.0 / frac;
107    }
108
109    let sign_bi = if sign < 0 {
110        -BigInt::one()
111    } else {
112        BigInt::one()
113    };
114    (sign_bi * h_prev, k_prev)
115}

117/// Internal: |x − num/den| as f64 (sufficient for the candidate
118/// comparison since both candidates are within Stern-Brocot's
119/// guaranteed envelope of f64's representable rationals).
120fn approx_error(num: &BigInt, den: &BigInt, x_abs: f64) -> f64 {
121    let p = rational_to_f64(num, den);
122    (x_abs - p).abs()
123}

125/// Element-wise [`tighten_f64`] over an `f64` slice.
126pub fn tighten_vec(xs: &[f64], q_max: u64) -> Vec<(BigInt, BigInt)> {
127    xs.iter().map(|&x| tighten_f64(x, q_max)).collect()
128}
129
130/// Project a tightened `(num, den)` rational back to `f64`.
131/// Lossy by construction; used for displaying / cross-checking.
132pub fn rational_to_f64(num: &BigInt, den: &BigInt) -> f64 {
133    use num_traits::ToPrimitive;
134    let n = num.to_f64().unwrap_or(f64::NAN);
135    let d = den.to_f64().unwrap_or(f64::NAN);
136    n / d
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142
143    #[test]
144    fn tighten_zero_is_zero_over_one() {
145        let (n, d) = tighten_f64(0.0, 100);
146        assert_eq!(n, BigInt::zero());
147        assert_eq!(d, BigInt::one());
148    }
149
150    #[test]
151    fn tighten_half_is_one_over_two() {
152        let (n, d) = tighten_f64(0.5, 100);
153        assert_eq!(n, BigInt::from(1));
154        assert_eq!(d, BigInt::from(2));
155    }
156
157    #[test]
158    fn tighten_one_third_is_one_over_three() {
159        let (n, d) = tighten_f64(1.0_f64 / 3.0_f64, 100);
160        assert_eq!(n, BigInt::from(1));
161        assert_eq!(d, BigInt::from(3));
162    }
163
164    #[test]
165    fn tighten_negative_two_thirds() {
166        let (n, d) = tighten_f64(-2.0_f64 / 3.0_f64, 100);
167        assert_eq!(n, BigInt::from(-2));
168        assert_eq!(d, BigInt::from(3));
169    }
170
171    #[test]
172    fn tighten_pi_at_denom_113_picks_355_113() {
173        // Best rational approximation of π with denom ≤ 113 is the
174        // Milü 355/113 (Zu Chongzhi).
175        let (n, d) = tighten_f64(std::f64::consts::PI, 113);
176        assert_eq!(n, BigInt::from(355));
177        assert_eq!(d, BigInt::from(113));
178    }
179
180    #[test]
181    fn tighten_qou_q0_at_denom_10000() {
182        // q_0 ≈ 1.10997859555418057.  Stern-Brocot at denom ≤ 10_000
183        // should give a nice small-denom approximation.
184        let q0 = 1.10997859555418057_f64;
185        let (n, d) = tighten_f64(q0, 10_000);
186        // At denom ≤ 10_000 we expect a good approximation; sanity
187        // check the round-trip drift is small.
188        let drift = (rational_to_f64(&n, &d) - q0).abs();
189        assert!(
190            drift < 1e-7,
191            "drift too large: {drift} (got {n}/{d})"
192        );
193        // Denominator bound respected
194        assert!(
195            d.to_string().parse::<u64>().unwrap() <= 10_000,
196            "denominator exceeded bound: {d}"
197        );
198    }
199
200    #[test]
201    fn tighten_vec_preserves_signs() {
202        let xs = [0.5, -0.5, 1.0, -2.0];
203        let qs = tighten_vec(&xs, 100);
204        for (x, (n, d)) in xs.iter().zip(qs.iter()) {
205            assert_eq!(
206                x.signum() * (rational_to_f64(n, d).abs()),
207                *x,
208                "sign mismatch at x={x}"
209            );
210        }
211    }
