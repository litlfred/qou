1//! Cross-level embedding consistency check (R4 of RUST_INTEGRATION.md).
2//!
3//! For a tower edge `parent ← {constituent_1, ..., constituent_k}`
4//! the canonical embedding
5//!
6//!     π : H_{n_1}(q_0) ⊗ ... ⊗ H_{n_k}(q_0)  ↪  H_{n_parent}(q_0)
7//!
8//! must send T_w(constituents) to T_w(parent) inside the relevant
9//! cosets.  At the level of Specht modules, restriction
10//!
11//!     S^λ |_{S_{n_1} × ... × S_{n_k}}  =
12//!         ⊕_{μ_1 ⊢ n_1, ..., μ_k ⊢ n_k}  c^λ_{μ_1, ..., μ_k}
13//!                                        · S^{μ_1} ⊠ ... ⊠ S^{μ_k}
14//!
15//! with multi-Littlewood-Richardson multiplicities.  This module
16//! computes those multiplicities (via pairwise composition of
17//! `lr_coefficient`) and emits the cross-level diagnostic.
18//!
19//! For Phase R4 PoC we check:
20//!   (a) embedding-dimension consistency: n_parent = Σ_i n_i
21//!   (b) per-partition restriction: d_λ = Σ_{μ-tuple} c^λ_{μ-tuple} · Π d_{μ_i}
22//!
23//! Trace-identity verification (check (c) in the design spec) requires
24//! already-evaluated parent + constituent Wedderburn data and is left
25//! as a separate `verify_trace_identity` function (called when
26//! evaluator output for parent + constituents is available).
27
28use crate::joint_tower_sdp_certificate::{CrossLevelEdge, JointTowerSdpCertificate};
29use crate::littlewood_richardson::lr_coefficient;
30use crate::seminormal::{partitions_of, standard_young_tableaux};

32/// Specification for a tower edge.
33pub struct TowerEdgeSpec<'a> {
34    pub parent_name: &'a str,
35    pub parent_n: usize,
36    pub constituents: Vec<(&'a str, usize)>,
37}
38
39/// Multi-LR coefficient c^λ_{μ_1, ..., μ_k}.
40///
41/// Computed by left-associative composition of pairwise LR:
42///   c^λ_{μ_1, μ_2, ..., μ_k}  =
43///       Σ_{ν_2}  c^λ_{ν_2, μ_k}  ·  c^{ν_2}_{μ_1, ..., μ_{k-1}}
44///
45/// recursive folding from k = 2 outward.
46pub fn multi_lr(lambda: &[usize], mus: &[Vec<usize>]) -> i64 {
47    if mus.is_empty() {
48        return if lambda.iter().all(|&x| x == 0) || lambda.is_empty() {
49            1
50        } else {
51            0
52        };
53    }
54    if mus.len() == 1 {
55        return if lambda == mus[0].as_slice() { 1 } else { 0 };
56    }
57    if mus.len() == 2 {
58        return lr_coefficient(lambda, &mus[0], &mus[1]);
59    }
60    // Fold: combine first k-1 into all possible ν, then LR(ν, μ_k).
61    let n_first: usize = mus[..mus.len() - 1].iter().flatten().sum();
62    let last_mu = mus.last().unwrap();
63    let mut total = 0i64;
64    for nu in partitions_of(n_first) {
65        let head_count = multi_lr(&nu, &mus[..mus.len() - 1].to_vec());
66        if head_count == 0 {
67            continue;
68        }
69        let lr = lr_coefficient(lambda, &nu, last_mu);
70        total += head_count * lr;
71    }
72    total
73}
74
75/// Dimension of S^λ = number of standard Young tableaux of shape λ.
76fn d_lambda(shape: &[usize]) -> usize {
77    if shape.is_empty() {
78        return 1;
79    }
80    standard_young_tableaux(shape).len()
81}

83/// Enumerate all tuples of partitions (μ_1, ..., μ_k) with μ_i ⊢ n_i.
84fn partition_tuples(ns: &[usize]) -> Vec<Vec<Vec<usize>>> {
85    if ns.is_empty() {
86        return vec![Vec::new()];
87    }
88    let head = partitions_of(ns[0]);
89    let tails = partition_tuples(&ns[1..]);
90    let mut out = Vec::new();
91    for h in head {
92        for t in &tails {
93            let mut combined = vec![h.clone()];
94            combined.extend(t.iter().cloned());
95            out.push(combined);
96        }
97    }
98    out
99}

101/// Verify cross-level embedding consistency for a tower edge.
102///
103/// Performs checks (a) and (b); returns one CrossLevelEdge per
104/// parent partition λ summarizing the LR-decomposition.
105pub fn verify_edge(spec: &TowerEdgeSpec) -> Vec<CrossLevelEdge> {
106    let mut out = Vec::new();
107    let n_parent = spec.parent_n;
108    let n_constituents: usize = spec.constituents.iter().map(|(_, n)| *n).sum();
109    // Check (a)
110    let dim_match = n_parent == n_constituents;
111    if !dim_match {
112        out.push(CrossLevelEdge {
113            parent: spec.parent_name.to_string(),
114            constituent: format!(
115                "{:?}",
116                spec.constituents.iter().map(|(s, _)| *s).collect::<Vec<_>>()
117            ),
118            embedding_codomain_n_0: n_parent,
119            embedding_domain_n_0: n_constituents,
120            embedding_consistent: false,
121            diagnostic: format!(
122                "(a) DIM MISMATCH: n_parent = {}, Σn_i = {}",
123                n_parent, n_constituents
124            ),
125        });
126        return out;
127    }
128
129    let constituent_ns: Vec<usize> = spec.constituents.iter().map(|(_, n)| *n).collect();
130    // Check (b): per-partition LR sum
131    for lambda in partitions_of(n_parent) {
132        let dim_lambda = d_lambda(&lambda);
133        let mut lr_sum: i64 = 0;
134        let mut tuple_terms: Vec<String> = Vec::new();
135        for tuple in partition_tuples(&constituent_ns) {
136            let c = multi_lr(&lambda, &tuple);
137            if c == 0 {
138                continue;
139            }
140            let prod_d: usize = tuple.iter().map(|m| d_lambda(m)).product();
141            lr_sum += c * (prod_d as i64);
142            tuple_terms.push(format!("{}·{:?}", c, tuple));
143        }
144        let consistent = lr_sum == dim_lambda as i64;
145        let diagnostic = format!(
146            "λ={:?}, d_λ={}, Σ c·Πd_μ = {}  [{}]",
147            lambda,
148            dim_lambda,
149            lr_sum,
150            tuple_terms.join(" + ")
151        );
152        out.push(CrossLevelEdge {
153            parent: spec.parent_name.to_string(),
154            constituent: format!(
155                "{:?}",
156                spec.constituents.iter().map(|(s, _)| *s).collect::<Vec<_>>()
157            ),
158            embedding_codomain_n_0: n_parent,
159            embedding_domain_n_0: n_constituents,
160            embedding_consistent: consistent,
161            diagnostic,
162        });
163    }
164    out
165}

184#[cfg(test)]
185mod tests {
186    use super::*;
187
188    #[test]
189    fn nucleon_from_3_quarks() {
190        // n_parent = 3, constituents = 3 × (n=1).
191        // Each quark partition is just (1) with d=1.
192        // Multi-LR c^λ_{(1)(1)(1)} = number of SYT of shape λ.
193        let edges = verify_edge(&TowerEdgeSpec {
194            parent_name: "nucleon",
195            parent_n: 3,
196            constituents: vec![("quark", 1), ("quark", 1), ("quark", 1)],
197        });
198        // Expect 3 edges (one per partition of 3): (3), (2,1), (1,1,1)
199        assert_eq!(edges.len(), 3);
200        for e in &edges {
201            assert!(
202                e.embedding_consistent,
203                "edge {:?} inconsistent: {}",
204                e.parent, e.diagnostic
205            );
206        }
207    }

209    #[test]
210    fn helium4_from_4_nucleons() {
211        // Tower edge ⁴He ← 4⊗nucleon: each nucleon has n=3, parent n=12.
212        // Partitions of 12 are numerous (77); LR multinomial check
213        // verifies dim(S^λ) = sum over (μ_1, μ_2, μ_3, μ_4) of
214        // products of c^λ_{...} · Π d_{μ_i}.
215        let edges = verify_edge(&TowerEdgeSpec {
216            parent_name: "4He",
217            parent_n: 12,
218            constituents: vec![
219                ("nucleon", 3),
220                ("nucleon", 3),
221                ("nucleon", 3),
222                ("nucleon", 3),
223            ],
224        });
225        assert_eq!(edges.len(), partitions_of(12).len()); // 77
226        // All consistent (LR identity is a theorem).
227        for e in &edges {
228            assert!(
229                e.embedding_consistent,
230                "edge for {:?} inconsistent: {}",
231                e.parent, e.diagnostic
232            );
233        }
234    }
