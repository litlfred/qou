1//! F3.2.γ scaffolding: Geck-Pfeiffer minimal-length reduction.
2//!
3//! For the Iwahori-Hecke algebra `H_n(q)`, every element `T_β` arising
4//! from a positive braid word has an expansion in the canonical basis
5//! `{T_w : w ∈ S_n}`:
6//!
7//!   `T_β = Σ_{w ∈ S_n} c_w(q) · T_w`
8//!
9//! where the coefficients `c_w(q)` are Laurent polynomials in q.  This
10//! module computes that expansion via the standard Hecke multiplication
11//! rules, exposing it for use by q-MN character evaluation and the
12//! discovery cache.
13//!
14//! ## Multiplication rules (Bourbaki LIE IV §2)
15//!
16//! For a simple reflection `s_i` and `w ∈ S_n`:
17//!
18//!   T_{s_i} · T_w = T_{s_i w}                 if ℓ(s_i w) > ℓ(w)
19//!   T_{s_i} · T_w = T_{s_i w} + (q − q^{-1}) · T_w   if ℓ(s_i w) < ℓ(w)
20//!
21//! Equivalently, for the right action:
22//!
23//!   T_w · T_{s_i} = T_{w s_i}                 if ℓ(w s_i) > ℓ(w)
24//!   T_w · T_{s_i} = T_{w s_i} + (q − q^{-1}) · T_w   if ℓ(w s_i) < ℓ(w)
25//!
26//! For inverse generators we use the Hecke relation `T_{s_i}^{-1} =
27//! T_{s_i} − (q − q^{-1})`.
28//!
29//! ## Status
30//!
31//! This is **scaffolding** for F3.2.γ.  The full implementation
32//! requires:
33//!
34//!   1. ✅ Permutation type (cycle representation or one-line form)
35//!   2. ✅ Length function ℓ(w) = #inversions(w)
36//!   3. ✅ Hecke basis multiplication primitives (Tw_mul_Ts)
37//!   4. ✅ Word-to-Hecke expansion: T_β = Σ c_w · T_w
38//!   5. 🚧 Memoization across multiplications
39//!   6. 🚧 PyO3 export
40//!   7. 🚧 Cross-validation with seminormal at q_0 on full atomic braids
41//!
42//! Cost: a single `T_β` expansion has worst-case O(n! · word_len)
43//! coefficients but typical (sparse) braid words give smaller spans.
44

48/// Permutation of `n` elements as a one-line representation (i.e.
49/// `perm[i] = w(i)`, 0-indexed).
50#[derive(Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
51pub struct Perm(pub Vec<usize>);
52
53impl Perm {
54    pub fn identity(n: usize) -> Self {
55        Self((0..n).collect())
56    }
57
58    pub fn n(&self) -> usize {
59        self.0.len()
60    }
61
62    /// Apply the simple transposition `s_i` (swap positions i, i+1)
63    /// on the right: `w · s_i`.  This is the same as swapping
64    /// `perm[i]` and `perm[i+1]`.
65    pub fn apply_s_right(&mut self, i: usize) {
66        if i + 1 < self.0.len() {
67            self.0.swap(i, i + 1);
68        }
69    }
70
71    /// Apply `s_i` on the left: `s_i · w`.  This swaps the positions
72    /// where the elements `i` and `i+1` appear in the one-line form.
73    pub fn apply_s_left(&mut self, i: usize) {
74        // Find positions of i and i+1 in self.0 and swap their values.
75        let mut pos_i = None;
76        let mut pos_i1 = None;
77        for (k, &v) in self.0.iter().enumerate() {
78            if v == i { pos_i = Some(k); }
79            if v == i + 1 { pos_i1 = Some(k); }
80        }
81        if let (Some(a), Some(b)) = (pos_i, pos_i1) {
82            self.0.swap(a, b);
83        }
84    }
85
86    /// Number of inversions = ℓ(w).
87    pub fn length(&self) -> usize {
88        let mut count = 0;
89        for i in 0..self.0.len() {
90            for j in (i + 1)..self.0.len() {
91                if self.0[i] > self.0[j] {
92                    count += 1;
93                }
94            }
95        }
96        count
97    }
98}

100/// Element of `H_n(q)` as a sparse map `w → c_w(q)` in the T_w basis.
101pub type HeckeElement = BTreeMap<Perm, LaurentPolyQ>;

103/// Identity in the Hecke basis: `1 = T_e`.
104pub fn hecke_identity(n: usize) -> HeckeElement {
105    let mut m = BTreeMap::new();
106    m.insert(Perm::identity(n), LaurentPolyQ::one());
107    m
108}

110/// Right-multiply a Hecke element `x` by `T_{s_i}` (positive simple
111/// generator).  Returns `x · T_{s_i}`.
112pub fn hecke_mul_ts_right(x: &HeckeElement, i: usize) -> HeckeElement {
113    let h = LaurentPolyQ::hecke_h();  // q − q^{-1}
114    let mut result: HeckeElement = BTreeMap::new();
115    for (w, c) in x {
116        let l_w = w.length();
117        let mut w_si = w.clone();
118        w_si.apply_s_right(i);
119        let l_w_si = w_si.length();
120        if l_w_si > l_w {
121            // T_w · T_{s_i} = T_{w s_i}.
122            let entry = result.entry(w_si).or_insert_with(LaurentPolyQ::zero);
123            *entry += c;
124        } else {
125            // T_w · T_{s_i} = T_{w s_i} + (q − q^{-1}) · T_w.
126            let entry = result.entry(w_si).or_insert_with(LaurentPolyQ::zero);
127            *entry += c;
128            let h_c = h.clone() * c.clone();
129            let entry2 = result.entry(w.clone()).or_insert_with(LaurentPolyQ::zero);
130            *entry2 += &h_c;
131        }
132    }
133    // Strip zero entries.
134    result.retain(|_, v| !v.is_zero());
135    result
136}

138/// **Left-multiply** a Hecke element `x` by `T_{s_i}` — returns
139/// `T_{s_i} · x`.
140///
141///   T_{s_i} · T_w = T_{s_i w}                 if ℓ(s_i w) > ℓ(w)
142///   T_{s_i} · T_w = T_{s_i w} + (q − q⁻¹) · T_w   if ℓ(s_i w) < ℓ(w)
143pub fn hecke_mul_ts_left(x: &HeckeElement, i: usize) -> HeckeElement {
144    let h = LaurentPolyQ::hecke_h();
145    let mut result: HeckeElement = BTreeMap::new();
146    for (w, c) in x {
147        let l_w = w.length();
148        let mut s_w = w.clone();
149        s_w.apply_s_left(i);
150        let l_s_w = s_w.length();
151        if l_s_w > l_w {
152            let entry = result.entry(s_w).or_insert_with(LaurentPolyQ::zero);
153            *entry += c;
154        } else {
155            let entry = result.entry(s_w).or_insert_with(LaurentPolyQ::zero);
156            *entry += c;
157            let h_c = h.clone() * c.clone();
158            let entry2 = result.entry(w.clone()).or_insert_with(LaurentPolyQ::zero);
159            *entry2 += &h_c;
160        }
161    }
162    result.retain(|_, v| !v.is_zero());
163    result
164}

166/// **Left-multiply** by `T_{s_i}^{-1} = T_{s_i} − (q − q⁻¹)`.
167pub fn hecke_mul_ts_inv_left(x: &HeckeElement, i: usize) -> HeckeElement {
168    let h = LaurentPolyQ::hecke_h();
169    let mut result = hecke_mul_ts_left(x, i);
170    for (w, c) in x {
171        let h_c = h.clone() * c.clone();
172        let entry = result.entry(w.clone()).or_insert_with(LaurentPolyQ::zero);
173        *entry -= &h_c;
174    }
175    result.retain(|_, v| !v.is_zero());
176    result
177}

179/// **Hecke conjugation by T_{s_i}**: returns `T_{s_i} · x · T_{s_i}⁻¹`.
180///
181/// For class function evaluation: `χ_λ(T_{s_i} · x · T_{s_i}⁻¹) = χ_λ(x)`.
182pub fn hecke_conjugate_by_s(x: &HeckeElement, i: usize) -> HeckeElement {
183    let after_left = hecke_mul_ts_left(x, i);
184    hecke_mul_ts_inv_right(&after_left, i)
185}

187/// Right-multiply by `T_{s_i}^{-1} = T_{s_i} − (q − q^{-1})`.
188pub fn hecke_mul_ts_inv_right(x: &HeckeElement, i: usize) -> HeckeElement {
189    let h = LaurentPolyQ::hecke_h();
190    // x · T_{s_i}^{-1} = x · T_{s_i} − (q − q^{-1}) · x.
191    let mut result = hecke_mul_ts_right(x, i);
192    for (w, c) in x {
193        let h_c = h.clone() * c.clone();
194        let entry = result.entry(w.clone()).or_insert_with(LaurentPolyQ::zero);
195        *entry -= &h_c;
196    }
197    result.retain(|_, v| !v.is_zero());
198    result
199}

201/// Right-multiply by `2 · (σ + σ⁻¹)/2 = σ + σ⁻¹`, i.e. the **doubled
202/// averaged crossing**.
203///
204/// **Why doubled**: the averaged Hecke crossing as a matrix is
205/// `(σ + σ⁻¹) / 2 = σ − h/2 · I` (via the Hecke relation
206/// `σ² = h · σ + I` ⇒ `σ⁻¹ = σ − h`).  Including the `/2` factor in
207/// the Hecke-basis expansion would require rational coefficients in
208/// `LaurentPolyQ`, which currently uses `BigInt`.  Instead we compute
209/// the **doubled** version `(σ + σ⁻¹) · x = 2 · σ x − h · x` here,
210/// and the caller tracks a separate `2^k` divisor (one factor of 2 per
211/// averaged crossing in the word).  [`chi_lambda_via_gp_hoefsmit`]
212/// divides the final χ^λ output by `2^k` to absorb the deferred
213/// normalisation.
214///
215/// Returns a single Hecke element — **no basis branching**.  Despite
216/// the half-sum semantics, the basis grows the same as for a single
217/// σ-application (NOT doubled per averaged crossing the way a literal
218/// 2^k expansion would).  This is the structural win that makes GP
219/// viable for canonical atomic braids with averaged crossings.
220pub fn hecke_mul_ts_averaged_right(x: &HeckeElement, i: usize) -> HeckeElement {
221    let h = LaurentPolyQ::hecke_h();
222    // (σ + σ⁻¹) · x = (σ · x) + (σ⁻¹ · x)
223    //              = (σ · x) + (σ · x − h · x)
224    //              = 2 · (σ · x) − h · x.
225    let sigma_x = hecke_mul_ts_right(x, i);
226    // result := 2 · sigma_x.
227    let mut result: HeckeElement = BTreeMap::new();
228    for (w, c) in &sigma_x {
229        let mut c2 = c.clone();
230        c2.scalar_mul_assign(2_i32);
231        if !c2.is_zero() {
232            result.insert(w.clone(), c2);
233        }
234    }
235    // result := 2 · sigma_x − h · x.
236    for (w, c) in x {
237        let h_c = h.clone() * c.clone();
238        let entry = result.entry(w.clone()).or_insert_with(LaurentPolyQ::zero);
239        *entry -= &h_c;
240    }
241    result.retain(|_, v| !v.is_zero());
242    result
243}

245/// Convert a braid word `[(sign, gen_1based), ...]` on `n` strands
246/// into its expansion `T_β = Σ c_w(q) · T_w` in the Hecke basis.
247pub fn braid_to_hecke(n: usize, word: &[(i32, u32)]) -> HeckeElement {
248    let (x, _avg_count) = braid_to_hecke_with_avg(n, word);
249    x
250}

252/// Like [`braid_to_hecke`] but also handles **averaged** crossings
253/// (sign == 0) and returns the count of averaged crossings encountered.
254///
255/// The averaged crossings are expanded via the doubled half-sum
256/// `(σ + σ⁻¹) · x = 2 σ x − h x` (see
257/// [`hecke_mul_ts_averaged_right`]).  Each averaged crossing contributes
258/// a deferred factor of `1/2` to the χ^λ output; the caller divides
259/// by `2^avg_count` at the χ^λ evaluation step.
260///
261/// **Why a separate `avg_count` return** instead of pre-normalising:
262/// `LaurentPolyQ` uses `BigInt` coefficients, so `h/2 = (q − q⁻¹)/2`
263/// cannot be represented directly without migrating the whole
264/// `LaurentPolyQ` arithmetic to rationals.  Tracking `2^k` as an
265/// external scalar avoids that refactor and matches the
266/// `averaging_denominator` convention already used by
267/// `tools/hecke-engine/src/wenzl_lr.rs::averaging_denominator`.
268///
269/// **Key structural property**: the basis size grows as if we'd done
270/// a single σ-application per averaged crossing — NOT the 2^k blow-up
271/// that a literal sub-word enumeration would produce.  The half-sum
272/// is computed as ONE Hecke element via the algebraic identity, not
273/// as two separate computations.  This is the structural win that
274/// makes GP viable on canonical atomic braids with averaged crossings.
275pub fn braid_to_hecke_with_avg(
276    n: usize,
277    word: &[(i32, u32)],
278) -> (HeckeElement, u32) {
279    let mut x = hecke_identity(n);
280    let mut avg_count: u32 = 0;
281    for &(sign, gen) in word {
282        let i = (gen as usize).saturating_sub(1);
283        if i + 1 >= n {
284            continue;
285        }
286        x = if sign > 0 {
287            hecke_mul_ts_right(&x, i)
288        } else if sign < 0 {
289            hecke_mul_ts_inv_right(&x, i)
290        } else {
291            // sign == 0: averaged crossing.
292            avg_count = avg_count
293                .checked_add(1)
294                .expect("braid_to_hecke_with_avg: averaged-crossing count overflow");
295            hecke_mul_ts_averaged_right(&x, i)
296        };
297    }
298    (x, avg_count)
299}

301/// **Compute minimal length within S_n conjugacy class** of cycle
302/// type `mu` (sorted descending).  Equals `Σ (μ_i - 1) = n - r(μ)`
303/// where `r(μ)` is the number of parts.
304fn min_length_for_cycle_type(mu: &[usize]) -> usize {
305    let n: usize = mu.iter().sum();
306    let r = mu.len();
307    n - r
308}

310/// **Reduced word for a permutation** (lex-first reduced expression).
311///
312/// Returns a sequence of 0-indexed simple-reflection indices `[i_1, i_2, ...]`
313/// such that `w = s_{i_1} · s_{i_2} · ... · s_{i_k}` and `k = ℓ(w)`.
314///
315/// Algorithm: while w is not identity, find the **leftmost** descent
316/// (smallest i such that w(i) > w(i+1)), apply s_i on the right
317/// (i.e. swap positions i and i+1 — this removes that descent and
318/// strictly decreases inversion count), and prepend i to the word.
319/// Any descent gives a valid reduced word; we take the leftmost one
320/// for determinism.  Per Copilot review #r3144... the docstring used
321/// to say "rightmost" while the code took the leftmost — corrected
322/// to match the implementation.
323pub fn reduced_word(w: &Perm) -> Vec<u32> {
324    let mut current = w.clone();
325    let n = current.n();
326    let mut word = Vec::new();
327    while current.length() > 0 {
328        // Find leftmost ascent in the current → identity reduction
329        // (= leftmost descent in current).
330        let mut found = None;
331        for i in 0..n.saturating_sub(1) {
332            if current.0[i] > current.0[i + 1] {
333                found = Some(i);
334                break;
335            }
336        }
337        match found {
338            Some(i) => {
339                current.0.swap(i, i + 1);
340                word.push((i + 1) as u32);  // 1-indexed
341            }
342            None => break,
343        }
344    }
345    word.reverse();
346    word
347}

349/// **F3.2.η — correct-but-slow chi via Hoefsmit per T_w.**
350///
351/// For each basis element T_w in the expansion of T_β, compute
352/// χ_λ(T_w; q) by finding a reduced word for w and evaluating
353/// the seminormal kernel on that word.  Sum: χ_λ(T_β) = Σ c_w · χ_λ(T_w).
354///
355/// **Trade-off**: correct at any q (matches Hoefsmit on β directly),
356/// but no MN speedup — cost is `O(|basis| · dim² · ℓ_max)` which
357/// is similar to direct Hoefsmit on β.  Useful as a CORRECTNESS
358/// BASELINE to validate the basis-expansion + LR composition
359/// infrastructure, even though F3.2.γ via q-MN would be faster
360/// once F3.2.η_full (proper Geck-Pfeiffer cuspidal moves) lands.
361pub fn chi_lambda_via_gp_hoefsmit(
362    lambda: &[usize],
363    n: usize,
364    word: &[(i32, u32)],
365    q: f64,
366) -> f64 {
367    let n_lambda: usize = lambda.iter().sum();
368    if n_lambda != n {
369        return 0.0;
370    }
371    // **Averaged-crossing handling (C-full Step 1, 2026-05-18)**:
372    // `braid_to_hecke_with_avg` expands each averaged crossing
373    // (sign == 0) via the doubled half-sum
374    // `(σ + σ⁻¹) · x = 2 σ x − h x` as a single Hecke element (no
375    // 2^k branching of the basis).  Each averaged crossing carries a
376    // deferred factor of `1/2` which we divide out at the χ^λ
377    // evaluation step here.
378    let (basis_expansion, avg_count) = braid_to_hecke_with_avg(n, word);
379    let mut total = 0.0_f64;
380    for (w, c_w) in &basis_expansion {
381        let red_word = reduced_word(w);
382        let positive_word: Vec<(i32, u32)> = red_word.iter().map(|&g| (1, g)).collect();
383        let chi_w = crate::seminormal::chi_lambda_braid(lambda, &positive_word, q);
384        let c_w_f64 = c_w.evaluate_f64(q);
385        total += c_w_f64 * chi_w;
386    }
387    // Apply the deferred `(1/2)^avg_count` averaging normalisation.
388    // **Why `f64::powi` instead of `1u64 << avg_count`**: the bit-
389    // shift is UB for `avg_count >= 64` (debug panic; release wraps).
390    // Per Copilot + Gemini review on PR #720, this is a real concern
391    // at the scaling targets — canonical atomic braid words for
392    // heavy atoms can exceed 64 averaged crossings (per
393    // `mass_at_3A_proper.atom_braid_word_3A`, each mixed-quark pair
394    // and each pn-interface contributes; for A=40 the count is
395    // ~120). f64::powi handles the full range up to f64::INFINITY
396    // (~2^1023) and underflows gracefully beyond f64 mantissa width
397    // (~1024) rather than silently corrupting the result.
398    if avg_count > 0 {
399        total /= 2.0_f64.powi(avg_count as i32);
400    }
401    total
402}

404/// **MPFR variant** of [`chi_lambda_via_gp_hoefsmit`] at arbitrary
405/// precision (≥ 50 dps recommended for the QOU canonical-χ targets).
406///
407/// C-full Step 2 (2026-05-18): per the C-gate audit
408/// (`docs/audits/2026-05-18-hoefsmit-c-gate-gp-cross-validate.md`),
409/// the GP path needs MPFR for production canonical |tr_M| compute.
410/// Uses [`crate::seminormal_mpfr::chi_lambda_braid_mpfr`] per T_w +
411/// [`LaurentPolyQ::evaluate_mpfr`] for the basis-coefficient
412/// polynomials, all at the precision implied by `q`.
413///
414/// **`q` is supplied as a decimal string** (not `f64`) to preserve
415/// arbitrary precision through the FFI; the caller passes the
416/// canonical 50-dps `Q_50_DIGIT_STR` (or higher) from
417/// `folio-assistant/computations/q_parameter.py`. This matches the
418/// convention of `chi_lambda_canonical_h_n_mpfr` in `wenzl_lr.rs`.
419///
420/// **Averaged-crossing handling**: same as the f64 variant — uses
421/// [`braid_to_hecke_with_avg`] and divides by `2^avg_count` at the
422/// end. The divisor is computed via the same `f64::powi` pattern
423/// (then cast to MPFR `Float`) to avoid the `1u64 << k` UB for
424/// large k (per Copilot/Gemini review on PR #720).
425///
426/// Returns the χ^λ value as a decimal string at `dps` precision
427/// (matches `chi_lambda_braid_mpfr` for FFI compatibility).
428pub fn chi_lambda_via_gp_hoefsmit_mpfr(
429    lambda: &[usize],
430    n: usize,
431    word: &[(i32, u32)],
432    q_str: &str,
433    dps: u32,
434) -> String {
435    use rug::Float;
436    let n_lambda: usize = lambda.iter().sum();
437    let prec = (dps as f64 * 3.322 + 16.0) as u32;
438    if n_lambda != n {
439        return Float::with_val(prec, 0).to_string_radix(10, Some(dps as usize));
440    }
441
442    // Parse q at MPFR precision. A parse failure here is a caller
443    // bug (caller is responsible for supplying a valid decimal
444    // string) — panic loudly per Gemini review on #725.
445    let q = Float::parse(q_str)
446        .map(|p| Float::with_val(prec, p))
447        .expect("chi_lambda_via_gp_hoefsmit_mpfr: q_str is not a valid decimal");
448
449    let (basis_expansion, avg_count) = braid_to_hecke_with_avg(n, word);
450    let mut total = Float::with_val(prec, 0);
451
452    for (w, c_w) in &basis_expansion {
453        let red_word = reduced_word(w);
454        let positive_word: Vec<(i32, u32)> =
455            red_word.iter().map(|&g| (1, g)).collect();
456        // Per-T_w character at MPFR precision.
457        let chi_w_str = crate::seminormal_mpfr::chi_lambda_braid_mpfr(
458            lambda, &positive_word, q_str, dps,
459        );
460        // chi_lambda_braid_mpfr produces valid decimal output; a
461        // parse failure here would indicate corruption, not a
462        // recoverable error.
463        let chi_w = Float::parse(&chi_w_str)
464            .map(|p| Float::with_val(prec, p))
465            .expect("chi_lambda_braid_mpfr returned an invalid decimal string");
466        // Basis coefficient c_w(q) evaluated at MPFR precision.
467        let c_w_mpfr = c_w.evaluate_mpfr(&q);
468        // total += c_w * chi_w (all MPFR).
469        let prod = Float::with_val(prec, &c_w_mpfr * &chi_w);
470        total += &prod;
471    }
472
473    // Deferred (1/2)^avg_count normalisation.
474    // **Why exact right-shift instead of `f64::powi` cast** (per
475    // Gemini review on #725): `rug::Float` supports `ShrAssign<u32>`
476    // which divides by exactly 2^k via the binary exponent field —
477    // no precision loss, no f64-range cap (handles all u32 values),
478    // exact in MPFR.  The `f64::powi` cast pattern from #720's
479    // Copilot/Gemini review was for the f64 path; in MPFR we have
480    // a strictly-better option.
481    if avg_count > 0 {
482        total >>= avg_count;
483    }
484    total.to_string_radix(10, Some(dps as usize))
485}

487/// **All partitions, MPFR, with per-T_w memoisation** (C-full Step 3).
488///
489/// Compute χ^λ for every partition λ ⊢ n on the same braid word, at
490/// MPFR precision, sharing the GP basis-expansion + per-T_w
491/// characters across partitions where possible.
492///
493/// **Why this is faster than calling `chi_lambda_via_gp_hoefsmit_mpfr`
494/// per partition in a loop**:
495///
496/// - `braid_to_hecke_with_avg(n, word)` runs ONCE (basis expansion is
497///   λ-independent).
498/// - `reduced_word(w)` for each `w` in the expansion runs ONCE.
499/// - The per-T_w MPFR character `chi_lambda_braid_mpfr(λ, positive_word, ...)`
500///   is computed once per (λ, distinct positive_word) pair, not once
501///   per (λ, basis-element). Different basis elements `T_w` and `T_w'`
502///   often have the same reduced word (in their minimal-length
503///   conjugacy class representative), so the cache hit rate grows
504///   with basis size.
505/// - The deferred `2^avg_count` divisor is computed once and applied
506///   per partition.
507///
508/// Returns `Vec<(partition, chi_decimal_string)>` mirroring
509/// `chi_lambda_canonical_all_partitions_mpfr`'s shape. Inner
510/// computation is parallel across partitions via rayon.
511///
512/// **Memoisation thread-safety**: each partition's rayon task builds
513/// its own λ-specific `chi_cache` (HashMap from reduced-word → χ^λ
514/// MPFR) over the deduplicated `distinct_words` list. The cache is
515/// **per-partition**, not shared across rayon tasks — cross-partition
516/// sharing would require cross-thread mutation under rayon, which we
517/// avoid. The λ-independent prefix (basis expansion, reduced-word
518/// extraction, distinct-word deduplication, c_w(q) MPFR evaluation)
519/// IS shared across all partitions: that part runs once before the
520/// rayon dispatch and is read-only inside the parallel loop.
521/// (Note corrected per Copilot review on PR #728.)
522pub fn chi_lambda_via_gp_hoefsmit_all_partitions_mpfr(
523    n: usize,
524    word: &[(i32, u32)],
525    q_str: &str,
526    dps: u32,
527) -> Vec<(Vec<usize>, String)> {
528    use rayon::prelude::*;
529    use rug::Float;
530    use std::collections::HashMap;
531
532    let prec = (dps as f64 * 3.322 + 16.0) as u32;
533
534    // 1. Basis expansion runs once for the whole sweep.
535    let (basis_expansion, avg_count) = braid_to_hecke_with_avg(n, word);
536
537    // 2. For each w in the expansion, pre-compute its reduced word
538    //    (a Vec<u32> of simple-transposition indices). The keying
539    //    by reduced word lets distinct `w`'s share their χ^λ
540    //    computation when they reduce to the same canonical form.
541    let basis_terms: Vec<(Vec<u32>, LaurentPolyQ)> = basis_expansion
542        .iter()
543        .map(|(w, c_w)| (reduced_word(w), c_w.clone()))
544        .collect();
545
546    // 3. Collect the distinct positive_word signatures across the
547    //    basis. The same reduced word can appear for multiple
548    //    basis elements; cache its char once per λ.
549    let mut distinct_words: Vec<Vec<u32>> = basis_terms
550        .iter()
551        .map(|(rw, _)| rw.clone())
552        .collect();
553    distinct_words.sort();
554    distinct_words.dedup();
555
556    // 4. Pre-compute the c_w(q) MPFR evaluations once (λ-independent).
557    // Per Gemini review on PR #728 line 553: parse failure is a
558    // caller bug, not a recoverable error — panic loudly rather
559    // than silently default to q = 1.
560    let q = Float::parse(q_str)
561        .map(|p| Float::with_val(prec, p))
562        .expect(
563            "chi_lambda_via_gp_hoefsmit_all_partitions_mpfr: \
564             q_str is not a valid decimal",
565        );
566    let c_w_mpfr_per_term: Vec<Float> = basis_terms
567        .iter()
568        .map(|(_, c_w)| c_w.evaluate_mpfr(&q))
569        .collect();
570
571    // 5. Deferred 2^avg_count normalisation will be applied per-
572    //    partition inside the rayon loop via exact `>>= avg_count`
573    //    on the running MPFR Float (mantissa-preserving via the
574    //    binary exponent field, range-safe across the full u32, no
575    //    f64-range cap).  Matches the convention in
576    //    `chi_lambda_via_gp_hoefsmit_mpfr` after the Gemini #728
577    //    review.
578
579    // 6. Per λ: build the per-distinct-word cache, then sum.
580    let partitions = crate::seminormal::partitions_of(n);
581    partitions
582        .par_iter()
583        .map(|lambda| {
584            // Per-T_w MPFR character cache for THIS λ.  Built sequentially
585            // inside the partition's rayon task (no cross-partition state).
586            let mut chi_cache: HashMap<Vec<u32>, Float> = HashMap::with_capacity(
587                distinct_words.len(),
588            );
589            for rw in &distinct_words {
590                let positive_word: Vec<(i32, u32)> =
591                    rw.iter().map(|&g| (1, g)).collect();
592                let chi_str = crate::seminormal_mpfr::chi_lambda_braid_mpfr(
593                    lambda, &positive_word, q_str, dps,
594                );
595                // Per Gemini + Copilot review on PR #728:
596                // chi_lambda_braid_mpfr emits valid decimal output;
597                // a parse failure here would indicate corruption
598                // and silently coercing to 0 would mask the bug and
599                // also look like a legitimate cancellation.  Panic
600                // loudly with the offending string instead.
601                let chi_f = Float::parse(&chi_str)
602                    .map(|p| Float::with_val(prec, p))
603                    .unwrap_or_else(|_| {
604                        panic!(
605                            "chi_lambda_braid_mpfr returned a non-decimal \
606                             string: {chi_str:?}"
607                        )
608                    });
609                chi_cache.insert(rw.clone(), chi_f);
610            }
611
612            // Sum c_w * χ^λ(T_w) over the basis.
613            let mut total = Float::with_val(prec, 0);
614            for ((rw, _c_w), c_w_mpfr) in basis_terms.iter().zip(c_w_mpfr_per_term.iter()) {
615                let chi = chi_cache
616                    .get(rw)
617                    .expect("chi_cache built from distinct_words, lookup must succeed");
618                let prod = Float::with_val(prec, c_w_mpfr * chi);
619                total += &prod;
620            }
621
622            // Apply the deferred 2^avg_count normalisation via
623            // exact MPFR right-shift (mantissa-preserving, full-u32
624            // safe).  Per Gemini #565 + Copilot #563 review on PR
625            // #728: the f64::powi cast pattern from the older
626            // divisor approach caps at f64 range ~2^1023 and
627            // overflows to inf for avg_count > 1023, collapsing
628            // results to 0.  `>>=` does the exact division via the
629            // binary exponent field.
630            if avg_count > 0 {
631                total >>= avg_count;
632            }
633            (lambda.clone(), total.to_string_radix(10, Some(dps as usize)))
634        })
635        .collect()
636}

638/// **F3.2.ζ — partial Hecke-conjugation reduction (simple-reflection only).**
639///
640/// Given a Hecke element `x = Σ c_w · T_w`, repeatedly find a basis
641/// element `T_w` whose `w` is NOT minimal-length within its
642/// `S_n` conjugacy class, and apply Hecke conjugation by a SIMPLE
643/// reflection `s_i` such that `ℓ(s_i w s_i) < ℓ(w)`.  The conjugation
644/// produces a sum of basis elements via the Hecke relations.
645///
646/// **Algorithmic limitation (motivates F3.2.η)**: not every non-minimal
647/// element of `S_n` has a length-reducing simple-reflection
648/// conjugation.  For example, in the conjugacy class of 6-cycles in
649/// `S_6`, the element `[1, 3, 4, 5, 0, 2]` has length 7 and minimal
650/// length 5, but no `s_i w s_i` reduces length.  The full
651/// Geck-Pfeiffer 2000 §3.2 algorithm uses "good cuspidal moves"
652/// (specific products of consecutive simple reflections) to reach
653/// minimal-length representatives in such cases.
654///
655/// **Empirical**: on the D atomic braid (n=6, word_len=23), this
656/// partial reduction takes the basis from ~700 elements down to 145,
657/// of which 17 are minimal-length and 128 still need the full
658/// cuspidal-move algorithm.
659///
660/// At q = 1, even partial reduction is sufficient because cycle type
661/// determines the character; the F3.2.γ wiring matches classical MN
662/// at q = 1 regardless of `reduce_basis_to_minimal_length` behaviour.
663pub fn reduce_basis_to_minimal_length(x: &HeckeElement) -> HeckeElement {
664    let mut current = x.clone();
665    // Iterate until fixed point.
666    loop {
667        // Find a non-minimal w in current.
668        let target = current.iter()
669            .find_map(|(w, _c)| {
670                let mu = perm_cycle_type(w);
671                let min_len = min_length_for_cycle_type(&mu);
672                if w.length() > min_len { Some(w.clone()) } else { None }
673            });
674        let Some(target_w) = target else {
675            return current;  // all basis elements are minimal-length
676        };
677        let n = target_w.n();
678
679        // Find a simple reflection s such that ℓ(s · target_w · s) < ℓ(target_w).
680        let target_len = target_w.length();
681        let mut chosen_s: Option<usize> = None;
682        for i in 0..n.saturating_sub(1) {
683            let mut sws = target_w.clone();
684            sws.apply_s_left(i);
685            sws.apply_s_right(i);
686            if sws.length() < target_len {
687                chosen_s = Some(i);
688                break;
689            }
690        }
691        let Some(s_idx) = chosen_s else {
692            // No length-reducing conjugation found.  This shouldn't
693            // happen for n ≤ 6 in S_n (every non-minimal element has
694            // a length-reducing s) but bail safely if it does.
695            return current;
696        };
697
698        // Apply Hecke conjugation: replace c_w · T_w with
699        // c_w · (T_s · T_w · T_s⁻¹) in current.
700        let coeff = current.remove(&target_w).unwrap_or_else(LaurentPolyQ::zero);
701        // Build singleton {target_w → coeff}.
702        let mut singleton: HeckeElement = BTreeMap::new();
703        singleton.insert(target_w.clone(), coeff);
704        let conjugated = hecke_conjugate_by_s(&singleton, s_idx);
705        // Merge into current.
706        for (v, c_v) in conjugated {
707            let entry = current.entry(v).or_insert_with(LaurentPolyQ::zero);
708            *entry += &c_v;
709        }
710        current.retain(|_, v| !v.is_zero());
711    }
712}

714/// **F3.2.ε — Geck-Pfeiffer cuspidal reduction by simple-reflection conjugation.**
715///
716/// Given `w ∈ S_n`, repeatedly conjugate by simple reflections `s`
717/// (i.e., `w ↦ s w s`) when this strictly decreases length, until
718/// no such reduction is possible.  The resulting element has the
719/// same cycle type as the input but minimal length within the
720/// conjugation orbit reachable by simple-reflection moves.
721///
722/// **Theoretical basis**: by Geck-Pfeiffer 2000 §3, two elements
723/// connected by a length-decreasing conjugation `w → sws` give the
724/// same value on every class function of `H_n(q)`:
725///
726///   χ_λ(T_w; q) = χ_λ(T_{sws}; q)   if ℓ(sws) < ℓ(w).
727///
728/// **Note**: this simple algorithm reaches MINIMAL representatives
729/// for all conjugacy classes in `S_n` for `n ≤ 6` (sufficient for
730/// D / T atomic braids).  For larger `n`, some classes may require
731/// "good cuspidal moves" beyond simple conjugation; for those the
732/// full Geck-Pfeiffer §3.2 algorithm is needed.
733pub fn reduce_to_cuspidal(w: &Perm) -> Perm {
734    let n = w.n();
735    let mut current = w.clone();
736    loop {
737        let cur_len = current.length();
738        if cur_len == 0 {
739            return current;  // identity is cuspidal
740        }
741        let mut reduced = false;
742        for i in 0..n.saturating_sub(1) {
743            // Compute s_i · w · s_i.
744            let mut candidate = current.clone();
745            candidate.apply_s_left(i);
746            candidate.apply_s_right(i);
747            if candidate.length() < cur_len {
748                current = candidate;
749                reduced = true;
750                break;
751            }
752        }
753        if !reduced {
754            return current;
755        }
756    }
757}

759/// Cycle type of a permutation (sorted descending).
760fn perm_cycle_type(perm: &Perm) -> Vec<usize> {
761    let n = perm.n();
762    let mut visited = vec![false; n];
763    let mut cycles = Vec::new();
764    for start in 0..n {
765        if visited[start] {
766            continue;
767        }
768        let mut len = 0;
769        let mut j = start;
770        while !visited[j] {
771            visited[j] = true;
772            j = perm.0[j];
773            len += 1;
774        }
775        cycles.push(len);
776    }
777    cycles.sort_by(|a, b| b.cmp(a));
778    cycles
779}

781/// **F3.2.γ wiring**: compute `χ_λ(T_β; q)` for an arbitrary braid
782/// word by expanding `T_β = Σ c_w · T_w` in the Hecke basis and
783/// applying q-MN on each `T_w`.
784///
785///   χ_λ(T_β; q) = Σ_w c_w(q) · χ_λ(T_w; q)
786///
787/// **CRITICAL CORRECTNESS NOTE — F3.2.γ is structurally incomplete.**
788///
789/// At q ≠ 1, two basis elements `T_w` and `T_{w'}` with the SAME
790/// cycle type but DIFFERENT lengths give DIFFERENT Hecke characters.
791/// This is NOT a class function on cycle type — it's a class
792/// function on `H_n(q)` (invariant under Hecke conjugation), which
793/// for symmetric-group permutations means: invariant ONLY among
794/// MINIMAL-LENGTH representatives of each conjugacy class.
795///
796/// Empirical verification (see q_mn_canonical.py cross-check):
797///   χ^(2,1)(T_{s_1};       q₀) ≈ +0.209  (length 1, minimal)
798///   χ^(2,1)(T_{s_1 s_2 s_1}; q₀) ≈ -0.000  (length 3, NOT minimal)
799///   Both have cycle type (2, 1).
800///
801/// The simplified q-MN treats both as the canonical-cycle-type
802/// character of the minimal-length representative — incorrect for
803/// non-minimal w.
804///
805/// **Two paths to a correct implementation (F3.2.δ):**
806///
807/// 1. **Geck-Pfeiffer cuspidal reduction**: for each `T_w` in the
808///    expansion, reduce w to its minimal-length representative
809///    `w_min` within its conjugacy class via "good conjugations"
810///    (s w s = w' with ℓ(w') ≤ ℓ(w) - 2), then apply q-MN to
811///    `cycle_type(w_min)`.  Reference: Geck-Pfeiffer 2000 §3.
812///
813/// 2. **Hoefsmit per T_w**: for each basis element, build the
814///    canonical reduced word `w_min` and evaluate via Hoefsmit
815///    seminormal.  Correct but no MN speedup.
816///
817/// At q = 1 the simplified path is correct (Hecke ↦ S_n; characters
818/// ARE class functions on cycle type alone).
819pub fn chi_lambda_via_geck_pfeiffer(
820    lambda: &[usize],
821    n: usize,
822    word: &[(i32, u32)],
823) -> LaurentPolyQ {
824    let n_lambda: usize = lambda.iter().sum();
825    if n_lambda != n {
826        return LaurentPolyQ::zero();
827    }
828    let basis_expansion = braid_to_hecke(n, word);
829    // F3.2.ζ: reduce ALL basis elements to minimal-length representatives
830    // via iterated Hecke conjugation.  This produces a sum where
831    // each basis member T_w has ℓ(w) = minimal length within its
832    // conjugacy class — at which point q-MN on cycle_type(w) gives
833    // the correct character.
834    let reduced = reduce_basis_to_minimal_length(&basis_expansion);
835    let mut total = LaurentPolyQ::zero();
836    for (w, c_w) in &reduced {
837        let mu = perm_cycle_type(w);
838        let chi_w = crate::seminormal_mn::chi_lambda_mn_qdef(lambda, &mu);
839        total += &(c_w.clone() * chi_w);
840    }
841    total
842}
