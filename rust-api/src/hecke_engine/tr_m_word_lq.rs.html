1//! Markov-axiom recursive ascent for the canonical Jones-Markov
2//! trace, ported from `folio-assistant/computations/laurent_q_trace.py`.
3//!
4//! Strategy #1 from
5//! `docs/audits/2026-05-20-markov-peel-bottleneck-profile.md`,
6//! Phase-2 (Rust port) per the user's "deeper Rust port" approval.
7//!
8//! Output type: [`ZHLaurent`] — a formal polynomial in (z, h) with
9//! [`LaurentPolyQ`] coefficients. The Python side calls
10//! `tr_m_word_lq_serialized(word, n_strands)` via PyO3, gets back
11//! a serialised ZHLaurent (`Vec<((i32, i32), Vec<(i32, BigInt)>)>`),
12//! and evaluates at q₀ to mpmath.mpf at production dps.
13//!
14//! No f64 anywhere in the recursion — every coefficient is exact
15//! `BigInt` via [`LaurentPolyQ`]. Per CLAUDE.md §Precision goals L1.
16

18use std::collections::{BTreeMap, HashMap};

21/// Formal polynomial in (z, h) with [`LaurentPolyQ`] coefficients.
22///
23/// Internal representation: `BTreeMap<(i32, i32), LaurentPolyQ>` —
24/// (z_exp, h_exp) → coefficient. BTreeMap for deterministic
25/// iteration order (canonical equality + cache key stability).
26#[derive(Clone, Debug, Default)]
27pub struct ZHLaurent {
28    pub terms: BTreeMap<(i32, i32), LaurentPolyQ>,
29}
30
31impl ZHLaurent {
32    pub fn zero() -> Self {
33        Self { terms: BTreeMap::new() }
34    }
35
36    pub fn one() -> Self {
37        let mut t = BTreeMap::new();
38        t.insert((0, 0), LaurentPolyQ::one());
39        Self { terms: t }
40    }
41
42    pub fn is_zero(&self) -> bool {
43        self.terms.is_empty()
44    }
45
46    /// Multiply by the variable z: shift each (a, b) → (a+1, b).
47    pub fn mul_by_z(&self) -> Self {
48        let mut out = BTreeMap::new();
49        for (&(a, b), c) in &self.terms {
50            out.insert((a + 1, b), c.clone());
51        }
52        Self { terms: out }
53    }
54
55    /// Multiply by the variable h: shift each (a, b) → (a, b+1).
56    pub fn mul_by_h(&self) -> Self {
57        let mut out = BTreeMap::new();
58        for (&(a, b), c) in &self.terms {
59            out.insert((a, b + 1), c.clone());
60        }
61        Self { terms: out }
62    }
63
64    /// `self + other` (dict-merge with LaurentPolyQ addition on collisions).
65    pub fn add_ref(&self, other: &ZHLaurent) -> Self {
66        let mut out: BTreeMap<(i32, i32), LaurentPolyQ> = self.terms.clone();
67        for (&k, v) in &other.terms {
68            match out.get_mut(&k) {
69                Some(existing) => {
70                    let sum = existing.clone() + v.clone();
71                    if sum.terms.is_empty() {
72                        out.remove(&k);
73                    } else {
74                        *existing = sum;
75                    }
76                }
77                None => {
78                    out.insert(k, v.clone());
79                }
80            }
81        }
82        Self { terms: out }
83    }
84
85    /// Serialise to a Vec for FFI. Each entry: `((z_exp, h_exp),
86    /// Vec<(q_half_exp, big_int_decimal_string)>)`. The BigInt is
87    /// emitted as a decimal string so PyO3 can ferry arbitrary-
88    /// precision integers across the boundary.
89    pub fn to_ffi(&self) -> Vec<((i32, i32), Vec<(i32, String)>)> {
90        self.terms
91            .iter()
92            .map(|(&(a, b), coef)| {
93                let coefs: Vec<(i32, String)> = coef
94                    .terms
95                    .iter()
96                    .map(|(&exp, c)| (exp, c.to_string()))
97                    .collect();
98                ((a, b), coefs)
99            })
100            .collect()
101    }
102}

108/// Cache key: (cyclic-canonical word, n_strands, max_depth).
109///
110/// **Tier 1.C — cyclic canonicalisation (2026-05-21).**  The
111/// Markov trace satisfies `tr_M(αβ) = tr_M(βα)`, so any two words
112/// that are cyclic rotations of each other have the same `tr_M`.
113/// The cache key uses the *lexicographically smallest* cyclic
114/// rotation so that all `n` rotations of an `n`-long word share
115/// one cache entry.  The recursion body still operates on the
116/// **original** word (the rotations in the algorithm body are
117/// semantic, not just cache canonicalisation) — only the cache
118/// key is canonicalised.  See the brief in
119/// `docs/audits/2026-05-20-compute-optimization-roadmap.md` §Tier 1.C.
120///
121/// `depth` is excluded — it is a recursion-guard parameter, not
122/// part of the function's semantic identity.
123type CacheKey = (Vec<u32>, usize, u32);
124
125/// Lexicographically smallest cyclic rotation of `word`.
126///
127/// `O(n²)` — adequate at typical atomic-braid word lengths
128/// (≤ a few hundred); could be tightened to `O(n log n)` via
129/// Booth's algorithm if a profile ever flags this.  See
130/// `docs/audits/2026-05-20-compute-optimization-roadmap.md`
131/// §Tier 1.C for the optimisation rationale.
132pub(crate) fn canonical_cyclic(word: &[u32]) -> Vec<u32> {
133    let n = word.len();
134    if n <= 1 {
135        return word.to_vec();
136    }
137    let mut best: Vec<u32> = word.to_vec();
138    let mut rot: Vec<u32> = Vec::with_capacity(n);
139    for i in 1..n {
140        rot.clear();
141        rot.extend_from_slice(&word[i..]);
142        rot.extend_from_slice(&word[..i]);
143        if rot < best {
144            best.clone_from(&rot);
145        }
146    }
147    best
148}

150/// Global cache (HashMap-bounded by maxsize). `Mutex<HashMap>`
151/// rather than a proper LRU for simplicity; at ~100K entries max
152/// the Mutex contention is negligible compared to the per-call
153/// recursion cost.
154static TR_M_CACHE: OnceLock<Mutex<HashMap<CacheKey, ZHLaurent>>> = OnceLock::new();
155
156fn cache() -> &'static Mutex<HashMap<CacheKey, ZHLaurent>> {
157    TR_M_CACHE.get_or_init(|| Mutex::new(HashMap::new()))
158}
159
160/// Clear the global recursion cache. Call between independent atoms
161/// to prevent unbounded memory growth in long-running sessions.
162pub fn clear_cache() {
163    if let Ok(mut g) = cache().lock() {
164        g.clear();
165    }
166}

168/// Canonical Markov-trace recursion at the word level.
169///
170/// `word`: tuple of generator indices (0-based by external convention;
171///         pass 1-based here for consistency with the rest of
172///         hecke-engine? No — markov_trace_canonical uses 0-based, so
173///         we use 0-based too here to match the Python ZHLaurent
174///         implementation exactly).
175/// `n_strands`: H_n strand count.
176///
177/// Mirrors `folio-assistant/computations/laurent_q_trace.py
178/// tr_M_word_lq` line-by-line. Phase-1+2 scope: empty / single-top /
179/// multi-top Case A / Case B. Case C raises (caller falls back to
180/// the sympy ocneanu path — same behaviour as Python).
181pub fn tr_m_word_lq(
182    word: &[u32],
183    n_strands: usize,
184    max_depth: u32,
185) -> Result<ZHLaurent, String> {
186    tr_m_word_lq_impl(word, n_strands, max_depth, 0)
187}
188
189fn tr_m_word_lq_impl(
190    word: &[u32],
191    n_strands: usize,
192    max_depth: u32,
193    depth: u32,
194) -> Result<ZHLaurent, String> {
195    if depth > max_depth {
196        return Err(format!(
197            "tr_m_word_lq recursion too deep: word={:?}, n={}",
198            word, n_strands
199        ));
200    }
201    // Tier 1.C: cache key is the cyclic-canonical rotation, but the
202    // recursion body uses the original `word` (the algorithmic
203    // rotations inside `tr_m_word_lq_body` carry semantic meaning,
204    // not just cache canonicalisation).
205    let key: CacheKey = (canonical_cyclic(word), n_strands, max_depth);
206    if let Some(cached) = cache().lock().unwrap().get(&key) {
207        return Ok(cached.clone());
208    }
209
210    let result = tr_m_word_lq_body(word, n_strands, max_depth, depth)?;
211
212    if let Ok(mut g) = cache().lock() {
213        // Bound the cache; drop everything if we hit the cap.
214        if g.len() >= 100_000 {
215            g.clear();
216        }
217        g.insert(key, result.clone());
218    }
219    Ok(result)
220}
221
222fn tr_m_word_lq_body(
223    word: &[u32],
224    n_strands: usize,
225    max_depth: u32,
226    depth: u32,
227) -> Result<ZHLaurent, String> {
228    // (1) Empty word — tr_M(1) = 1
229    if word.is_empty() {
230        return Ok(ZHLaurent::one());
231    }
232
233    let max_gen = *word.iter().max().unwrap() as usize;
234
235    // (2) Generator support below n - 2 — recurse at H_{max_gen + 2}
236    if max_gen + 2 < n_strands {
237        return tr_m_word_lq_impl(word, max_gen + 2, max_depth, depth + 1);
238    }
239
240    let top = (n_strands as i32 - 2) as u32;
241    let top_positions: Vec<usize> = word
242        .iter()
243        .enumerate()
244        .filter_map(|(i, &g)| if g == top { Some(i) } else { None })
245        .collect();
246
247    // (3) Single top — cyclic rotate + Markov II: tr_M(w·σ_top) = z·tr_M(rest)
248    if top_positions.len() == 1 {
249        let pos = top_positions[0];
250        let mut rest: Vec<u32> = Vec::with_capacity(word.len() - 1);
251        rest.extend_from_slice(&word[pos + 1..]);
252        rest.extend_from_slice(&word[..pos]);
253        let inner = tr_m_word_lq_impl(&rest, n_strands - 1, max_depth, depth + 1)?;
254        return Ok(inner.mul_by_z());
255    }
256
257    // (4) Multiple tops — rotate so last top at rightmost position.
258    let last_top = *top_positions.last().unwrap();
259    let mut rotated: Vec<u32> = Vec::with_capacity(word.len());
260    rotated.extend_from_slice(&word[last_top + 1..]);
261    rotated.extend_from_slice(&word[..=last_top]);
262    let rest = &rotated[..rotated.len() - 1];
263
264    // Defensive guard (matches Python + markov_trace_canonical reference).
265    if !rest.contains(&top) {
266        let inner = tr_m_word_lq_impl(rest, n_strands - 1, max_depth, depth + 1)?;
267        return Ok(inner.mul_by_z());
268    }
269
270    // Find the position of the second-to-last top (the "inner" one).
271    let inner_pos = rest.len() - 1
272        - rest.iter().rev().position(|&g| g == top).unwrap();
273    let segment = &rest[inner_pos + 1..];
274
275    // (5) Case A — no top-1 in segment: apply σ² = hσ + 1 after commute
276    if !segment.iter().any(|&g| g == top - 1) {
277        let prefix = &rest[..inner_pos];
278        let mut new_base: Vec<u32> = Vec::with_capacity(prefix.len() + segment.len());
279        new_base.extend_from_slice(prefix);
280        new_base.extend_from_slice(segment);
281        // term1 = tr_M(new_base + σ_top)
282        let mut new_base_with_top = new_base.clone();
283        new_base_with_top.push(top);
284        let term1 = tr_m_word_lq_impl(&new_base_with_top, n_strands, max_depth, depth + 1)?;
285        // term2 = tr_M(new_base)
286        let term2 = tr_m_word_lq_impl(&new_base, n_strands, max_depth, depth + 1)?;
287        return Ok(term1.mul_by_h().add_ref(&term2));
288    }
289
290    // (6) Case B — top-1 in segment: apply braid relation
291    //     σ_top · σ_{top-1} · σ_top = σ_{top-1} · σ_top · σ_{top-1}
292    let first_tm1_in_seg = segment.iter().position(|&g| g == top - 1).unwrap();
293    let alpha = &segment[..first_tm1_in_seg];
294    if alpha.iter().any(|&g| g == top - 1) {
295        return Err(format!(
296            "tr_m_word_lq: unexpected top-1 in alpha segment; word={:?}, n={}",
297            word, n_strands
298        ));
299    }
300    let beta = &segment[first_tm1_in_seg + 1..];
301    if beta.iter().any(|&g| g == top - 1) {
302        // Case C — multiple top-1 in segment. Python raises
303        // NotImplementedError and caller routes through the sympy
304        // ocneanu fallback. We do the same here so the Python
305        // caller knows to fall back.
306        return Err(format!(
307            "tr_m_word_lq: Case C (multiple top-1 in segment) \
308             not implemented; word={:?}, n={}",
309            word, n_strands
310        ));
311    }
312
313    // Build rewritten word:
314    //   rest[:inner_pos] · alpha · σ_{top-1} · σ_top · σ_{top-1} · beta
315    let mut new_word: Vec<u32> = Vec::with_capacity(rest.len() + 2);
316    new_word.extend_from_slice(&rest[..inner_pos]);
317    new_word.extend_from_slice(alpha);
318    new_word.push(top - 1);
319    new_word.push(top);
320    new_word.push(top - 1);
321    new_word.extend_from_slice(beta);
322    tr_m_word_lq_impl(&new_word, n_strands, max_depth, depth + 1)
323}

334    /// `tr_M(σ_0) = z` — single top at H_2.
335    #[test]
336    fn test_single_top() {
337        clear_cache();
338        let result = tr_m_word_lq(&[0], 2, 200).unwrap();
339        // Should be ZHLaurent { (1, 0): 1 }
340        assert_eq!(result.terms.len(), 1, "expected 1 term, got {:?}", result.terms);
341        let key = (1i32, 0i32);
342        assert!(result.terms.contains_key(&key),
343                "expected (z^1 h^0), got keys {:?}",
344                result.terms.keys().collect::<Vec<_>>());
345        // coefficient should be 1 (i.e. LaurentPolyQ::one())
346        let c = &result.terms[&key];
347        assert!(c.terms.contains_key(&0), "coef should be q^0 = 1");
348    }

378    #[test]
379    fn canonical_cyclic_smoke() {
380        // Length 0 / 1 — passthrough.
381        let v0: Vec<u32> = vec![];
382        assert_eq!(canonical_cyclic(&v0), v0);
383        let v1 = vec![7u32];
384        assert_eq!(canonical_cyclic(&v1), v1);
385
386        // (0, 1, 2) and (1, 2, 0) and (2, 0, 1) all canonicalise to (0, 1, 2).
387        let canon = vec![0u32, 1, 2];
388        assert_eq!(canonical_cyclic(&[0, 1, 2]), canon);
389        assert_eq!(canonical_cyclic(&[1, 2, 0]), canon);
390        assert_eq!(canonical_cyclic(&[2, 0, 1]), canon);
391
392        // (2, 1, 0), (1, 0, 2), (0, 2, 1) — pick (0, 2, 1).
393        let canon2 = vec![0u32, 2, 1];
394        assert_eq!(canonical_cyclic(&[2, 1, 0]), canon2);
395        assert_eq!(canonical_cyclic(&[1, 0, 2]), canon2);
396        assert_eq!(canonical_cyclic(&[0, 2, 1]), canon2);
397    }

399    #[test]
400    fn cache_hits_on_cyclic_rotation() {
401        // Two words that are cyclic rotations of each other should
402        // share a cache entry.  Verifies via cache-size increment.
403        clear_cache();
404        let _ = tr_m_word_lq(&[0, 1], 3, 200).unwrap();
405        let size1 = cache().lock().unwrap().len();
406        // Rotation [1, 0] = σ_1 σ_0.  tr_M is cyclic-invariant so
407        // tr_M(σ_0 σ_1) = tr_M(σ_1 σ_0); the cache canonicalises
408        // both to the same key.
409        let _ = tr_m_word_lq(&[1, 0], 3, 200).unwrap();
410        let size2 = cache().lock().unwrap().len();
411        // Without cyclic canonicalisation the second call would
412        // add a new entry; with canonicalisation it must hit the
413        // existing one.  (The body may add intermediate entries
414        // for recursive sub-calls — we only check that the *top-
415        // level* cyclic-rotation call did not increment.)
416        // The two calls share the same recursion tree, so size
417        // should stay equal at the top-level invocation count.
418        assert_eq!(size1, size2,
419                   "cyclic rotation should hit existing cache entry; \
420                    size before {} != after {}",
421                   size1, size2);
422    }
