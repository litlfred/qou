1//! Joint-tower SDP certificate format.
2//!
3//! Production-scale Rust analogue of the design-spec certificate
4//! emitted by the Python proof-of-concept
5//! [`gb_filtration_jet_tracker.py`](../../../folio-assistant/computations/gb_filtration_jet_tracker.py).
6//!
7//! Schema: matches `gb-filtration-jet-order.witness.json` (Python PoC)
8//! and `joint-tower-sdp-h3.witness.json` / `joint-tower-sdp-h6-h9.witness.json`
9//! (Python solver outputs).  The Rust extension lets us scale to
10//! H_18 (⁶Li), H_21 (⁷Li), and α-cluster atoms (⁸Be → ⁴⁰Ca) where
11//! the Python implementation does not.
12//!
13//! See [RUST_INTEGRATION.md](../../RUST_INTEGRATION.md) §"Phase R1".
14
15use serde::Serialize;
16use std::collections::BTreeMap;
17
18/// One reduction event in the jet-order log.  Matches the Python
19/// `JetEvent` dataclass.
20#[derive(Serialize, Clone, Debug)]
21pub struct JetEvent {
22    pub step: u64,
23    pub multiplier: i32,
24    pub predecessor: Vec<usize>,
25    pub successor: Vec<usize>,
26    pub filtration_before: u32,
27    pub filtration_after: u32,
28    /// "ascending" | "hecke-quadratic"
29    pub relation: String,
30    /// SymPy-equivalent coefficient mutation, stored as a string.
31    pub coefficient_change: String,
32}
33
34/// One basis element T_w with its symbolic-q coefficient at a given
35/// filtration grade.
36#[derive(Serialize, Clone, Debug)]
37pub struct FiltrationTerm {
38    /// Permutation tuple (1-based) representing w ∈ S_n.
39    pub perm: Vec<usize>,
40    /// Canonical reduced word in σ-generators (1-based).
41    pub canonical_word: Vec<u32>,
42    /// Coefficient as a string (rational function in q for now;
43    /// SparsePoly Laurent later).
44    pub coefficient_in_q: String,
45}
46
47/// Full GB-filtration certificate at one Hecke element.
48/// Maps grade → list of (perm, canonical_word, coefficient).
49#[derive(Serialize, Clone, Debug, Default)]
50pub struct FiltrationCertificate {
51    pub by_grade: BTreeMap<u32, Vec<FiltrationTerm>>,
52    pub shape: BTreeMap<u32, usize>,
53    pub max_grade: u32,
54}

56/// Per-Wedderburn-block PSD report (R3 — populated by
57/// `wedderburn_psd::evaluate`).  Empty when block evaluation
58/// is deferred.
59#[derive(Serialize, Clone, Debug, Default)]
60pub struct WedderburnBlockReport {
61    pub partition: Vec<usize>,
62    pub d_lambda: usize,
63    pub matrix_at_q_0_sym_eigvals: Vec<f64>,
64    pub min_eigenvalue: f64,
65    pub max_eigenvalue: f64,
66    pub psd_symmetric_part: bool,
67}

69/// Cross-level consistency edge for tower embeddings (R4 —
70/// populated by `cross_level_embedding::verify`).
71#[derive(Serialize, Clone, Debug)]
72pub struct CrossLevelEdge {
73    pub parent: String,
74    pub constituent: String,
75    pub embedding_codomain_n_0: usize,
76    pub embedding_domain_n_0: usize,
77    pub embedding_consistent: bool,
78    pub diagnostic: String,
79}

81/// Methodology classification matching `methodology-deprecation.witness.json`.
82#[derive(Serialize, Clone, Debug)]
83pub enum Methodology {
84    #[serde(rename = "principled-derivation")]
85    PrincipledDerivation,
86    #[serde(rename = "principled-gb-nf")]
87    PrincipledGbNf,
88    #[serde(rename = "manual-historical-volume-match")]
89    ManualHistoricalVolumeMatch,
90    #[serde(rename = "magic-A-formula")]
91    MagicAFormula,
92    #[serde(rename = "recursive-on-4He")]
93    RecursiveOn4He,
94    #[serde(rename = "deprecated-volume-match")]
95    DeprecatedVolumeMatch,
96}

98/// Per-atom output of the joint-tower SDP solver.
99#[derive(Serialize, Clone, Debug)]
100pub struct JointTowerSdpCertificate {
101    pub name: String,
102    pub atomic_n_0: usize,
103    pub native_strand_count: usize,
104    pub braid_word: Vec<i32>,
105
106    pub filtration_certificate: FiltrationCertificate,
107    pub jet_log: Vec<JetEvent>,
108    pub n_steps: u64,
109    pub n_hecke_relations_fired: u64,
110
111    /// R3 output — empty if Wedderburn PSD evaluation is deferred.
112    #[serde(skip_serializing_if = "Vec::is_empty")]
113    pub wedderburn_blocks: Vec<WedderburnBlockReport>,
114
115    /// R4 output — empty if cross-level checks not run.
116    #[serde(skip_serializing_if = "Vec::is_empty")]
117    pub cross_level_edges: Vec<CrossLevelEdge>,
118
119    /// R5-minimal output: PSD-cone α* and gap.  None if not computed.
120    #[serde(skip_serializing_if = "Option::is_none")]
121    pub psd_cone_alpha_star: Option<f64>,
122    #[serde(skip_serializing_if = "Option::is_none")]
123    pub psd_cone_gap: Option<f64>,
124
125    pub rho_factor: u32,
126    /// MeV (when computed).
127    #[serde(skip_serializing_if = "Option::is_none")]
128    pub b_pred_mev: Option<f64>,
129    #[serde(skip_serializing_if = "Option::is_none")]
130    pub b_ame_mev: Option<f64>,
131    #[serde(skip_serializing_if = "Option::is_none")]
132    pub err_ppb: Option<i64>,
133
134    pub methodology: Methodology,
135}
