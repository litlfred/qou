1//! End-to-end MPFR evaluation of `tr_M(β_atomic(Z, N), q₀)`.
2//!
3//! Combines:
4//! - Tier 1.A reduce ([`crate::reduce_laurent::incremental_reduce`])
5//! - Phase-2 trace recursion ([`crate::tr_m_word_lq::tr_m_word_lq`])
6//! - MPFR-precision substitution at `q₀`
7//!
8//! into ONE Rust function callable through ONE PyO3 binding.  The
9//! caller ships `(signed_word, n_strands, q_str, dps)` and gets back
10//! a single decimal string for `mpmath.mpf` to parse on the Python
11//! side.  No per-word PyO3 calls; no Python mpmath combine loop;
12//! no Python ↔ Rust shuttling of BigInt decimal strings per
13//! coefficient.
14//!
15//! This is the user-directed "put all that in rust" / "stop back
16//! and forth rust and python of data" closure of Tier 1.A.
17//!
18//! Per CLAUDE.md §Precision goals L1 (50-dps compute floor): every
19//! intermediate quantity is exact (BigRational / BigInt) up to the
20//! final MPFR substitution; precision is set by the `dps` argument
21//! (production default 50, research-grade 100+).
22

30/// Decimal precision (`dps`) → MPFR binary precision (bits).
31///
32/// `bits = ⌈dps · log₂10⌉ + 32 guard bits` — log₂(10) ≈ 3.3219.
33/// Using `3.4` as the slope is conservative (an upper bound) but
34/// integer-truncation (`as u32`) would still under-allocate by
35/// one bit at small `dps` (e.g. `dps = 2` → `6.8` truncates to
36/// `6`, but `⌈2 · log₂10⌉ = 7`).  `ceil()` enforces the
37/// documented bound at every `dps`.
38fn dps_to_prec(dps: u32) -> u32 {
39    ((dps as f64 * 3.4).ceil() as u32) + 32
40}

42/// **Tier 1.A entry point — full pipeline in Rust + MPFR.**
43///
44/// Pipeline (all Rust, no Python crossings between stages):
45///
46/// 1. Reduce ([`reduce_laurent::incremental_reduce`]) — R1+R2+R3
47///    fixpoint with exact BigRational coefficients.
48/// 2. For each reduced σ-word:
49///    a. Evaluate the BigRational coefficient at `q` via MPFR
50///       string round-trip (preserves precision past f64 cap).
51///    b. Compute `tr_M(word)` ZHLaurent via
52///       [`tr_m_word_lq::tr_m_word_lq`] (already Rust-native).
53///    c. Evaluate the ZHLaurent at `(q, z, h)` via MPFR, where
54///       `z = 1/(q^{1/2}+q^{-1/2})`, `h = q − q^{-1}`.
55///    d. Accumulate into the running total.
56/// 3. Return `total.to_string_radix(10, Some(dps + 5))` so the
57///    Python caller can parse a single decimal string via
58///    `mpmath.mpf(result_str)`.
59///
60/// `q_str`: substrate parameter as a decimal string (avoids f64
61/// lossy conversion at the boundary; matches the existing
62/// `chi_lambda_*_mpfr` API convention).
63///
64/// `dps`: working precision in decimal places.  Production floor
65/// is 50 (CLAUDE.md §Precision goals L1); use 100+ for substrate-
66/// derivation cross-checks.
67///
68/// Returns `Err(message)` if the trace recursion hits Case C
69/// (multiple top-1 in segment) on any reduced word — caller routes
70/// to the sympy ocneanu fallback, same contract as
71/// `tr_m_word_lq::tr_m_word_lq`.
72pub fn tr_m_atomic_mpfr(
73    signed_word: &[(i8, i32)],
74    n_strands: usize,
75    q_str: &str,
76    dps: u32,
77) -> Result<String, String> {
78    let (val, _final) = tr_m_atomic_mpfr_with_state(
79        signed_word,
80        n_strands,
81        q_str,
82        dps,
83        crate::laurent_hecke_element::LaurentHeckeElement::identity(),
84    )?;
85    Ok(val)
86}

88/// **Resume-from-state variant of [`tr_m_atomic_mpfr`].**
89///
90/// Takes an `initial_state` (a pre-reduced
91/// [`crate::laurent_hecke_element::LaurentHeckeElement`]) and the
92/// `signed_word` *suffix* to apply on top of it.  Returns
93/// `(mpf_decimal_string, final_reduced_state)` so the caller can
94/// persist the final state as a cache entry for future resumes.
95///
96/// Use case (per `markov_peel_rust_bridge` prefix-cache wiring):
97/// computing `tr_M(β_{6Li})` after `β_{5Li}` is already cached.
98/// Since the 5Li signed word is a strict prefix of the 6Li one,
99/// the 5Li reduced state is the correct initial accumulator —
100/// only the extra 12 crossings need to be applied (the part where
101/// term-count growth would otherwise blow up).
102///
103/// Identity-state input (the `tr_m_atomic_mpfr` wrapper above)
104/// reproduces the from-scratch path exactly.
105pub fn tr_m_atomic_mpfr_with_state(
106    signed_word_suffix: &[(i8, i32)],
107    n_strands: usize,
108    q_str: &str,
109    dps: u32,
110    initial_state: crate::laurent_hecke_element::LaurentHeckeElement,
111) -> Result<(String, crate::laurent_hecke_element::LaurentHeckeElement), String> {
112    let reduced = reduce_laurent::incremental_reduce_from_state(
113        initial_state,
114        signed_word_suffix,
115        10_000,
116    );
117    let val = eval_reduced_to_mpfr(&reduced, n_strands, q_str, dps)?;
118    Ok((val, reduced))
119}

121/// **Rust-side cached variant of [`tr_m_atomic_mpfr`].**
122///
123/// Looks up the longest cached prefix of `signed_word` in the
124/// in-process [`atomic_reduce_cache`], resumes the reduce from
125/// that state, applies the suffix, and caches the final state
126/// under the full signed-word's hash for future calls.
127///
128/// **Zero FFI marshalling on cache hit.**  The cached
129/// [`LaurentHeckeElement`] never crosses the Python ↔ Rust
130/// boundary — Python passes only the signed word, hash + lookup
131/// happen entirely in Rust.  This is the productive variant of
132/// the prefix-cache infra (the disk-resident JSON path in
133/// `_prefix_cache.py` is a net loss per
134/// `docs/audits/2026-05-22-prefix-cache-negative-bench.md`).
135pub fn tr_m_atomic_mpfr_cached(
136    signed_word: &[(i8, i32)],
137    n_strands: usize,
138    q_str: &str,
139    dps: u32,
140) -> Result<String, String> {
141    let (prefix_len, initial_state_opt) =
142        atomic_reduce_cache::find_longest_cached_prefix(signed_word);
143    let initial = initial_state_opt.unwrap_or_else(LaurentHeckeElement::identity);
144    let suffix = &signed_word[prefix_len..];
145    let reduced =
146        reduce_laurent::incremental_reduce_from_state(initial, suffix, 10_000);
147    atomic_reduce_cache::save_prefix_state(signed_word, &reduced);
148    eval_reduced_to_mpfr(&reduced, n_strands, q_str, dps)
149}

151/// MPFR evaluation of a reduced `LaurentHeckeElement` at `q₀`.
152/// Shared by `tr_m_atomic_mpfr*` entry points (factored out so
153/// the from-scratch / resume / cached paths reuse it bit-for-bit).
154fn eval_reduced_to_mpfr(
155    reduced: &LaurentHeckeElement,
156    n_strands: usize,
157    q_str: &str,
158    dps: u32,
159) -> Result<String, String> {
160    let prec = dps_to_prec(dps);
161    let q = Float::with_val(
162        prec,
163        Float::parse(q_str)
164            .map_err(|e| format!("tr_m_atomic_mpfr: q_str parse error: {}", e))?,
165    );
166    let q_half = Float::with_val(prec, q.clone().sqrt());
167    let q_inv = Float::with_val(prec, q.clone().recip());
168    let q_half_inv = Float::with_val(prec, q_half.clone().recip());
169    // z = 1 / (q^{1/2} + q^{-1/2})
170    let z = Float::with_val(prec, Float::with_val(prec, &q_half + &q_half_inv).recip());
171    // h = q − q^{-1}
172    let h = Float::with_val(prec, &q - &q_inv);
173
174    let mut total = Float::with_val(prec, 0);
175    for (word, coef) in &reduced.terms {
176        // Skip formal-variable (negative gen) entries — they make
177        // no contribution to the Markov trace (matches the Python
178        // convention in `tr_M_element`).
179        if word.iter().any(|&g| g < 0) {
180            continue;
181        }
182        let coef_mpfr = coef.evaluate_mpfr(&q);
183        if coef_mpfr.is_zero() {
184            continue;
185        }
186        // Re-cast non-negative i32 → u32 for the trace API.  Bit-
187        // identical for non-negative values; the filter above
188        // guarantees safety.
189        let word_u32: Vec<u32> = word.iter().map(|&g| g as u32).collect();
190        let zh = tr_m_word_lq::tr_m_word_lq(&word_u32, n_strands, 200)?;
191        let mut zh_val = Float::with_val(prec, 0);
192        for (&(z_exp, h_exp), poly) in &zh.terms {
193            let c_zh = poly.evaluate_mpfr(&q);
194            let z_pow: Float = Float::with_val(prec, z.clone().pow(z_exp));
195            let h_pow: Float = Float::with_val(prec, h.clone().pow(h_exp));
196            let mut term = Float::with_val(prec, &c_zh * &z_pow);
197            term *= &h_pow;
198            zh_val += term;
199        }
200        total += Float::with_val(prec, &coef_mpfr * &zh_val);
201    }
202
203    // Emit at full precision so mpmath.mpf can reconstruct without
204    // loss.  `Some(dps + 5)` gives a small over-precision buffer
205    // so the decimal representation is unambiguous to `dps` places.
206    Ok(total.to_string_radix(10, Some((dps + 5) as usize)))
207}

211#[cfg(test)]
212mod tests {
213    use super::*;
214
215    /// `q₀` — the canonical substrate value.
216    const Q0_STR: &str = "1.10997859555418057528159407960950937799328227995870";
217
218    #[test]
219    fn proton_word_smoke() {
220        // The proton atom is a non-trivial atomic braid; here we
221        // exercise the smallest non-empty signed word — a single
222        // Sigma+ crossing at gen 0 — and verify the result is
223        // finite and parseable as a Float.
224        let signed_word = vec![(1i8, 0i32)];
225        let result_str = tr_m_atomic_mpfr(&signed_word, 3, Q0_STR, 50).unwrap();
226        let parsed = Float::parse(&result_str)
227            .expect("result string must parse as Float");
228        let _: Float = Float::with_val(180, parsed);
229    }
230
231    #[test]
232    fn empty_word_is_one() {
233        // tr_M(identity) = 1 at any q.
234        let signed_word: Vec<(i8, i32)> = vec![];
235        let result_str = tr_m_atomic_mpfr(&signed_word, 3, Q0_STR, 50).unwrap();
236        let parsed = Float::parse(&result_str).unwrap();
237        let v = Float::with_val(180, parsed);
238        let one = Float::with_val(180, 1);
239        let diff = Float::with_val(180, &v - &one);
240        let diff_f64: f64 = diff.to_f64().abs();
241        assert!(diff_f64 < 1e-40, "tr_M(identity) should be 1, got {}", result_str);
242    }
243
244    #[test]
245    fn averaged_squared_at_q0() {
246        // (σ_0 − h/2)² as derived in reduce_laurent::tests::
247        //     incremental_reduce_averaged_squared:
248        // reduces to q²/4 + 1/2 + q⁻²/4  on the empty word.
249        // Markov trace of empty word is 1, so the contribution is
250        // 1·(q²/4 + 1/2 + q⁻²/4).
251        let signed_word = vec![(0i8, 0i32), (0i8, 0i32)];
252        let result_str = tr_m_atomic_mpfr(&signed_word, 3, Q0_STR, 50).unwrap();
253        let parsed = Float::parse(&result_str).unwrap();
254        let v = Float::with_val(180, parsed);
255        // Expected: q²/4 + 1/2 + q⁻²/4 at q₀.
256        let q = Float::with_val(180, Float::parse(Q0_STR).unwrap());
257        let q_sq = Float::with_val(180, q.clone() * &q);
258        let q_inv_sq = Float::with_val(180, q_sq.clone().recip());
259        let expected = Float::with_val(
260            180,
261            Float::with_val(180, &q_sq / 4)
262                + Float::with_val(180, 1) / 2
263                + Float::with_val(180, &q_inv_sq / 4),
264        );
265        let diff = Float::with_val(180, &v - &expected);
266        let diff_f64: f64 = diff.to_f64().abs();
267        assert!(
268            diff_f64 < 1e-40,
269            "averaged-squared mismatch at q₀: got {}, expected {}",
270            result_str,
271            expected.to_string_radix(10, Some(55))
272        );
273    }
