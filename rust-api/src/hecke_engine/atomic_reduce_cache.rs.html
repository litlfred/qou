1//! Rust-side cache of reduced [`LaurentHeckeElement`]s keyed by
2//! signed-word-prefix hash.  **Zero FFI marshalling on hit.**
3//!
4//! The disk-resident JSON cache landed in
5//! `_prefix_cache.py` was empirically a net loss on the canonical
6//! sweep (12% slower at ⁶Li) because shipping the cached state
7//! Python → Rust through decimal-string FFI exceeded the saving
8//! from skipping the prefix's reduce work.  See
9//! [`docs/audits/2026-05-22-prefix-cache-negative-bench.md`].
10//!
11//! This in-process cache fixes that: when a Python caller invokes
12//! [`crate::tr_m_atomic_mpfr::tr_m_atomic_mpfr_cached`], the
13//! signed-word prefix is hashed *inside Rust*, the cached
14//! `LaurentHeckeElement` is fetched by hash (no marshalling), the
15//! suffix is reduced from that state, and the final state is
16//! re-cached under the full signed-word's hash.
17//!
18//! Cache lifetime: process-bound (no persistence across `python3`
19//! invocations).  For mass-table sweeps that compute many atoms in
20//! one process, the cumulative wall is essentially the wall of the
21//! largest atom (suffix-only reduce on each).
22
23use crate::laurent_hecke_element::LaurentHeckeElement;
24use rustc_hash::FxHasher;
25use std::collections::HashMap;
26use std::hash::Hasher;
27use std::sync::{Mutex, OnceLock};
28
29/// Per-entry memory is dominated by `LaurentHeckeElement` size,
30/// which scales as `terms × coefs_per_term`.  ⁶Li alone is ~50 MB.
31/// Cap on entry count: 64 — comfortably covers a full mass-table
32/// sweep (8 atoms × small variants) without thrashing.
33const MAX_ENTRIES: usize = 64;
34
35static REDUCE_CACHE: OnceLock<Mutex<HashMap<u64, LaurentHeckeElement>>> = OnceLock::new();
36
37fn cache() -> &'static Mutex<HashMap<u64, LaurentHeckeElement>> {
38    REDUCE_CACHE.get_or_init(|| Mutex::new(HashMap::new()))
39}
40
41/// FxHash of a signed-word prefix.  `FxHasher` is non-cryptographic
42/// but fast and well-distributed at our scale (≤ thousands of
43/// prefixes per session); 64-bit space makes collisions astronomically
44/// unlikely for the kinds of prefix sets atomic-braid sweeps produce.
45pub fn hash_prefix(signed_word_prefix: &[(i8, i32)]) -> u64 {
46    let mut h = FxHasher::default();
47    // Manually feed each (sign, gen) pair so the hash depends on the
48    // full content, not just the slice length / address.
49    for &(sign, gen) in signed_word_prefix {
50        h.write_i8(sign);
51        h.write_i32(gen);
52    }
53    h.finish()
54}
55
56/// Find the longest cached prefix of `signed_word` (possibly the
57/// full word — exact-match cache hits return `prefix_len ==
58/// signed_word.len()` with an empty suffix, which is the
59/// expected behavior when the same atom is recomputed).
60///
61/// Returns `(prefix_len, Option<state_clone>)`.  The state is
62/// cloned out of the cache because the caller will mutate it
63/// (apply the suffix); the cache itself stays warm.
64pub fn find_longest_cached_prefix(
65    signed_word: &[(i8, i32)],
66) -> (usize, Option<LaurentHeckeElement>) {
67    let g = match cache().lock() {
68        Ok(g) => g,
69        Err(_) => return (0, None),  // poisoned mutex: degrade gracefully
70    };
71    // Try prefix lengths in decreasing order so we find the LONGEST
72    // matching prefix first.  Range is `1..=len` (inclusive) so the
73    // full-word cache hit is returned when the same atom is
74    // recomputed in the same process — the resume-from-state path
75    // then runs zero crossings and just re-evaluates MPFR.
76    for prefix_len in (1..=signed_word.len()).rev() {
77        let h = hash_prefix(&signed_word[..prefix_len]);
78        if let Some(state) = g.get(&h) {
79            return (prefix_len, Some(state.clone()));
80        }
81    }
82    (0, None)
83}
84
85/// Insert or update the cache entry for `signed_word` with the
86/// reduced `state`.  Cache entries are bounded by [`MAX_ENTRIES`];
87/// when the cap is hit, the cache is fully cleared (simple, fast,
88/// adequate for the mass-table-sweep workload).
89pub fn save_prefix_state(signed_word: &[(i8, i32)], state: &LaurentHeckeElement) {
90    let h = hash_prefix(signed_word);
91    if let Ok(mut g) = cache().lock() {
92        if g.len() >= MAX_ENTRIES && !g.contains_key(&h) {
93            // Drop everything when we hit the cap.  Matches the
94            // pattern in `tr_m_word_lq::cache()` and avoids the
95            // complexity of an LRU at our scale.
96            g.clear();
97        }
98        g.insert(h, state.clone());
99    }
100}
101
102/// Empty the cache.  Call between independent sessions (or between
103/// runs that should not share state) to avoid unbounded growth.
104pub fn clear_cache() {
105    if let Ok(mut g) = cache().lock() {
106        g.clear();
107    }
108}
109
110/// Number of cached entries (for stats / debug).
111pub fn cache_size() -> usize {
112    cache().lock().map(|g| g.len()).unwrap_or(0)
113}
