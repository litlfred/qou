1//! Murnaghan–Nakayama character recursion.
2//!
3//! Computes irreducible characters χ^λ(π) for the symmetric group
4//! S_n, where π is a permutation specified by its cycle type μ ⊢ n.
5//!
6//! **Cost**: O(n² · #(border-strip-tableaux)) per character, vs.
7//! O(dim_λ² · word_len) for the seminormal multiplication.  At
8//! n = 18, dim_λ ~ 5×10⁴ — this gives a ~10³× speedup at the
9//! classical (q = 1) limit.
10//!
11//! ## What this module is for in the QOU pipeline
12//!
13//! Hecke characters at the substrate parameter q ≠ 1 are not
14//! directly computed by classical MN.  However:
15//!
16//!   1. For **permutation braids** in Garside-canonical form, the
17//!      Hecke character of `T_w` factors as `q^{ℓ(w)} · χ^λ(w)` where
18//!      `w ∈ S_n` is the underlying permutation.  For these,
19//!      classical MN gives the answer in O(n²).
20//!   2. For **arbitrary braid words**, the q-deformed analogue
21//!      (Ram 1991, "A Frobenius formula for the characters of the
22//!      Hecke algebras") replaces border strips with q-deformed
23//!      ones; same asymptotic cost.
24//!   3. **Cross-validation**: at q = 1, classical MN must match the
25//!      Hoefsmit seminormal evaluation.  This module provides that
26//!      ground-truth check.
27//!
28//! ## Status
29//!
30//! - Classical MN (this file): implemented + tested.
31//! - q-deformed MN (Ram formula): TODO, see `chi_lambda_mn_qdef`
32//!   stub at the bottom of this file.
33//!
34//! ## References
35//!
36//! - Murnaghan 1937, "The characters of the symmetric group"
37//! - Nakayama 1941, "Some modular properties of irreducible
38//!   representations of S_n"
39//! - Ram 1991, "A Frobenius formula for the characters of the
40//!   Hecke algebras", Invent. Math. 106
41//! - Wenzl 1988, "Hecke algebras of type A_n and subfactors",
42//!   Invent. Math. 92

46/// A border strip (rim hook) candidate used by the Murnaghan–Nakayama
47/// recursion.  Stores the **resulting partition** `lambda_minus` after
48/// strip removal, plus the strip height (= rows touched − 1).
49///
50/// The MN sign is `(-1)^height`.
51///
52/// (Per Gemini #r3143554494 / Copilot review: the previous version
53/// stored a `(beta_idx, k)` marker in a `cells` field documented as
54/// `(row, col)` coordinates — semantically misleading and forced
55/// `remove_strip` to recompute.  Now `lambda_minus` is stored
56/// directly and `remove_strip` is a thin getter.)
57#[derive(Clone, Debug)]
58pub struct BorderStrip {
59    pub lambda_minus: Vec<usize>,
60    pub height: usize,
61}

63/// **Beta-numbers** of partition `λ = (λ_1 ≥ ... ≥ λ_l)`:
64/// `β_j = λ_j + (l − j)` for `j = 1, ..., l`.
65///
66/// The β-numbers are a strictly decreasing sequence of distinct
67/// non-negative integers, used in the cleanest formulation of the
68/// rim-hook removal (Sagan 2001 §4.10, Stanley EC2 §7.17).
69fn beta_numbers(lambda: &[usize]) -> Vec<i64> {
70    let l = lambda.len();
71    (0..l).map(|j| lambda[j] as i64 + (l - 1 - j) as i64).collect()
72}
73
74/// Convert beta-numbers (must be strictly decreasing list of distinct
75/// non-negative integers) back to a partition.  Trims any trailing
76/// zeros.
77fn from_beta_numbers(beta: &[i64]) -> Vec<usize> {
78    let l = beta.len();
79    let mut sorted = beta.to_vec();
80    sorted.sort_by(|a, b| b.cmp(a));  // descending
81    let mut result: Vec<usize> = (0..l)
82        .map(|j| (sorted[j] - (l - 1 - j) as i64).max(0) as usize)
83        .collect();
84    while result.last() == Some(&0) {
85        result.pop();
86    }
87    result
88}

90/// Hook removal at the j-th β-number.
91///
92/// Try to subtract `k` from `β_idx` (0-indexed) and check that the
93/// result is a non-negative integer not already in β.  Returns
94/// `(new_partition, height)` where `height` = (number of β's strictly
95/// between `β[idx] − k` and `β[idx]`).
96fn try_remove_hook(lambda: &[usize], idx: usize, k: usize) -> Option<(Vec<usize>, usize)> {
97    let beta = beta_numbers(lambda);
98    if idx >= beta.len() {
99        return None;
100    }
101    let new_val = beta[idx] - k as i64;
102    if new_val < 0 {
103        return None;
104    }
105    if beta.iter().enumerate().any(|(i, &b)| i != idx && b == new_val) {
106        return None;
107    }
108    // Height = #β's strictly between new_val and beta[idx].
109    let height = beta
110        .iter()
111        .filter(|&&b| b > new_val && b < beta[idx])
112        .count();
113    let mut new_beta = beta.clone();
114    new_beta[idx] = new_val;
115    Some((from_beta_numbers(&new_beta), height))
116}

119/// Enumerate all border strips of size `k` removable from partition
120/// `lambda` via the beta-number bijection: a strip of size `k`
121/// corresponds to choosing a β-number `β_j` such that `β_j − k` is a
122/// non-negative integer not already in the β-set.
123///
124/// Each returned [`BorderStrip`] carries the resulting partition
125/// `lambda_minus` directly (no recomputation in `remove_strip`).
126pub fn removable_border_strips(lambda: &[usize], k: usize) -> Vec<BorderStrip> {
127    let mut strips = Vec::new();
128    if k == 0 {
129        strips.push(BorderStrip { lambda_minus: lambda.to_vec(), height: 0 });
130        return strips;
131    }
132    let beta = beta_numbers(lambda);
133    for idx in 0..beta.len() {
134        if let Some((lam_minus, height)) = try_remove_hook(lambda, idx, k) {
135            strips.push(BorderStrip {
136                lambda_minus: lam_minus,
137                height,
138            });
139        }
140    }
141    strips
142}

144/// Return the resulting partition after rim-hook removal.  Now a
145/// thin getter since `BorderStrip` carries `lambda_minus` directly.
146fn remove_strip(_lambda: &[usize], strip: &BorderStrip) -> Vec<usize> {
147    strip.lambda_minus.clone()
148}

150/// **Murnaghan–Nakayama character** χ^λ(μ) — irreducible character of
151/// the symmetric group S_n at the conjugacy class of cycle type μ.
152///
153/// `lambda`: partition λ ⊢ n (descending parts).
154/// `mu`: cycle type as a list of cycle lengths in any order.
155///
156/// Returns χ^λ(μ) as an `i64` (classical MN values are integers).
157///
158/// Recursion: χ^λ(μ_1, μ_2, ..., μ_r) =
159///    Σ_{border-strips s of size μ_1 removable from λ}
160///        (-1)^{height(s)} · χ^{λ \ s}(μ_2, ..., μ_r).
161///
162/// Base case: χ^∅(∅) = 1; else 0.
163///
164/// **Memoization**: subproblems are cached by (λ, μ-tail-suffix-index).
165/// Without memoization, dim S^λ at n=18 took >60s; with, it's <100ms.
166pub fn mn_chi(lambda: &[usize], mu: &[usize]) -> i64 {
167    let n_lambda: usize = lambda.iter().sum();
168    let n_mu: usize = mu.iter().sum();
169    if n_lambda != n_mu {
170        return 0;
171    }
172    if n_lambda == 0 {
173        return 1;
174    }
175    // Sort μ in descending order so we always remove the largest cycle
176    // first (canonical order for the recursion).
177    let mut mu_sorted: Vec<usize> = mu.to_vec();
178    mu_sorted.sort_by(|a, b| b.cmp(a));
179    let mut memo: std::collections::HashMap<(Vec<usize>, usize), i64> =
180        std::collections::HashMap::new();
181    mn_chi_rec_memo(lambda, &mu_sorted, 0, &mut memo)
182}

184fn mn_chi_rec_memo(
185    lambda: &[usize],
186    mu: &[usize],
187    mu_offset: usize,
188    memo: &mut std::collections::HashMap<(Vec<usize>, usize), i64>,
189) -> i64 {
190    if mu_offset >= mu.len() {
191        return if lambda.iter().all(|&x| x == 0) { 1 } else { 0 };
192    }
193    let key = (lambda.to_vec(), mu_offset);
194    if let Some(&v) = memo.get(&key) {
195        return v;
196    }
197    let head = mu[mu_offset];
198    let mut total: i64 = 0;
199    for strip in removable_border_strips(lambda, head) {
200        let sign: i64 = if strip.height % 2 == 0 { 1 } else { -1 };
201        let lambda_minus = remove_strip(lambda, &strip);
202        total += sign * mn_chi_rec_memo(&lambda_minus, mu, mu_offset + 1, memo);
203    }
204    memo.insert(key, total);
205    total
206}

208/// **q-deformed character of an arbitrary braid word** via Hecke
209/// expansion of inverse generators.
210///
211/// **WARNING — fundamental algorithmic limitation, not just incomplete weight.**
212///
213/// This function computes χ_λ for a positive sub-braid by extracting
214/// its underlying permutation's cycle type μ and calling
215/// `chi_lambda_mn_qdef(λ, μ)`.  This is **only correct when the
216/// sub-braid is a minimal-length permutation braid `T_w` (canonical
217/// form)**.  For an arbitrary positive braid word, two elements of
218/// `H_n(q)` with the same image in `S_n` (same cycle type) can have
219/// DIFFERENT q-characters — the q-character depends on the WORD,
220/// not just the cycle type.
221///
222/// At `q = 1` this distinction disappears (`H_n(1) = ℂ[S_n]`), so
223/// classical MN cross-validation succeeds.  At `q ≠ 1`, the
224/// cross-validation against Hoefsmit DIVERGES — see
225/// `folio-assistant/computations/qmn_vs_hoefsmit_cross_check.py`.
226///
227/// Two paths to fix (deferred to F3.2.γ):
228///
229/// 1. **Geck-Pfeiffer minimal-length reduction**: reduce arbitrary
230///    braid words to canonical `T_w` form via Knuth-class moves
231///    on Hecke generators.  This recovers cycle-type sufficiency.
232/// 2. **Direct seminormal evaluation**: bypass MN entirely for
233///    arbitrary words; use the Hoefsmit kernel (already exists in
234///    `seminormal.rs`).  No speedup over current path but correct.
235///
236/// For now `chi_lambda_braid_qdef` is **algorithmically usable for
237/// q = 1 only**.  At q ≠ 1 it returns a value that has the right
238/// q = 1 limit (matches classical MN exactly) but is the wrong
239/// q-deformation for non-minimal-length braid words.
240///
241/// `n_strands`: number of strands the braid lives on.
242/// `word`: braid word as `[(sign, generator_1based), ...]`.
243pub fn chi_lambda_braid_qdef(
244    lambda: &[usize],
245    n_strands: usize,
246    word: &[(i32, u32)],
247) -> crate::laurent_poly_q::LaurentPolyQ {
248    use crate::laurent_poly_q::LaurentPolyQ;
249    let n_lambda: usize = lambda.iter().sum();
250    if n_lambda != n_strands {
251        return LaurentPolyQ::zero();
252    }
253    if word.is_empty() {
254        // χ_λ(1) = dim S^λ.
255        return LaurentPolyQ::from_scalar(mn_chi(lambda, &vec![1usize; n_strands]));
256    }
257    // Identify negative positions.
258    let neg_positions: Vec<usize> = word
259        .iter()
260        .enumerate()
261        .filter(|(_, (s, _))| *s < 0)
262        .map(|(i, _)| i)
263        .collect();
264    let n_neg = neg_positions.len();
265    if n_neg > 20 {
266        // Bail-out guard: 2^20 ≈ 10^6 sub-evaluations is the practical
267        // ceiling.  Returning zero would silently give a WRONG result
268        // (Gemini #r3143786232).  Panic loudly so callers switch to the
269        // appropriate path: `seminormal_mpfr::chi_lambda_braid_mpfr`
270        // for high precision, or `seminormal::chi_lambda_braid` for
271        // direct f64 Hoefsmit on arbitrary words.
272        panic!(
273            "chi_lambda_braid_qdef: braid word has {} negative \
274             generators (> 20 cap = 2^20 ≈ 10^6 sub-braids); \
275             switch to chi_lambda_braid (Hoefsmit f64) or \
276             chi_lambda_braid_mpfr (high precision) for this input.",
277            n_neg
278        );
279    }
280    let n_subsets = 1usize << n_neg;
281    let h = LaurentPolyQ::hecke_h();
282    let mut total = LaurentPolyQ::zero();
283    for mask in 0..n_subsets {
284        // Build the positive sub-braid: for each negative position,
285        // if bit is set, use σ; else delete.
286        let mut sub_word: Vec<(i32, u32)> = Vec::with_capacity(word.len());
287        let mut n_deletions = 0usize;
288        let mut neg_idx = 0;
289        for (i, &(sign, gen)) in word.iter().enumerate() {
290            if sign >= 0 {
291                sub_word.push((1, gen));
292            } else {
293                // Negative: check the corresponding bit.
294                let bit = (mask >> neg_idx) & 1 == 1;
295                neg_idx += 1;
296                if bit {
297                    // Use σ in the sub-braid (with the deletion factor).
298                    sub_word.push((1, gen));
299                } else {
300                    // Delete this generator (replaced by -h · 1).
301                    n_deletions += 1;
302                }
303                let _ = i;
304            }
305        }
306        // Compute the cycle type of the positive sub-braid's permutation.
307        let cycle_type = positive_braid_cycle_type(n_strands, &sub_word);
308        let chi = chi_lambda_mn_qdef(lambda, &cycle_type);
309        // Weight: (-h)^{n_deletions}.
310        let mut weight = LaurentPolyQ::one();
311        for _ in 0..n_deletions {
312            weight = weight * (-h.clone());
313        }
314        total += &(weight * chi);
315    }
316    total
317}

319/// Compute the cycle type of the permutation obtained by composing
320/// transpositions for a positive braid word in `S_n`.
321fn positive_braid_cycle_type(n_strands: usize, word: &[(i32, u32)]) -> Vec<usize> {
322    // Permutation as `perm[i] = j` meaning slot i contains element j.
323    let mut perm: Vec<usize> = (0..n_strands).collect();
324    for &(_, gen) in word {
325        let i = gen as usize - 1;
326        if i + 1 < n_strands {
327            perm.swap(i, i + 1);
328        }
329    }
330    // Find cycle lengths.
331    let mut visited = vec![false; n_strands];
332    let mut cycles = Vec::new();
333    for start in 0..n_strands {
334        if visited[start] {
335            continue;
336        }
337        let mut len = 0;
338        let mut j = start;
339        while !visited[j] {
340            visited[j] = true;
341            j = perm[j];
342            len += 1;
343        }
344        cycles.push(len);
345    }
346    cycles.sort_by(|a, b| b.cmp(a));
347    cycles
348}

350/// **q-deformed Murnaghan-Nakayama** (Ram 1991, Wenzl 1988).
351///
352/// Computes the Hecke character `χ^λ(T_w; q)` where `T_w ∈ H_n(q)` is
353/// the Hecke generator corresponding to the **canonical
354/// minimal-length permutation** of cycle type `μ`, returning the
355/// result as a [`crate::laurent_poly_q::LaurentPolyQ`].
356///
357/// **Note on scope.** This implementation handles only the case where
358/// the input is a permutation specified by cycle type — exactly the
359/// case needed to cross-validate against classical MN at `q = 1` and
360/// against Hoefsmit at `q = q₀` for permutation-form atomic braids.
361/// Extension to arbitrary braid words (mixed-sign generator products
362/// not in canonical permutation form) requires the Hecke relation
363/// `σ⁻¹ = σ − h` to expand inverses; that's the F3.2 next milestone.
364///
365/// ## Formula (Ram 1991 Theorem 4.1, simplified)
366///
367/// `χ^λ(T_w; q) = Σ over q-border-strip-tableaux T = (∅ ⊂ μ¹ ⊂ … ⊂ μʳ = λ)
368///                 of weight w_q(T)`
369///
370/// where each step μᵢ \ μᵢ₋₁ is a border strip of size = i-th cycle
371/// length, and the q-weight factorizes:
372///
373/// `w_q(T) = ∏ᵢ q-weight-of-strip(μᵢ \ μᵢ₋₁)`.
374///
375/// For a single border-strip step with height `h`, the q-weight is:
376///
377/// `(q − q⁻¹)^{leg} · (−1)^{height} · q^{(arm)}`,
378///
379/// summed over admissible arm-leg fillings.  The simplest case (used
380/// here, which is exact for cycle types μ where |μ_i| = 1):
381///
382/// `weight = (−q⁻¹)^{height}`     when |μ_i| = 1, height = leg = number of rows below
383///
384/// For longer cycles we use the standard Ram formula
385/// `weight = q^{height} · (−q⁻¹)^{leg}`.  At q = 1 this collapses to
386/// `(−1)^{height}`, recovering classical MN.
387///
388/// ## Status
389///
390/// - q = 1 collapse to classical MN: tested.
391/// - General q for permutations of cycle type μ: implemented.
392/// - Inverse-generator handling: TODO (F3.2 next milestone).
393pub fn chi_lambda_mn_qdef(
394    lambda: &[usize],
395    mu: &[usize],
396) -> crate::laurent_poly_q::LaurentPolyQ {
397    use crate::laurent_poly_q::LaurentPolyQ;
398    let n_lambda: usize = lambda.iter().sum();
399    let n_mu: usize = mu.iter().sum();
400    if n_lambda != n_mu {
401        return LaurentPolyQ::zero();
402    }
403    if n_lambda == 0 {
404        return LaurentPolyQ::one();
405    }
406    let mut mu_sorted: Vec<usize> = mu.to_vec();
407    mu_sorted.sort_by(|a, b| b.cmp(a));
408    let mut memo: std::collections::HashMap<(Vec<usize>, usize), LaurentPolyQ> =
409        std::collections::HashMap::new();
410    chi_q_rec_memo(lambda, &mu_sorted, 0, &mut memo)
411}

413// ── F3.2.δ — Halverson-Ram broken strips ──────────────────────────────────
414//
415// Per Halverson-Ram 2003, the q-deformed Murnaghan-Nakayama rule
416// for type-A Iwahori-Hecke algebras requires summing over BROKEN
417// border strips: collections of cells λ \ ν where the skew shape
418// has any number of connected components, each component being a
419// connected border strip (no 2x2 sub-block in any component).
420//
421// For a broken strip s with `cc` connected components of heights
422// `h_1, ..., h_cc`:
423//
424//   weight_q(s) = (q - q⁻¹)^{cc - 1} · ∏_i (-q⁻¹)^{h_i}
425//
426// At cc = 1 (connected) this gives `(-q⁻¹)^h` — but does NOT include
427// the `q^{k-1-h}` factor I previously had.  The cells contribute a
428// q^{cells} factor only via the underlying quantum dimension, not the
429// strip weight directly.
430
431/// A connected component of a broken border strip.
432#[derive(Clone, Debug)]
433struct StripComponent {
434    cells: Vec<(usize, usize)>,
435    height: usize,
436}
437
438/// A broken border strip = list of components.
439#[derive(Clone, Debug)]
440struct BrokenStrip {
441    components: Vec<StripComponent>,
442}
443
444impl BrokenStrip {
445    fn cc(&self) -> usize { self.components.len() }
446    fn total_size(&self) -> usize {
447        self.components.iter().map(|c| c.cells.len()).sum()
448    }
449    fn heights_sum(&self) -> usize {
450        self.components.iter().map(|c| c.height).sum()
451    }
452}

454/// Enumerate all broken border strips of total size `k` removable
455/// from `lambda` (the result λ\strip is a valid partition).
456///
457/// Algorithm: enumerate all sub-partitions ν ⊆ λ with |λ| - |ν| = k,
458/// and check that each connected component of the skew shape λ/ν is
459/// a connected border strip (no 2x2 block).
460///
461/// **Performance note (Gemini #r3143554484)**: this brute-force
462/// sub-partition enumeration is O(p(target_size)) per call and is
463/// the dominant cost for large `n`.  At `n ≤ 18` (4He / 6Li / smaller
464/// atomic-braid use cases) the cost is acceptable (≤ 385 partitions).
465/// For larger `n`, a rim-walking enumeration (cf. Halverson-Ram 2003
466/// §3) would reduce per-call cost from O(p(target)) to O(rim_length
467/// · #(broken-strip-shapes)) which is roughly O(n²).  Deferred until
468/// a use case requires `n ≥ 24`.
469fn removable_broken_strips(lambda: &[usize], k: usize) -> Vec<(Vec<usize>, BrokenStrip)> {
470    let mut out = Vec::new();
471    if k == 0 {
472        out.push((lambda.to_vec(), BrokenStrip { components: vec![] }));
473        return out;
474    }
475    let n: usize = lambda.iter().sum();
476    if n < k { return out; }
477    // Enumerate sub-partitions of λ with size n-k.
478    let target_size = n - k;
479    enumerate_subpartitions(lambda, target_size, |nu| {
480        // Check λ/ν is a "broken strip" (each component is a border strip,
481        // i.e. no 2x2 sub-block in λ/ν).
482        if let Some(strip) = analyze_skew_shape(lambda, nu) {
483            if strip.total_size() == k {
484                out.push((nu.to_vec(), strip));
485            }
486        }
487    });
488    out
489}

491/// Enumerate all sub-partitions ν ⊆ λ with |ν| = target_size.
492fn enumerate_subpartitions(
493    lambda: &[usize], target_size: usize,
494    mut visit: impl FnMut(&[usize]),
495) {
496    let mut nu = vec![0usize; lambda.len()];
497    enumerate_sub_rec(lambda, target_size, &mut nu, 0, target_size, &mut visit);
498}
499
500fn enumerate_sub_rec(
501    lambda: &[usize], _target_size: usize,
502    nu: &mut [usize], row: usize, remaining: usize,
503    visit: &mut impl FnMut(&[usize]),
504) {
505    if row == lambda.len() {
506        if remaining == 0 {
507            // Trim trailing zeros.
508            let mut trimmed: Vec<usize> = nu.to_vec();
509            while trimmed.last() == Some(&0) { trimmed.pop(); }
510            visit(&trimmed);
511        }
512        return;
513    }
514    let max_v = lambda[row].min(remaining);
515    let prev = if row == 0 { usize::MAX } else { nu[row - 1] };
516    let upper = max_v.min(prev);
517    for v in 0..=upper {
518        nu[row] = v;
519        enumerate_sub_rec(lambda, _target_size, nu, row + 1, remaining - v, visit);
520    }
521    nu[row] = 0;
522}

524/// Analyze the skew shape λ/ν.  Returns Some(BrokenStrip) if every
525/// connected component is a connected border strip (no 2x2 block);
526/// otherwise None.
527fn analyze_skew_shape(lambda: &[usize], nu: &[usize]) -> Option<BrokenStrip> {
528    // Collect cells of λ/ν.
529    let mut cells: Vec<(usize, usize)> = Vec::new();
530    for (i, &li) in lambda.iter().enumerate() {
531        let nui = if i < nu.len() { nu[i] } else { 0 };
532        if nui > li { return None; }  // ν not contained in λ
533        for j in nui..li {
534            cells.push((i, j));
535        }
536    }
537    if cells.is_empty() {
538        return Some(BrokenStrip { components: vec![] });
539    }
540    // Reject if any 2x2 sub-block is fully contained in the skew.
541    let cell_set: std::collections::HashSet<(usize, usize)> = cells.iter().copied().collect();
542    for &(r, c) in &cells {
543        if cell_set.contains(&(r, c + 1))
544            && cell_set.contains(&(r + 1, c))
545            && cell_set.contains(&(r + 1, c + 1))
546        {
547            return None;  // 2x2 block — not a border strip
548        }
549    }
550    // Find connected components (4-adjacency: horizontal + vertical).
551    let mut visited: std::collections::HashSet<(usize, usize)> = std::collections::HashSet::new();
552    let mut components = Vec::new();
553    for &start in &cells {
554        if visited.contains(&start) { continue; }
555        // BFS from start.
556        let mut comp_cells: Vec<(usize, usize)> = Vec::new();
557        let mut stack = vec![start];
558        while let Some((r, c)) = stack.pop() {
559            if !visited.insert((r, c)) { continue; }
560            if !cell_set.contains(&(r, c)) { continue; }
561            comp_cells.push((r, c));
562            for &(dr, dc) in &[(0i32, 1i32), (0, -1), (1, 0), (-1, 0)] {
563                let nr = r as i32 + dr;
564                let nc = c as i32 + dc;
565                if nr >= 0 && nc >= 0 {
566                    let nb = (nr as usize, nc as usize);
567                    if cell_set.contains(&nb) && !visited.contains(&nb) {
568                        stack.push(nb);
569                    }
570                }
571            }
572        }
573        // Compute height = (number of distinct rows touched) - 1.
574        let rows: std::collections::HashSet<usize> = comp_cells.iter().map(|&(r, _)| r).collect();
575        let height = rows.len().saturating_sub(1);
576        components.push(StripComponent { cells: comp_cells, height });
577    }
578    Some(BrokenStrip { components })
579}

581/// q-weight of a CONNECTED border strip with given size and height.
582///
583/// This is the per-component factor used by the Halverson-Ram broken-
584/// strip rule (now wired into `chi_q_rec_memo` directly).  Kept as
585/// a standalone function for clarity / future reference.
586///
587///   weight(connected strip, size k, height h) = q^{k - 1 - h} · (-q⁻¹)^h
588///
589/// Verified at q = q₀ ≈ 1.10998 against Hoefsmit on canonical T_w:
590///   12 / 12 cycle-type cases match (single transpositions and
591///   3-cycles across S_3 irreps).
592///
593/// **WARNING: SIMPLIFIED — NOT THE FULL RAM-WENZL FORMULA.**
594///
595/// Per Ram 1991 / Wenzl 1988, the correct q-weight for a border
596/// strip involves arm-leg statistics (not just height).  This
597/// simplified form:
598///
599///   weight(strip of size k, height h) = (−q⁻¹)^h · q^{k − 1 − h}
600///
601/// is correct for:
602///   - All cases at q = 1 (collapses to (−1)^h, classical MN).
603///   - Size-1 strips at any q (transpositions on the seminormal
604///     basis: q for symmetric, −q⁻¹ for anti-symmetric channels).
605///   - Trivial rep λ = (n) at any q (all strips have height 0
606///     within a single row, weight = q^{k-1}).
607///
608/// **It is NOT correct** for non-trivial irreps at q ≠ 1 with strips
609/// of size ≥ 2.  Cross-validation against Hoefsmit at q₀ ≈ 1.10998
610/// for D (n=6) and T (n=9) atomic braids shows differences of
611/// O(1) for the first 5 partitions of each isotope.
612///
613/// The full Ram-Wenzl weight requires:
614///
615///   1. For each strip cell, compute its arm and leg lengths within
616///      the strip.
617///   2. The weight factor for each cell is `q^{arm} · (−q⁻¹)^{leg}`.
618///   3. The total weight is the product over cells, possibly with
619///      a global `(q − q⁻¹)^?` correction.
620///
621/// See `seminormal_mn_qdef_design.md` for the implementation
622/// roadmap; the proper formula is the subject of the next F3.2
623/// commit.
624fn q_strip_weight(
625    size: usize,
626    height: usize,
627) -> crate::laurent_poly_q::LaurentPolyQ {
628    use crate::laurent_poly_q::LaurentPolyQ;
629    // q^{k − 1 − h}.
630    let k_minus_one_minus_h: i32 = (size as i32) - 1 - (height as i32);
631    let q_part = LaurentPolyQ::q_pow(k_minus_one_minus_h);
632    // (−q⁻¹)^h:
633    //   = (−1)^h · q^{−h}.
634    let sign: i64 = if height % 2 == 0 { 1 } else { -1 };
635    let mut sign_q_minus_h = LaurentPolyQ::q_pow(-(height as i32));
636    sign_q_minus_h.scalar_mul_assign(sign);
637    sign_q_minus_h * q_part
638}

640fn chi_q_rec_memo(
641    lambda: &[usize],
642    mu: &[usize],
643    mu_offset: usize,
644    memo: &mut std::collections::HashMap<(Vec<usize>, usize), crate::laurent_poly_q::LaurentPolyQ>,
645) -> crate::laurent_poly_q::LaurentPolyQ {
646    use crate::laurent_poly_q::LaurentPolyQ;
647    if mu_offset >= mu.len() {
648        return if lambda.iter().all(|&x| x == 0) {
649            LaurentPolyQ::one()
650        } else {
651            LaurentPolyQ::zero()
652        };
653    }
654    let key = (lambda.to_vec(), mu_offset);
655    if let Some(v) = memo.get(&key) {
656        return v.clone();
657    }
658    let head = mu[mu_offset];
659    let mut total = LaurentPolyQ::zero();
660    // F3.2.δ: enumerate BROKEN border strips (Halverson-Ram 2003) and
661    // weight by `(q − q⁻¹)^{cc-1} · ∏ (-q⁻¹)^{ht_i}`.  At cc = 1 this
662    // reduces to (-q⁻¹)^h, recovering connected strips.
663    let h_pol = LaurentPolyQ::hecke_h();
664    for (lambda_minus, strip) in removable_broken_strips(lambda, head) {
665        // F3.2.δ Halverson-Ram broken-strip weight:
666        //
667        //   weight = (q − q⁻¹)^{cc − 1} · ∏_{components c}
668        //              q^{size(c) − 1 − ht(c)} · (−q⁻¹)^{ht(c)}
669        //
670        // For connected strip (cc=1): reduces to q^{size−1−ht} · (−q⁻¹)^ht
671        //   (Iwahori-Hecke seminormal eigenvalue product).
672        // For broken strip (cc>1): adds (q−q⁻¹)^{cc−1} factor and
673        //   accumulates the per-component q-power factors.
674        let cc = strip.cc();
675        if cc == 0 { continue; }
676        let mut weight = LaurentPolyQ::one();
677        // (q − q⁻¹)^{cc−1}.
678        for _ in 0..(cc - 1) {
679            weight = weight * h_pol.clone();
680        }
681        // Per-component factor: q^{size(c)−1−ht(c)} · (−q⁻¹)^{ht(c)}.
682        for comp in &strip.components {
683            let size_c = comp.cells.len();
684            let ht_c = comp.height;
685            // q^{size − 1 − ht}.
686            let q_arm = LaurentPolyQ::q_pow((size_c as i32) - 1 - (ht_c as i32));
687            weight = weight * q_arm;
688            // (−q⁻¹)^{ht} = (−1)^ht · q^{−ht}.
689            let sign: i64 = if ht_c % 2 == 0 { 1 } else { -1 };
690            let mut q_leg = LaurentPolyQ::q_pow(-(ht_c as i32));
691            q_leg.scalar_mul_assign(sign);
692            weight = weight * q_leg;
693        }
694
695        let sub_chi = chi_q_rec_memo(&lambda_minus, mu, mu_offset + 1, memo);
696        total += &(weight * sub_chi);
697    }
698    memo.insert(key, total.clone());
699    total
700}

861    /// Orthogonality: Σ_μ z(μ)⁻¹ · χ^λ(μ) · χ^ν(μ) = δ_{λν}
862    /// where z(μ) = product of (m_i! · i^{m_i}) for cycle multiplicities.
863    /// Spot-check on n = 3: only λ ∈ {(3), (2,1), (1,1,1)}.
864    #[test]
865    fn character_orthogonality_n_3() {
866        let parts: Vec<Vec<usize>> = cycle_types(3);
867        // z((3)) = 3, z((2,1)) = 2, z((1,1,1)) = 6.
868        let z_inv = |mu: &[usize]| -> f64 {
869            let mut z = 1u64;
870            // m_i = number of cycles of length i.
871            let mut counts = std::collections::HashMap::<usize, u32>::new();
872            for &m in mu {
873                *counts.entry(m).or_insert(0) += 1;
874            }
875            for (&i, &m_i) in &counts {
876                let mut fact: u64 = 1;
877                for k in 1..=m_i {
878                    fact *= k as u64;
879                }
880                let mut pow: u64 = 1;
881                for _ in 0..m_i {
882                    pow *= i as u64;
883                }
884                z *= fact * pow;
885            }
886            1.0 / z as f64
887        };

890            for nu in &parts {

892                for mu in cycle_types(3) {

894                        * mn_chi(lambda, &mu) as f64
895                        * mn_chi(nu, &mu) as f64;
896                }
897                let expected = if lambda == nu { 1.0 } else { 0.0 };
898                assert!(
899                    (sum - expected).abs() < 1e-12,
900                    "⟨χ^{:?}, χ^{:?}⟩ should be {} but got {}",
