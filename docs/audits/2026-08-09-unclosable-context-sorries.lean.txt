/-- The class constrains none of its four real fields, so the descent
    identity fails on an instance that simply picks bad numbers. -/
theorem mass_categorical_descent_unconstrained_false :
    ¬ ∀ ctx : MassCategoricalDescentContext, ctx.B = ctx.B_desc + ctx.B_trans := by
  intro h
  have := h { CategoricalObj := Unit, m_endo := id,
              B := 0, B_desc := 1, B_trans := 0, B_BW := 0 }
  norm_num at this

end QOU.ParticleInteractions

/-- `N_modes` is carried opaquely with nothing relating it to `q` or `Re_q`,
    so the cascade-depth bound fails on an instance that picks it large. -/

    ¬ ∀ ctx : QOU.FluidDynamics.QTurbulenceSpectrumContext,
        (ctx.N_modes : ℝ)
          ≤ (3 : ℝ) / 4 * Real.log ctx.Re_q / Real.log ctx.q := by

  have := h { q := 2, q_gt_one := by norm_num,
              Re_q := 2 / (2 - 1), Re_q_eq := rfl,
              N_modes := 1000000, ε := 1, ε_pos := by norm_num,
              C_q := 1, C_q_pos := by norm_num }
  norm_num at this

end RefuteProbe

/-- `Vol_41` and `δΦ` are unrelated fields: set the volume to 0 and the
    perturbation to 1. -/

    ¬ ∀ ctx : QOU.BraidKnot.VolumePerturbationContext,
        ∀ k : ℕ, 4 ≤ k →
          ctx.δΦ k = ctx.z ^ 2 * ctx.Vol_41
            * QOU.BraidKnot.qIntSym ctx.q ((k : ℤ) - 3) := by
  intro h
  have := h { q := 1, z := 1 / (Real.sqrt 1 + 1 / Real.sqrt 1), Vol_41 := 0,
              δΦ := fun _ => 1, z_def := rfl } 4 (by norm_num)
  norm_num at this

              hbar_q := 1 - (1 : ℝ)⁻¹, hbar_q_def := rfl, mass := 1 }
  norm_num at this

              A_eq := rfl, a := 0, b := 0, sign_val := 0, m_e := 0,
              Q_value := 0, S_Delta := 0, Sigma_FP := 0 }
  norm_num at this
