% ── Preliminary: rigid monoidal categories ───────────────────────────────────

\begin{definition}[Rigid monoidal category]
\label{def:rigid-monoidal-category}
A \emph{monoidal category} $(\mathcal{C}, \otimes, \mathbf{1})$ is a
category equipped with a bifunctor
$\otimes \colon \mathcal{C} \times \mathcal{C} \to \mathcal{C}$ (the
\emph{tensor product}), a unit object~$\mathbf{1}$, and natural
isomorphisms expressing associativity and unitality.

The category is \emph{rigid} if every object $V$ has a \emph{dual}
$V^{\vee}$ together with morphisms
\[
  \mathrm{ev}_V \colon V \otimes V^{\vee} \to \mathbf{1},
  \qquad
  \mathrm{coev}_V \colon \mathbf{1} \to V^{\vee} \otimes V,
\]
satisfying the snake identities
$(\mathrm{ev}_V \otimes \mathrm{id}_V) \circ
(\mathrm{id}_V \otimes \mathrm{coev}_V) = \mathrm{id}_V$
(and symmetrically for~$V^{\vee}$).

We say $\mathcal{C}$ is \emph{$R$-linear} (for a commutative ring~$R$)
if every hom-set is an $R$-module and composition is $R$-bilinear.
\end{definition}

\begin{example}[Functions and representations]
\label{ex:rigid-monoidal-vect}
The category $\mathrm{Vect}_{\mathbb{C}}$ of finite-dimensional complex
vector spaces is rigid monoidal: the tensor product is
$V \otimes W$, the unit is $\mathbf{1} = \mathbb{C}$, and the dual is
$V^{\vee} = \mathrm{Hom}(V, \mathbb{C})$ with evaluation
$\mathrm{ev}(v \otimes f) = f(v)$.

Consider the polynomial ring $\mathbb{C}[z]$ as functions on the complex
line.  The category $\mathrm{Rep}(\mathrm{SU}(2))$ of
finite-dimensional representations of $\mathrm{SU}(2)$ is a
$\mathbb{C}$-linear rigid monoidal category whose simple objects are the
spin-$j$ representations $V_j$ ($j = 0, \tfrac12, 1, \tfrac32, \ldots$,
$\dim V_j = 2j+1$).  The fundamental object is
$V = V_{1/2} \cong \mathbb{C}^2$.  The coordinate ring
$\mathcal{O}(\mathrm{SU}(2))$ --- the algebra of polynomial functions
on the group, generated by the matrix entries
$\alpha, \beta, \bar\alpha, \bar\beta$ with
$\alpha\bar\alpha + \beta\bar\beta = 1$ --- is the bridge between the
geometry and the representation theory, playing a role analogous to
$\mathbb{C}[z]$ for the affine line.
\end{example}

\begin{definition}[Quantum Universe]
\label{def:quantum-universe}
\lean{QOU.QuantumUniverse}\leanok
Let $R$ be a commutative ring.  A \emph{quantum universe over~$R$} is a
triple $(\mathcal{C}, A, \omega)$ where:
\begin{enumerate}
  \item $(\mathcal{C}, \otimes, \mathbf{1})$ is an $R$-linear rigid
        monoidal category
        (Definition~\ref{def:rigid-monoidal-category});
  \item $A$ is a \emph{Frobenius--Hopf object} in~$\mathcal{C}$: an
        object carrying both a Hopf algebra structure
        $(\mu, \eta, \Delta, \varepsilon, S)$ and a Frobenius form
        $\lambda \colon A \to \mathbf{1}$ making the pairing
        $A \otimes A \xrightarrow{\mu} A \xrightarrow{\lambda} \mathbf{1}$
        non-degenerate;
  \item $\omega \colon \mathcal{C} \to R\text{-}\mathrm{Mod}$ is a
        \emph{fibre functor}: an exact, faithful, $R$-linear monoidal
        functor;
  \item $A$ admits a \emph{differential calculus}: the module of
        K\"ahler differentials $\Omega^1(A)$ exists in~$\mathcal{C}$.
\end{enumerate}
\end{definition}

\begin{example}[The classical quantum universe]
\label{ex:quantum-universe-su2}
Take $R = \mathbb{C}$ and $G = \mathrm{SU}(2)$.  Just as
$\mathbb{C}[z]$ is the ring of polynomial functions on the affine line,
the \emph{coordinate ring} $\mathcal{O}(\mathrm{SU}(2))$ is the ring of
polynomial functions on the group.  Concretely, it is generated by four
matrix entries $\alpha, \beta, \bar\alpha, \bar\beta$ with the single
relation $\alpha\bar\alpha + \beta\bar\beta = 1$ (the determinant
condition).  The quantum universe is:
\begin{itemize}
  \item $\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$, with simple
        objects $V_j$ ($\dim V_j = 2j+1$) as in
        Example~\ref{ex:rigid-monoidal-vect};
  \item $A = \mathcal{O}(\mathrm{SU}(2))$, with Frobenius form
        $\lambda$ given by the Haar integral
        ($\lambda(f) = \int_G f\, d\mu$ is the unique bi-invariant
        integral on the group);
  \item $\omega \colon \mathrm{Rep}(\mathrm{SU}(2)) \to
        \mathrm{Vect}_{\mathbb{C}}$, the forgetful functor
        (``forget the group action, keep the vector space'').
\end{itemize}
\end{example}

\begin{definition}[Quantum Observable Universe]
\label{def:quantum-observable-universe}
\lean{QOU.QuantumObservableUniverse}\leanok
A quantum universe $(\mathcal{C}, A, \omega)$ over~$R$
(Definition~\ref{def:quantum-universe}) is \emph{observable} if:
\begin{enumerate}
  \item \emph{(non-degeneracy)} the Frobenius form $\lambda$ on $A$ is
        non-degenerate;
  \item \emph{(observability)} $\mathrm{Hom}_{\mathcal{C}}(\mathbf{1},\,
        \mathrm{Ad}(A)) = 0$, where $\mathrm{Ad}(A)$ is the adjoint object
        ($A$ acted on by conjugation); and
  \item \emph{(categorical volume constraint)}
        \label{def:categorical-volume-constraint}
        let $V$ denote the fundamental object of~$\mathcal{C}$ (the simple
        object of categorical dimension $[2]_q$); there exists a
        non-degenerate epimorphism
        \begin{equation}\label{eq:categorical-volume-constraint}
          \epsilon \colon \Lambda^2 V \;\longrightarrow\; \mathbf{1}
        \end{equation}
        preserved by the rigid dualities of~$\mathcal{C}$.
\end{enumerate}
\end{definition}

\begin{example}[The standard quantum observable universe]
\label{ex:qou-su2}
Continue from Example~\ref{ex:quantum-universe-su2}.  Write
$V = V_{1/2} \cong \mathbb{C}^2$ for the fundamental (spin-$\tfrac12$)
representation:
\begin{enumerate}
  \item \emph{Non-degeneracy.}  The Haar integral $\lambda = \int_G$ on
        $\mathcal{O}(\mathrm{SU}(2))$ is faithful: if
        $\lambda(f^* f) = 0$ then $f = 0$.
  \item \emph{Observability.}  The adjoint representation
        $\mathrm{Ad} \cong V_1$ (the spin-$1$ or $3$-dimensional
        representation) is irreducible, so it contains no copy of the
        trivial representation: $\mathrm{Hom}(\mathbf{1}, V_1) = 0$.
  \item \emph{Volume constraint.}  For $v, w \in V = \mathbb{C}^2$,
        the determinant
        $\epsilon(v \wedge w) = v_1 w_2 - v_2 w_1$
        defines a non-degenerate map
        $\epsilon \colon \Lambda^2 V \to \mathbb{C}$.
        This is the content of
        Lemma~\ref{lem:volume-from-rigidity}: the spin-$\tfrac12$
        representation is symplectically self-dual ($V \cong V^{\vee}$ via
        the alternating form $\epsilon$), which is precisely what makes
        $\mathrm{SU}$ (determinant~$1$) rather than $\mathrm{GL}$ (any
        determinant).
\end{enumerate}
\end{example}

\begin{remark}[Deformation parameter and gauge group]
\label{rem:quantum-universe-explication}
The deformation parameter~$q$ is not part of the categorical
definition; it enters via the substrate parameter
(Definition~\ref{def:substrate-parameter}) when the Hopf structure on~$A$
is specialised.  The \emph{quantum gauge group} is the affine $R$-group
scheme $G = \mathrm{Aut}^{\otimes}(\omega)$, reconstructed via
Tannaka--Krein duality
(see also Definition~\ref{def:quantum-gauge-group} for the archimedean
case).
\end{remark}

\begin{lemma}[Volume form from rigidity and symplectic self-duality]
\label{lem:volume-from-rigidity}
Let $(\mathcal{C}, \otimes, \mathbf{1})$ be an $R$-linear rigid monoidal
category, and let $V$ be an object of categorical dimension~$2$ that is
\emph{symplectically self-dual}: there exists an isomorphism
$\phi \colon V \xrightarrow{\;\sim\;} V^{\vee}$ satisfying
$\phi^{\vee} = -\phi$ (alternating).  Then the composite
\[
  \epsilon
  \;\colon\;
  V \otimes V
  \xrightarrow{\;\mathrm{id} \otimes \phi\;}
  V \otimes V^{\vee}
  \xrightarrow{\;\mathrm{ev}_V\;}
  \mathbf{1}
\]
factors through $\Lambda^2 V$ and is non-degenerate; that is,
condition~\textup{(iii)} of
Definition~\textup{\ref{def:quantum-observable-universe}} holds.
\end{lemma}

\begin{proof}
Because $\phi$ is alternating, the pairing
$B = \mathrm{ev}_V \circ (\mathrm{id} \otimes \phi)$ satisfies
$B \circ \tau_{V,V} = -B$, where $\tau_{V,V}$ is the symmetry (or
braiding) isomorphism.  Hence $B$ vanishes on symmetric tensors and
factors as $B = \epsilon \circ \pi$, where
$\pi \colon V \otimes V \twoheadrightarrow \Lambda^2 V$ is the
canonical projection.  Non-degeneracy of~$\epsilon$ follows from
non-degeneracy of~$\mathrm{ev}_V$ (guaranteed by rigidity) and the
isomorphism~$\phi$.  The duality axioms
$(\mathrm{ev}_V \otimes \mathrm{id}) \circ
(\mathrm{id} \otimes \mathrm{coev}_V) = \mathrm{id}_V$ ensure that
$\epsilon$ is preserved by the rigid dualities of~$\mathcal{C}$.
\end{proof}

\begin{theorem}[Emergence of the quantum gauge group]
\label{thm:su2-emergence}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}).  Then
$G_q = \mathrm{Aut}^{\otimes}(\omega) \cong \mathrm{SU}_q(2)$
(or $\mathrm{SL}_q(2, R)$ depending on the signature of the Frobenius
pairing).
\end{theorem}

\begin{proof}
The quantum gauge group $G_q$ is defined via Tannaka--Krein duality as
the group scheme of tensor-preserving natural automorphisms of the fibre
functor~$\omega$.  An element $g \in G_q$ assigns an invertible linear map
$g_X \colon \omega(X) \to \omega(X)$ to every object $X \in \mathcal{C}$.

Because $g$ is a natural transformation, it commutes with all morphisms
in~$\mathcal{C}$, including the volume constraint
$\epsilon \colon \Lambda^2 V \to \mathbf{1}$
\eqref{eq:categorical-volume-constraint}.  Applying~$\omega$ yields the
commutative diagram of $R$-modules
\begin{equation}\label{eq:su2-emergence-diagram}
\begin{CD}
\Lambda^2 \omega(V)
  @>{\Lambda^2 g_V}>>
  \Lambda^2 \omega(V) \\
@V{\omega(\epsilon)}VV
  @VV{\omega(\epsilon)}V \\
\omega(\mathbf{1})
  @>>{g_{\mathbf{1}} \,=\, \mathrm{id}}>
  \omega(\mathbf{1})
\end{CD}
\end{equation}
The fundamental object $V$ has categorical dimension $[2]_q$, so
$\omega(V)$ is a rank-$2$ free $R$-module and $\Lambda^2 \omega(V)$ has
rank~$1$.  The induced operator $\Lambda^2 g_V$ acts on this rank-$1$
module as scalar multiplication by $\det(g_V)$.  Following
diagram~\eqref{eq:su2-emergence-diagram} from the top-left to the
bottom-right:
\[
  \omega(\epsilon) \circ \Lambda^2 g_V
  \;=\;
  g_{\mathbf{1}} \circ \omega(\epsilon)
  \;=\;
  \omega(\epsilon).
\]
Since $\omega(\epsilon)$ is non-degenerate (hence non-zero), this forces
\begin{equation}\label{eq:det-one}
  \det(g_V) \;=\; 1,
\end{equation}
restricting the natural automorphisms from $\mathrm{GL}_q(2)$ to the
special subgroup $\mathrm{SL}_q(2)$.

The QOU observability condition ensures the Frobenius form
$\lambda \colon A \to \mathbf{1}$ is non-degenerate, inducing the
Frobenius self-duality $A \cong A^{\vee}$
(Proposition~\ref{prop:fiber-adjoint-involution}).  This rigid
self-duality equips $\mathcal{C}$ with a positive-definite
$*$-structure.  Preserving this duality forces
$g_V^{-1} = g_V^{\dagger}$, locking the group into its unitary real
form $\mathrm{SU}_q(2)$.
\end{proof}

\begin{remark}[Geometric realisation of the volume constraint]
\label{rem:volume-constraint-geometric}
In the archimedean realisation (Example~\ref{def:c-quantum-universe}),
the categorical volume constraint~\eqref{eq:categorical-volume-constraint}
is a consequence of the contact geometry.  The contact condition requires
that $d\theta|_{H \times H}$ be non-degenerate on the Levi distribution
$H = \ker\theta$; this non-degenerate alternating $2$-form is the Levi
form $L_\theta$.  Because $L_\theta$ is alternating and non-degenerate on
the rank-$4$ real distribution~$H$, it instantiates the categorical
volume form $\epsilon \colon \Lambda^2 V \to \mathbf{1}$ of
Lemma~\ref{lem:volume-from-rigidity}.  The Hodge star $*_H$ associated to
the Webster metric~$g_\theta$ then splits $\Lambda^2 H^* = \Lambda^+\!H^*
\oplus \Lambda^-\!H^*$, forcing the local structure group of the instanton
connection to be $\mathrm{SU}(2)$.
\end{remark}

\begin{definition}[Projective line object]
\label{def:projective-line-object}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over~$R$
(Definition~\ref{def:quantum-universe}) equipped with substrate
parameter~$q$ (Definition~\ref{def:substrate-parameter}), and let $V$
denote the \emph{fundamental object} of~$\mathcal{C}$ (the simple object
of categorical dimension $[2]_q$, corresponding to the spin-$\tfrac12$
representation under $\omega$).  The \emph{projective line object} is the
coset object
\[
  \mathbb{P}(V) \;=\; G_q \,/\, T,
\]
where $T = \mathrm{Aut}^{\otimes}(\omega|_{\mathcal{C}_0})$ is the maximal
torus of $G_q$ (the sub-group scheme preserving the weight decomposition).
The function algebra of $\mathbb{P}(V)$ is the $T$-coinvariant subalgebra
$\mathcal{O}_q(\mathbb{P}(V)) = A^{\mathrm{co}\,T}$.
\end{definition}

\begin{example}[Archimedean projective line]
\label{ex:projective-line-archimedean}
In the archimedean realisation
(Example~\ref{def:c-quantum-universe}), the projective line object is
$\mathbb{P}(V) \cong \mathrm{SU}(2)/\mathrm{U}(1) \cong \mathbb{CP}^1$.
\end{example}

\begin{proposition}[Fiber--adjoint-fiber involution]
\label{prop:fiber-adjoint-involution}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}).  The Frobenius form
$\lambda \colon A \to \mathbf{1}$ induces an isomorphism
$A \xrightarrow{\;\sim\;} A^{\vee}$ (the Frobenius self-duality), under
which:
\begin{enumerate}
  \item the projective line object $\mathbb{P}(V)$
        (Definition~\ref{def:projective-line-object}) embeds in the adjoint
        object $\mathrm{Ad}(A)$ as the coadjoint orbit of a highest-weight
        vector;
  \item the assignment $\sigma \colon \mathbb{P}(V) \hookrightarrow
        \mathrm{Ad}(A)$ is an involution in the sense that $\sigma$
        determines $\mathbb{P}(V)$ from $\mathrm{Ad}(A)$ and vice versa;
  \item the non-degeneracy of the Frobenius pairing forces the observability
        condition $\mathrm{Hom}(\mathbf{1},\, \mathrm{Ad}(A)) = 0$.
\end{enumerate}
In the archimedean realisation, $\sigma$ is the identification
$\mathbb{CP}^1 \cong S^2 \subset \mathfrak{su}(2)$ (the unit sphere in the
Lie algebra under the Killing form).  For topological $R$-modules of
rank~$2$, $\mathbb{P}(V) = \mathrm{Gr}(1, M)$ (rank-$1$ direct summands of
$M \cong R^2$), $\mathrm{Ad}(A) \cong \mathrm{End}_0(M) \cong
\mathfrak{sl}_2(R)$ (trace-free endomorphisms), and $\sigma$ sends a line
$\ell \subset M$ to the rank-$1$ projector
$\mathrm{proj}_\ell - \tfrac12\,\mathrm{id}_M$.
\end{proposition}

\begin{proof}
The Frobenius isomorphism $A \cong A^{\vee}$ sends $a \mapsto \lambda \circ
\mu(a \otimes -)$, which is non-degenerate by the Frobenius--Hopf axiom.
Under this identification, the coset object $\mathbb{P}(V) = G_q/T$ embeds
in $\mathrm{Ad}(A)$ via the orbit map $gT \mapsto \mathrm{Ad}(g)(h)$ for a
fixed highest-weight element $h \in \mathrm{Ad}(A)$.  The orbit is closed
(by rigidity of~$\mathcal{C}$) and determines a sub-object of
$\mathrm{Ad}(A)$ isomorphic to $\mathbb{P}(V)$.  Conversely, given the
sub-object, the coset $G_q/T$ is recovered as its stabiliser quotient.  Item
(3) follows because $\mathrm{Hom}(\mathbf{1}, \mathrm{Ad}(A))$ would provide
a $G_q$-fixed point in $\mathrm{Ad}(A)$, but the coadjoint orbit
$\mathbb{P}(V)$ is a homogeneous space with no fixed points (the $T$-action
on $\mathbb{P}(V)$ is free away from the poles, and non-degeneracy of
$\lambda$ excludes the degenerate case).
\end{proof}

\begin{definition}[Categorical connection and curvature]
\label{def:categorical-connection}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over~$R$ with
differential calculus $\Omega^1(A)$
(Definition~\ref{def:quantum-universe}).  Write
$\Omega^k(A) = \bigwedge_A^k \Omega^1(A)$ for the higher differential
forms (external tensor powers of the $A$-bimodule~$\Omega^1(A)$).

A \emph{connection} on~$A$ is a morphism
$\nabla \colon A \to A \otimes \Omega^1(A)$ in~$\mathcal{C}$ satisfying the
Leibniz rule
\[
  \nabla(a \cdot b)
  \;=\; \nabla(a) \cdot b \;+\; a \cdot \nabla(b),
  \qquad a, b \in A.
\]
The \emph{curvature} of~$\nabla$ is the composition
\[
  F_\nabla
  \;=\; \nabla^2
  \;\colon\; A \;\longrightarrow\; A \otimes \Omega^2(A),
\]
which is $A$-linear (a standard consequence of the Leibniz rule).
\end{definition}

\begin{definition}[$q$-Hodge star]
\label{def:q-hodge-star}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over~$R$ with
connection~$\nabla$ (Definition~\ref{def:categorical-connection}), and
let $q$ denote the substrate parameter
(Definition~\ref{def:substrate-parameter}).  The \emph{$q$-Hodge star}
is the involution
$\star_q \colon \Omega^k(A) \to \Omega^{n-k}(A)$ defined by
\[
  \alpha \wedge_q \star_q \beta
  \;=\; q \cdot \lambda\bigl(\alpha^* \cdot \beta\bigr)\;\mathrm{vol},
\]
where $\lambda \colon A \to \mathbf{1}$ is the Frobenius form,
$\alpha^* = S(\alpha)$ is the antipode-conjugate, and
$\mathrm{vol} \in \Omega^n(A)$ is the volume form
(the unique $G_q$-invariant top form normalised by
$\lambda(\mathrm{vol}) = 1$).
\end{definition}

\begin{remark}[Notation for the categorical action]
\label{rem:categorical-action-notation}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) equipped with
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}),
a connection $\nabla$ (Definition~\ref{def:categorical-connection}),
and a section $\psi \colon \mathbf{1} \to A$ (a ``state'').
The following notation is used in
Definition~\ref{def:categorical-action}:
\begin{itemize}
  \item $\mathrm{tr} \colon \mathrm{End}_{\mathcal{C}}(A) \to R$ is the
        \emph{categorical trace} (the composition
        $\mathrm{coev} \to \mathrm{id} \to \mathrm{ev}$ via the rigid
        structure).
  \item $R_\lambda \in \mathrm{End}_{\mathcal{C}}(A)$ is the
        \emph{Frobenius--Ricci endomorphism}: the curvature of the
        Frobenius form~$\lambda$ itself, defined by
        $R_\lambda = (\mathrm{id}_A \otimes \lambda) \circ F_{\nabla_\lambda}$,
        where $\nabla_\lambda$ is the canonical connection on~$A$ induced
        by~$\lambda$ via $\nabla_\lambda(a) = \lambda(a_{(1)}) \otimes
        a_{(2)} - a \otimes 1$.
  \item $\mathrm{tr}_A \colon
        \mathrm{End}_{\mathcal{C}}(A \otimes \Omega^{\bullet}(A)) \to
        \mathrm{End}_{\mathcal{C}}(\Omega^{\bullet}(A))$ is the
        \emph{partial trace} over the~$A$ factor.
  \item $\psi^* = \lambda \circ S(\psi) \colon A \to \mathbf{1}$ is the
        \emph{adjoint section} (antipode composed with the Frobenius form).
  \item $D_\nabla = q^{-1}\,\nabla + (1 - q^{-1})\,\iota_\xi\,\nabla$ is
        the \emph{$q$-deformed covariant derivative}, where $\xi$ is the
        Reeb derivation of~$A$
        (Definition~\ref{def:reeb-decomposition}).
\end{itemize}
\end{remark}

\begin{definition}[Categorical action]
\label{def:categorical-action}
\uses{def:quantum-universe, def:quantum-observable-universe, def:categorical-connection}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) equipped with
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}),
a connection $\nabla$ (Definition~\ref{def:categorical-connection}),
and a section $\psi \colon \mathbf{1} \to A$ (a ``state'').
The \emph{categorical action} is the $R$-valued functional
\begin{equation}\label{eq:categorical-action}
  \mathcal{S}_q[\nabla,\, \lambda,\, \psi]
  \;=\;
  \underbrace{%
    \mathrm{tr}\bigl(q \cdot R_\lambda\bigr)
  }_{\text{gravity}}
  \;+\;
  \underbrace{%
    \lambda \circ \mathrm{tr}_A\bigl(
      F_\nabla \wedge_q \star_q F_\nabla
    \bigr)
  }_{\text{gauge}}
  \;+\;
  \underbrace{%
    \lambda\bigl(\psi^* \cdot D_\nabla\,\psi\bigr)
  }_{\text{matter}},
\end{equation}
with notation as in Remark~\ref{rem:categorical-action-notation}.
\end{definition}

\begin{proposition}[Variational equations of the categorical action]
\label{prop:variational-equations}
Let $\mathcal{S}_q$ be the categorical action
(Definition~\ref{def:categorical-action}).  The three variational
equations are:
\begin{enumerate}
  \item \emph{Gravity (vary~$\lambda$)}:
    $\delta\mathcal{S}_q / \delta\lambda = 0$ yields the
    \emph{categorical Einstein equation}:
    \[
      R_\lambda - \tfrac{1}{2}\,\mathrm{tr}(R_\lambda)\,\mathrm{id}_A
      \;=\; T_q[\nabla, \psi],
    \]
    where $T_q$ is the stress-energy endomorphism constructed from the
    gauge and matter terms.
  \item \emph{Gauge (vary~$\nabla$)}:
    $\delta\mathcal{S}_q / \delta\nabla = 0$ yields the
    \emph{categorical Yang--Mills equation}:
    \[
      \nabla \star_q F_\nabla + q \cdot [F_\nabla,\, \star_q F_\nabla]
      \;=\; J_\psi,
    \]
    where $J_\psi = \psi^* \otimes D_\nabla\psi - D_\nabla\psi^* \otimes \psi$
    is the matter current.
  \item \emph{Matter (vary~$\psi$)}:
    $\delta\mathcal{S}_q / \delta\psi = 0$ yields the
    \emph{categorical transport equation}:
    \[
      D_\nabla\,\psi \;=\; 0.
    \]
\end{enumerate}
\end{proposition}

\begin{remark}[Archimedean realisation of the variational equations]
\label{rem:variational-archimedean}
In the archimedean realisation
(Example~\ref{def:c-quantum-observable-universe}):
\begin{itemize}
  \item The categorical Einstein equation reduces to
    $\mathrm{Ric}_{g_\theta} = T^{(q)}$
    (Remark~\ref{rem:gr-substrate-geometry}).
  \item The categorical transport equation reduces to the Schr\"odinger
    equation $J\hbar_q\,\iota_{\xi_q} D_A \psi = \hat{H}_q \psi$
    (Remark~\ref{rem:schrodinger-jet-coaction}).
\end{itemize}
\end{remark}

\begin{remark}[Base-ring independence]
\label{rem:base-field-independence}
Definitions~\ref{def:quantum-universe} and~\ref{def:quantum-observable-universe}
are formulated over an arbitrary commutative ring~$R$.  The Frobenius--Hopf axioms
replace the analytic vanishing conditions of the archimedean realisation
(Example~\ref{def:c-quantum-observable-universe}) with a single algebraic
non-degeneracy condition on the Frobenius form.  Concrete realisations
include:
\begin{itemize}
  \item $R = K$ an archimedean field (a field with an archimedean absolute
        value, e.g.\ $\mathbb{Q}$, $\mathbb{R}$, or $\mathbb{C}$): the
        setting of CR geometry and contact manifolds
        (Example~\ref{def:c-quantum-universe});
  \item $R = \mathbb{Q}_\pp$ (the $\pp$-adic field): the gauge datum becomes
        an $F$-isocrystal with Frobenius lift
        (Example~\ref{ex:p-adic-realisation},
        Chapter~\ref{chap:p-adic});
  \item $R = \mathbb{F}_\pp$ (a finite field): representations of a quantum
        group over~$\mathbb{F}_\pp$
        (Example~\ref{ex:finite-field-realisation});
  \item $R = \mathbb{Z}[q, q^{-1}]$: the universal (Lusztig) integral form.
\end{itemize}
The fibre functor $\omega$ ensures that $G_q = \mathrm{Aut}^{\otimes}(\omega)$
is well-defined over any~$R$.
\end{remark}

\begin{example}[Archimedean realisation: quantum universe]
\label{def:c-quantum-universe}
\lean{QOU.CRManifoldData}\leanok
Let $K$ be an archimedean field.  A quantum universe over~$K$
(Definition~\ref{def:quantum-universe}) is realised by a $5$-tuple
$(M, \theta, G, P, E)$ where:
\begin{enumerate}
  \item $M$ is a CR manifold of hypersurface type
        (real dimension $2n+1$, $n \geq 1$) with non-degenerate Levi form
        of signature $(r,s)$, $r + s = n$;
  \item $\theta$ is a pseudo-Hermitian structure (contact form) on $M$,
        determining the Levi distribution $H = \ker\theta$, the Reeb vector
        field $\xi$, the Webster metric $g_\theta$ (of signature $(2r,2s)$
        on $H$), and the Levi form $L_\theta$;
  \item $G = \mathrm{SU}(2)$;
  \item $P \to M$ is a principal $G$-bundle whose holonomy reduces to
        $\mathrm{SO}(3) \subset \mathrm{SU}(2)/\{\pm I\}$;
  \item $E = P \times_G \mathbb{CP}^1 \to M$ is the associated
        \emph{state bundle}, where $\mathrm{SU}(2)$ acts on
        $\mathbb{CP}^1$ by M\"obius transformations.
\end{enumerate}
The monoidal category is $\mathcal{C} = \mathrm{Rep}(\mathrm{SU}_{q(p)}(2))$
(the representation category of the quantum group at the substrate
parameter~$q(p)$, Definition~\ref{def:substrate-parameter}), the
Frobenius--Hopf object is
$A = \mathcal{O}_q(\mathrm{SU}(2))$ with Frobenius form given by the
Haar state, and the fibre functor $\omega$ is the forgetful functor
to $K\text{-}\mathrm{Mod}$.
\end{example}

\begin{remark}[Notation and special cases]
\label{rem:notation-special-cases}
When $\theta$ is understood from context, we abbreviate the $5$-tuple to
$(M, G, P, E)$.  The \emph{strictly pseudoconvex} case corresponds to
signature $(n,0)$, where $L_\theta$ is positive definite.
\end{remark}

\begin{example}[Archimedean realisation: quantum observable universe]
\label{def:c-quantum-observable-universe}
\lean{QOU.CQuantumObservableUniverse}\leanok
An \emph{archimedean quantum observable universe} is an archimedean quantum
universe $(M, \theta, G, P, E)$
(Example~\ref{def:c-quantum-universe}) that satisfies
the vanishing conditions
\[
  H^0(M, \mathrm{Ad}(P)) = 0
  \qquad\text{and}\qquad
  \Gamma_c(M, E) = 0.
\]
These analytic conditions are the archimedean realisation of the
categorical observability conditions of
Definition~\ref{def:quantum-observable-universe}: the Frobenius
non-degeneracy of the Haar state on $\mathcal{O}_q(\mathrm{SU}(2))$,
combined with irreducibility of the instanton connection
(Lemma~\ref{lem:irreducible-connection}), forces
$H^0(M, \mathrm{Ad}(P)) = 0$; the section-vanishing
$\Gamma_c(M, E) = 0$ follows from the non-existence of
$A$-equivariant sections of the associated $\mathbb{CP}^1$-bundle.
\end{example}

\begin{definition}[Instanton connection]
\label{def:instanton}
\lean{QOU.InstantonConnection}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum universe with
$\dim_{\mathbb{R}} H = 4$ ($n = 2$).  An \emph{instanton connection} on $P$
is a connection $A$ whose curvature $F_A$, restricted to the horizontal
distribution $H = \ker\theta$, is self-dual:
$F_A|_H = {*_H}\,F_A|_H$, where $*_H$ is the Hodge star on
$\Lambda^2 H^*$ determined by the Webster metric $g_\theta$.  More generally,
when $\dim_{\mathbb{R}} H \neq 4$, an instanton connection is a
Yang--Mills connection that minimises the functional
$\int_M \lVert F_A|_H \rVert^2\,\theta \wedge (d\theta)^n$.
\end{definition}

\begin{lemma}[Irreducibility of the connection]
\label{lem:irreducible-connection}
\lean{QOU.irreducible_connection}\leanok
Let $(M, G, P, E)$ be an archimedean quantum observable universe.  Then the
vanishing condition $H^0(M, \mathrm{Ad}(P)) = 0$ implies that every
instanton connection $A$ on $P$ (Definition~\ref{def:instanton}) is
irreducible.
\end{lemma}

\begin{proof}
Suppose for contradiction that an instanton connection $A$ on $P$ is
reducible.  Then the stabiliser of $A$ under the gauge group contains a
non-trivial element $s \in \Gamma(M,\, \mathrm{Ad}(P))$ satisfying
$D_A s = 0$.  Such $s$ is a global parallel section of $\mathrm{Ad}(P)$,
and hence a non-zero element of $H^0(M, \mathrm{Ad}(P))$, contradicting
$H^0(M, \mathrm{Ad}(P)) = 0$.  Hence $A$ is irreducible.
\end{proof}

\begin{proposition}[Self-duality of the curvature]
\label{prop:self-dual-curvature}
\lean{QOU.self_dual_curvature}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with $\dim_{\mathbb{R}} M = 5$ (i.e.\ the CR dimension $n = 2$, so that the
Levi distribution $H = \ker\theta$ has $\dim_{\mathbb{R}} H = 4$).  Let $A$
be an instanton connection on $P$.  Then the curvature $F_A$, restricted to
$H$, is self-dual with respect to the Hodge star $*_H$ determined by the
Webster metric $g_\theta$ and the complex structure $J$:
\[
  F_A\big|_H \;=\; {*_H}\, F_A\big|_H.
\]
\end{proposition}

\begin{proof}
The contact form $\theta$ (Example~\ref{def:c-quantum-universe}) splits
the tangent bundle as $TM = H \oplus \mathbb{R}\xi$, where
$H = \ker\theta$ has real dimension $4$.  The curvature $F_A$ is an
$\mathrm{Ad}(P)$-valued $2$-form on $M$.  By the Reeb decomposition
(Definition~\ref{def:reeb-decomposition}), $F_A$ splits into horizontal and
vertical parts; we consider $F_A|_H \in \Gamma(\Lambda^2 H^* \otimes
\mathrm{Ad}(P))$.

Since $\dim_{\mathbb{R}} H = 4$, the operator $*_H$ maps $2$-forms to
$2$-forms and satisfies $*_H^2 = \mathrm{id}$ on $\Lambda^2 H^*$.  This
decomposes $\Lambda^2 H^* = \Lambda^+\! H^* \oplus \Lambda^-\! H^*$ into
self-dual and anti-self-dual parts.

An instanton connection $A$ minimises the Yang--Mills functional.  By the
Bogomolny argument, the integrated norm satisfies
\[
  \int_M \lVert F_A|_H \rVert^2 \,\theta \wedge (d\theta)^2
  \;=\; 2\!\int_M \lVert F_A^-|_H \rVert^2 \,\theta \wedge (d\theta)^2
        + \int_M \mathrm{tr}(F_A|_H \wedge F_A|_H)
          \wedge \theta,
\]
where the topological term
$\int_M \mathrm{tr}(F_A|_H \wedge F_A|_H) \wedge \theta$ is fixed by the
characteristic class of $P$.  Minimising the functional forces
$F_A^-|_H = 0$, which is precisely the self-duality condition
$F_A|_H = {*_H}\,F_A|_H$.
\end{proof}

\section{Substrate parameter and deformation}
\label{sec:substrate-parameter}

\begin{definition}[Substrate parameter]
\label{def:substrate-parameter}
\lean{QOU.SubstrateParameter}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over a ring~$R$
(Definition~\ref{def:quantum-universe}).  Let
$\mathcal{U}^{\mathbb{Z}}_q(\mathfrak{su}(2))$ denote the
\emph{Lusztig integral form}~\cite{lusztig1993}: the
$\mathbb{Z}[q,q^{-1}]$-subalgebra of $\mathcal{U}_q(\mathfrak{su}(2))$
generated by $K^{\pm 1}$ and the divided powers
$E^{(n)} = E^n/[n]_q!$, $F^{(n)} = F^n/[n]_q!$.
The \emph{substrate parameter} is a ring homomorphism
\[
  \iota \;\colon\; \mathbb{Z}[q, q^{-1}] \;\longrightarrow\; k,
  \qquad q \;\longmapsto\; q_0 \in k^{\times},
\]
and the base-changed algebra
$\mathcal{U}^{\mathbb{Z}}_q(\mathfrak{su}(2))
  \otimes_{\mathbb{Z}[q,q^{-1}]} k$
is the quantised enveloping algebra at the specialised value~$q_0$.
When the quantum universe is fibred over a space~$M$ (as in the
archimedean realisation, Example~\ref{def:c-quantum-universe}), $q_0$
may vary from point to point, yielding a \emph{substrate field}
$q \colon M \to k^{\times}$.
\end{definition}

\begin{remark}[Archimedean substrate via the Levi form]
\label{rem:c-substrate-via-levi}
In the archimedean quantum observable universe
(Example~\ref{def:c-quantum-observable-universe}), let $A$ be an
instanton connection on~$P$
(Definition~\ref{def:instanton}), $F_A^+$ its self-dual curvature
(Proposition~\ref{prop:self-dual-curvature}), and
$L_\theta \colon H \times H \to \mathbb{C}$ the Levi form on
$H = \ker\theta$, defined by
$L_\theta(V, W) = -i\,d\theta(V, \bar{W})$.
The specialisation map of
Definition~\ref{def:substrate-parameter} sends
\[
  q \;\longmapsto\;
  q(p) \;=\; 1 + \lvert\langle L_\theta,\, F_A^+ \rangle\rvert(p),
  \qquad p \in M,
\]
where $\langle L_\theta, F_A^+\rangle$ is the natural contraction
(Remark~\ref{rem:substrate-contraction}) and $\lvert\cdot\rvert$ denotes
the absolute value.  This defines a smooth field
$q \colon M \to [1,\infty) \subset \mathbb{C}^{\times}$ with $q(p) = 1$
at points where $F_A^+ = 0$ (classicality) and $q(p) \to \infty$ in the
deep-quantum regime.  The non-archimedean analogues are developed in
Chapter~\ref{chap:p-adic}: over~$\mathbb{Q}_\pp$ the Levi-form
contraction is replaced by the Frobenius endomorphism and
$q = \pp^{\,-v_\pp(\det\varphi)}$
(Example~\ref{ex:p-adic-realisation}); over~$\mathbb{F}_\pp$, $q$
is a primitive root of unity
(Example~\ref{ex:finite-field-realisation}).
\end{remark}

\begin{remark}[Algebraic vs.\ exponential specialisation]
\label{rem:substrate-exponential}
Over~$\mathbb{Q}$ (or any subfield of~$\mathbb{C}$), the classical
exponential $\exp(\lvert\langle L_\theta, F_A^+\rangle\rvert)$ is an
equally valid specialisation map.  The formula
$q(p) = 1 + \lvert\langle L_\theta, F_A^+\rangle\rvert(p)$
(Remark~\ref{rem:c-substrate-via-levi}) is the first-order approximation
of $\exp\!\bigl(\lvert\langle L_\theta, F_A^+\rangle\rvert(p)\bigr)$.
Both define smooth fields $M \to [1,\infty)$ with the same boundary
behaviour ($q = 1$ at classicality, $q \to \infty$ deep-quantum), and
all downstream formulas ($\hbar_q = 1 - q^{-1}$, the Heisenberg
commutator, the Schr\"odinger equation, the Yang--Mills functional) are
algebraic in~$q$ and therefore insensitive to the choice between $1+x$
and $\exp(x)$.  The algebraic form is adopted in
Definition~\ref{def:substrate-parameter} because it is meaningful over
every base ring~$R$, whereas $\exp(x)$ requires an archimedean topology.
\end{remark}

\begin{remark}[Natural contraction and field character of $q$]
\label{rem:substrate-contraction}
The Levi form $L_\theta$ is a section of $\Lambda^{1,1} H^*$
(the $(1,1)$-component of $\Lambda^2 H^*$ with respect to the complex
structure $J$).  The self-dual curvature $F_A^+$ is a section of
$\Lambda^+ H^* \otimes \mathrm{Ad}(P)$.  On a $4$-dimensional oriented
inner-product space, $\Lambda^{1,1} \subset \Lambda^+$, so $L_\theta$ and
$F_A^+$ live in the same subbundle.  The Webster metric $g_\theta$
provides a fibrewise inner product on $\Lambda^+ H^*$; composing this
with the Killing form on $\mathrm{Ad}(P)$ yields the natural contraction
\[
  \langle L_\theta,\, F_A^+ \rangle_p
  \;\in\; \mathbb{R}, \qquad p \in M.
\]
The substrate parameter is not a
fixed constant but a field on $M$: its value varies from point to point
according to the local coupling between the Levi form and the self-dual
curvature.
\end{remark}

\begin{definition}[Quantum Gauge Group]
\label{def:quantum-gauge-group}
\lean{QOU.QuantumGaugeGroup}\leanok
Let $(M, G, P, E)$ be an archimedean quantum observable universe with substrate
parameter $q$ (Definition~\ref{def:substrate-parameter}).  The
\emph{quantum gauge group} is the field of quantum groups
$\{\mathrm{SU}_{q(p)}(2)\}_{p \in M}$ obtained by deforming the structure
group $G = \mathrm{SU}(2)$ pointwise at the parameter $q(p)$.
\end{definition}

\begin{definition}[Quantum Planck Constant]
\label{def:quantum-planck-constant}
\lean{QOU.quantumPlanckConstant}\leanok
Let $(M, G, P, E)$ be an archimedean quantum observable universe with substrate
parameter $q$ (Definition~\ref{def:substrate-parameter}).  The
\emph{quantum Planck constant} is the smooth field
$\hbar_q \colon M \to [0,1)$ defined by
\[
  \hbar_q(p) \;=\; 1 - q(p)^{-1}, \qquad p \in M.
\]
\end{definition}

\begin{remark}[Limiting regimes of $\hbar_q$]
\label{rem:planck-limits}
Like $q$ itself, $\hbar_q$ is not a universal constant but a field on $M$.
As $q(p) \to 1$ (classicality), $\hbar_q(p) \to 0$ and quantum effects
vanish; as $q(p) \to \infty$ (deep-quantum regime), $\hbar_q(p) \to 1$.
\end{remark}

\begin{remark}[Heisenberg uncertainty as commutator]
\label{rem:heisenberg-commutator}
In the archimedean quantum observable universe the Heisenberg uncertainty
principle is expressed through the $q$-deformed commutator.  Let $\hat{X}$ and
$\hat{P}$ denote position and momentum observables on the fibre
$\mathbb{CP}^1$.  Let $J \colon H \to H$ be the complex structure operator
on the Levi distribution ($J^2 = -\mathrm{Id}$); throughout this paper every
occurrence of the imaginary unit~$i$ in a physical equation is identified with
the geometric action of~$J$ on horizontal planes.  We define their
$q$-dependent commutator at $p \in M$ by
\[
  [\hat{X},\, \hat{P}]_q \;:=\; J\,\hbar_q(p),
\]
where $\hbar_q(p) = 1 - q(p)^{-1}$
(Definition~\ref{def:quantum-planck-constant}).  In this remark the subscript
$q$ serves only to record the dependence of the commutator on the substrate
field $q$ and should not be confused with any $q$-deformed Lie bracket
$[\cdot,\cdot]_q$ on vector fields that may appear elsewhere.  The commutator
encodes the local uncertainty: when $q(p) \to 1$ (classicality),
$[\hat{X}, \hat{P}]_q \to 0$ and observables commute; when $q(p) \to \infty$
(deep-quantum regime), $[\hat{X}, \hat{P}]_q \to J$ and maximal uncertainty
prevails.
\end{remark}

\begin{remark}[Commutator and the Lie bracket analogy]
\label{rem:commutator-lie-analogy}
The position--momentum commutator $[\hat{X},\hat{P}]_q$
(Remark~\ref{rem:heisenberg-commutator}) is the quantum-mechanical
counterpart of the Lie bracket
$[e_a,e_b]=\epsilon_{abc}\,e_c$ in $\mathfrak{su}(2)$
(Definition~\ref{def:su2-basis}).  Both commutators measure the
first-order non-commutativity of their respective algebras
(Remark~\ref{rem:first-order-taylor}).  In the braid-group setting,
this non-commutativity appears as the defining relation
$\sigma_1\sigma_2\sigma_1=\sigma_2\sigma_1\sigma_2$
(Definition~\ref{def:braid-group}); on the Bring surface, it is
realised by the $S_3$ permutation action on the spectral resolvents
(Remark~\ref{rem:braid-group-action}).
\end{remark}

\begin{remark}[Commutator as Reeb vector]
\label{rem:commutator-reeb-vector}
The commutator $[\hat{X},\hat{P}]_q$
(Remark~\ref{rem:heisenberg-commutator}) may be identified with the
value of the $q$-deformed Reeb vector field~$\xi_q$
(Definition~\ref{def:q-reeb-vector-field}) evaluated on the contact
form.  Concretely, the Reeb field is the unique direction along which
the symplectic pairing $d\theta$ degenerates, so $\theta(\xi_q)$
encodes precisely the obstruction to commutativity of canonical
coordinates.  On the exceptional divisor $\mathbb{CP}^1$ produced by
blowing up a spacetime defect on the Bring surface
(Remark~\ref{rem:brings-smoothness}), the Reeb vector field generates
the $U(1)$ fibration of the Hopf bundle $S^3\to\mathbb{CP}^1$, and
its ``strength'' is set by~$\hbar_q(p)$.
\end{remark}

\begin{definition}[$q$-Harmonic form (Particle)]
\label{def:q-harmonic-form}
\lean{QOU.QHarmonicForm}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe with
substrate parameter $q$ (Definition~\ref{def:substrate-parameter}).  Let $d_q$
denote the $q$-deformed exterior derivative
(Definition~\ref{def:q-ext-deriv} below) and let $\delta_q$ denote its formal
adjoint with respect to the inner product on
$\Gamma(M,\, \Lambda^k T^*M \otimes J^1 E)$ induced by the Webster metric
$g_\theta$.  A \emph{particle} is a section
$\alpha \in \Gamma(M,\, \Lambda^k T^*M \otimes J^1 E)$ that is
$q$-harmonic, i.e.\ lies in the kernel of the $q$-Laplacian:
\[
  \Delta_q \alpha
  \;=\; (d_q\,\delta_q + \delta_q\,d_q)\,\alpha
  \;=\; 0.
\]
\end{definition}

\begin{definition}[Reeb decomposition]
\label{def:reeb-decomposition}
\lean{QOU.ReebDecomposition}\leanok
Let $(M, \theta)$ be a CR manifold with non-degenerate Levi form
(Example~\ref{def:c-quantum-universe}) and let $\xi$ denote the
associated \emph{Reeb vector field}, characterised by
$\iota_\xi\,d\theta = 0$ and $\theta(\xi) = 1$.  Every differential
$k$-form $\alpha$ ($k \geq 1$) on $M$ decomposes uniquely as
\[
  \alpha \;=\; \alpha_H + \theta \wedge \iota_\xi\,\alpha,
\]
where $\alpha_H := \alpha - \theta \wedge \iota_\xi\,\alpha$ is the
\emph{horizontal component} (satisfying $\iota_\xi\,\alpha_H = 0$) and
$\alpha_V := \theta \wedge \iota_\xi\,\alpha$ is the \emph{vertical
component}.  A form is \emph{purely horizontal} if and only if
$\iota_\xi\,\alpha = 0$.
\end{definition}

\begin{definition}[Fermion]
\label{def:fermion}
\lean{QOU.Fermion}\leanok
A \emph{fermion} is a $q$-harmonic form
(Definition~\ref{def:q-harmonic-form}) that has a non-trivial vertical
component in the Reeb decomposition
(Definition~\ref{def:reeb-decomposition}):
\[
  \Delta_q\,\alpha = 0
  \qquad\text{and}\qquad
  \iota_\xi\,\alpha_V \neq 0.
\]
\end{definition}

\begin{definition}[Boson]
\label{def:boson}
\lean{QOU.Boson}\leanok
A \emph{boson} is a $q$-harmonic form
(Definition~\ref{def:q-harmonic-form}) that is purely horizontal in the
Reeb decomposition (Definition~\ref{def:reeb-decomposition}):
\[
  \Delta_q\,\alpha = 0
  \qquad\text{and}\qquad
  \alpha = \alpha_H
  \quad(\text{equivalently, } \iota_\xi\,\alpha = 0
        \text{ by Definition~\ref{def:reeb-decomposition}}).
\]
\end{definition}

\begin{definition}[Observation]
\label{def:observation}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe over a
ring~$R$ (Definition~\ref{def:quantum-observable-universe}), and let $q$
denote the substrate parameter
(Definition~\ref{def:substrate-parameter}).  An
\emph{observation} is a morphism
\[
  \mathcal{O} \;\colon\; A \;\longrightarrow\; A
  \qquad\text{in } \mathcal{C},
\]
satisfying:
\begin{enumerate}
  \item \textbf{Frobenius compatibility.}\; $\lambda \circ \mathcal{O}
        = \lambda$ (the observation preserves the Frobenius form);
  \item \textbf{Substrate perturbation.}\; the induced map on the
        deformation parameter shifts $q \mapsto q' = q + \delta q$, where
        $\delta q$ is determined by the Hopf-algebra coaction of~$A$ on
        itself:
        \[
          \delta q \;=\; \lambda\bigl(\mathcal{O}(a_{(1)})\,S(a_{(2)})\bigr)
          \quad\text{for } a \in A,
        \]
        with $S$ the antipode and $\Delta(a) = a_{(1)} \otimes a_{(2)}$
        the coproduct.
\end{enumerate}
\end{definition}

\begin{remark}[Archimedean realisation of observation]
\label{rem:observation-archimedean}
In the archimedean realisation
(Example~\ref{def:c-quantum-observable-universe}), an observation
(Definition~\ref{def:observation}) is realised as a continuous open map
from the spacetime base~$M$ to the bundle of endomorphisms
$\operatorname{End}(E) \to M$, encoding the coordinate change in torsion
energy of the bundle~$E$.  Because the substrate parameter~$q(p)$
(Definition~\ref{def:substrate-parameter}) is perturbed by
condition~(2), every observation shifts the non-commutative geometry:
observing the universe changes it.
\end{remark}

\begin{remark}[Two-observer parallelogram]
\label{rem:two-observer-parallelogram}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe and
let $q$ denote the substrate parameter.  Let
$\alpha$ be a fermion (Definition~\ref{def:fermion}) --- a
$q$-harmonic form with non-trivial vertical component
$\iota_\xi\,\alpha_V \neq 0$.  Consider two independent observations
$\mathcal{O}_1, \mathcal{O}_2 \colon A \to A$
(Definition~\ref{def:observation}) of the same event.  These data
assemble into a \emph{parallelogram of observations}:
\[
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  & \mathcal{O}_1(\alpha) \arrow[dr, "\text{compare}"] & \\
  \alpha \arrow[ur, "\mathcal{O}_1"]
         \arrow[dr, "\mathcal{O}_2"']
  & & \delta\alpha \\
  & \mathcal{O}_2(\alpha) \arrow[ur, "\text{compare}"'] &
\end{tikzcd}
\]
The fermion $\alpha$ is the \emph{source vertex}: it is the
physical state prior to measurement, carrying a non-trivial vertical
(Reeb) component that couples it to the spin direction.  From the
source, the two observers independently apply their observation
morphisms, producing $\mathcal{O}_1(\alpha)$ and
$\mathcal{O}_2(\alpha)$ at the lateral vertices of the parallelogram.
The two observed values are then compared at the \emph{far vertex},
yielding the discrepancy
\[
  \delta\alpha
  \;=\; \mathcal{O}_1(\alpha) - \mathcal{O}_2(\alpha).
\]

\textbf{Relation to uncertainty.}\;
Each observation perturbs the substrate parameter by $\delta q_i$
(Definition~\ref{def:observation}, condition~(2)), so the two
observers induce distinct shifted parameters $q_1' = q + \delta q_1$
and $q_2' = q + \delta q_2$.  The discrepancy $\delta\alpha$ is
controlled by the difference in these perturbations.  In the
archimedean realisation, the Heisenberg commutator
(Remark~\ref{rem:heisenberg-commutator}) gives
\[
  \lVert \delta\alpha \rVert
  \;\geq\;
  \bigl\lvert \hbar_{q_1'}(p) - \hbar_{q_2'}(p) \bigr\rvert
  \;=\;
  \bigl\lvert q_1'(p)^{-1} - q_2'(p)^{-1} \bigr\rvert,
\]
where $\hbar_q = 1 - q^{-1}$
(Definition~\ref{def:quantum-planck-constant}).  Thus the observed
discrepancy at the far vertex of the parallelogram is bounded below
by the difference of the local Planck constants induced by the two
observations.

The parallelogram is \emph{degenerate} (i.e.\ $\delta\alpha = 0$)
only when both observations perturb the substrate identically
($\delta q_1 = \delta q_2$), which requires the two observation
morphisms to agree on the adjoint coaction of~$A$.  Generically, the
non-commutativity of the Hopf structure prevents this, and the
parallelogram opens: independent observations of a fermion
necessarily disagree, with the magnitude of disagreement governed by
the uncertainty encoded in the substrate field.
\end{remark}

\begin{definition}[$q$-Deformed exterior derivative]
\label{def:q-ext-deriv}
\lean{QOU.qExtDeriv}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with substrate parameter $q$ (Definition~\ref{def:substrate-parameter}).
The \emph{$q$-deformed exterior derivative}
$d_q \colon \Omega^k(M) \to \Omega^{k+1}(M)$ is defined on $k$-forms by
\[
  d_q\,\alpha
  \;=\; q^{-1}\,d\alpha
        + (1 - q^{-1})\,\theta \wedge \mathcal{L}_\xi\,\alpha,
\]
where $d$ is the ordinary exterior derivative, $\theta$ the contact form,
and $\mathcal{L}_\xi$ the Lie derivative along the Reeb field.
\end{definition}

\begin{definition}[$q$-Deformed Lie bracket]
\label{def:q-lie-bracket}
\lean{QOU.qLieBracket}\leanok
Let $V, W \in \mathfrak{X}(M)$ be smooth vector fields.  The
\emph{$q$-deformed Lie bracket} is
\[
  [V, W]_q
  \;=\; q^{-1}\,[V, W]
        + (1 - q^{-1})\,\bigl(\theta(V)\,\mathcal{L}_\xi W
                              - \theta(W)\,\mathcal{L}_\xi V\bigr),
\]
where $[\,\cdot\,,\cdot\,]$ is the standard Lie bracket on $\mathfrak{X}(M)$.
\end{definition}

\begin{definition}[$q$-Deformed connection]
\label{def:q-connection}
\lean{QOU.qConnection}\leanok
Let $A$ be an instanton connection on $P$
(Lemma~\ref{lem:irreducible-connection}).  The \emph{$q$-deformed
connection} $A_q$ is defined by
\[
  A_q
  \;=\; q^{-1}\,A
        + (1 - q^{-1})\,\iota_\xi\,A\;\theta,
\]
\end{definition}

\begin{definition}[$q$-Deformed Reeb vector field]
\label{def:q-reeb-vector-field}
\lean{QOU.qReebVectorField}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with substrate parameter $q$ (Definition~\ref{def:substrate-parameter})
and Reeb vector field $\xi$ (Definition~\ref{def:reeb-decomposition}).  The
\emph{$q$-deformed Reeb vector field} is the vector field $\xi_q$ on $M$
defined by its action on smooth functions $f \in C^\infty(M)$:
\[
  \xi_q(f) \;=\; q^{-1}\,\xi(f) + (1 - q^{-1})\,[\xi, X_f]_q,
\]
where $X_f$ is the horizontal lift of $\mathrm{grad}_{g_\theta}\,f$ to
$H = \ker\theta$ (using the Webster metric $g_\theta$) and
$[\,\cdot\,,\cdot\,]_q$ is the $q$-deformed Lie bracket
(Definition~\ref{def:q-lie-bracket}).
\end{definition}

\begin{definition}[$q$-Deformed Reeb flow]
\label{def:q-reeb-flow}
\lean{QOU.QReebFlow}\leanok
Let $\xi_q$ be the $q$-deformed Reeb vector field
(Definition~\ref{def:q-reeb-vector-field}).  The \emph{$q$-deformed Reeb flow}
is the one-parameter family of diffeomorphisms
$\phi_t^q \colon M \to M$, $t \in \mathbb{R}$, generated by $\xi_q$:
\[
  \frac{d}{dt}\,\phi_t^q \;=\; \xi_q \circ \phi_t^q,
  \qquad \phi_0^q = \mathrm{id}_M.
\]
\end{definition}

\begin{definition}[$q$-Torsion leaf]
\label{def:q-torsion-leaf}
\lean{QOU.QTorsionLeaf}\leanok
Let $H = \ker\theta \subset TM$ be the horizontal distribution on the
CR manifold $M$.  A \emph{$q$-torsion leaf} is a maximal integral
submanifold $L \hookrightarrow M$ of $H$ that is invariant under the
$q$-deformed Reeb flow (Definition~\ref{def:q-reeb-flow}):
\[
  T_p L \;\subset\; H_p
  \quad\text{for all } p \in L,
  \qquad\text{and}\qquad
  \phi_t^q(L) = L
  \quad\text{for all } t \in \mathbb{R}.
\]
\end{definition}

\begin{remark}[$q$-Torsion leaves: existence and Frobenius obstruction]
\label{rem:q-torsion-existence}
The horizontal distribution $H = \ker\theta$ on a contact manifold is
maximally non-integrable: $d\theta|_{H \times H}$ is non-degenerate.
By the Frobenius theorem, maximal integral submanifolds of $H$ have
dimension at most~$n$ (half the rank of~$H$); these are the
\emph{Legendrian submanifolds}.  Definition~\ref{def:q-torsion-leaf}
therefore requires $L$ to be a Legendrian submanifold invariant under the
$q$-deformed Reeb flow.

Existence is not automatic: it depends on the dynamics of~$\xi_q$.
When $q \to 1$ and $\xi_q \to \xi$, invariant Legendrian submanifolds
exist whenever the Reeb flow is periodic or integrable (e.g.\ on
$S^3/\Gamma$).  For $q > 1$, the deformation generically breaks
periodicity, and one must appeal to KAM-type persistence results.

In the category of Lie algebroids over~$M$, the Frobenius integrability
condition is equivalent to $H$ carrying a Frobenius structure
(Definition~\ref{def:quantum-universe}).  Since the Levi form
$d\theta|_{H \times H}$ is non-degenerate, this Frobenius structure
\emph{fails} --- and it is precisely this failure that forces the
dimension bound on integral submanifolds and gives $q$-torsion leaves
their rigidity.
\end{remark}

\begin{remark}[Physical interpretation of the Reeb decomposition]
\label{rem:reeb-physical}
The Reeb vector field $\xi$ singles out the direction of internal spin on the
CR manifold.  Forms that ``see'' this direction (fermions) carry half-integer
spin; forms confined to the horizontal distribution (bosons) do not couple to
the Reeb flow and carry integer spin.  The decomposition
$\alpha = \alpha_H + \alpha_V$ is therefore the geometric origin of the
spin--statistics distinction in this framework.  Experimental confirmation of
this picture, including photonic quantum Hall measurements and many-body
interference in crystal lattices, is discussed in
Chapter~\ref{chap:experimental}.
\end{remark}

\begin{definition}[$q$-Weighted Yang--Mills--Levi functional]
\label{def:q-yang-mills-levi}
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with instanton connection~$A$ and substrate parameter~$q$
(Definition~\ref{def:substrate-parameter}).  The
\emph{$q$-weighted Yang--Mills--Levi functional} is
\[
  S_q[A, \theta]
  \;=\; \int_M q(p)\,\lVert F_A^+|_H \rVert^2\,
        \theta \wedge (d\theta)^n,
\]
where $q(p) = 1 + \lvert\langle L_\theta, F_A^+\rangle\rvert(p)$
(Remark~\ref{rem:c-substrate-via-levi}).
\end{definition}

\begin{remark}[General Relativity as the ``Substrate Geometry'']
\label{rem:gr-substrate-geometry}
In this model the metric is effectively replaced by the Levi form
$L_\theta$ and its coupling to the gauge curvature $F_A$.  Gravity is
expressed through the Variational Bicomplex as the back-reaction of the
gauge fields on the CR manifold $M$.
Variation of $S_q$ (Definition~\ref{def:q-yang-mills-levi}) with respect
to the Levi form $L_\theta$ yields field equations of the schematic form
\[
  \frac{\delta S_q}{\delta L_\theta}
  \;=\; 0
  \qquad\Longrightarrow\qquad
  \mathrm{Ric}_{g_\theta} \;=\; T^{(q)},
\]
where $\mathrm{Ric}_{g_\theta}$ is the Webster Ricci curvature and
$T^{(q)}$ is a $q$-weighted stress-energy $2$-tensor on $H$ (both
coordinate-free objects).  The precise derivation of the field equations
will be given in Chapter~\ref{chap:experimental}; here we note only that
the geometry of $M$ is forced to curve to accommodate the tension of the
$\mathrm{SU}(2)$ instantons.
\end{remark}

\begin{proposition}[Main Commutative Diagram]
\label{prop:main-commutative-diagram}
\lean{QOU.main_commutative_diagram}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with instanton connection $A$ (Definition~\ref{def:instanton}) and substrate
parameter $q$ (Definition~\ref{def:substrate-parameter}).  Let
$\mathbf{E}$ denote the Euler--Lagrange operator of the
$q$-weighted Yang--Mills--Levi functional $S_q$
(Definition~\ref{def:q-yang-mills-levi}), and let $\nabla^A$ denote
parallel transport along $A$.  The following diagram commutes:
\[
\begin{CD}
J^1E @>{\mathbf{E}[S_q]}>>
  \bigl\{ L_\theta : \mathrm{Ric}_{g_\theta} = T^{(q)} \bigr\} \\
@V{\text{jet evaluation}}VV
  @VV{q = 1 + \lvert\langle L_\theta, F_A^+\rangle\rvert}V \\
\Gamma(E_{\mathrm{lin}}) @>{\nabla^A}>>
  \bigl\{ \psi : J\hbar_q\,\iota_{\xi_q}D_A\psi = \hat{H}_q\psi \bigr\}
\end{CD}
\]
where the top row sends a jet (a first-order perturbation of the gauge field)
to the Einstein-type field equation via the Euler--Lagrange operator; the
left column evaluates a jet at a fibre point to produce a linearised section;
the right column is the coupling through the substrate field $q$; and the
bottom row is the parallel-transport equation whose linearisation is the
Schr\"odinger equation.
\end{proposition}

\begin{proof}[Proof sketch]
The top row is the variational equation $\delta S_q / \delta L_\theta = 0$,
which involves $q(p)$ through the functional $S_q$.  The bottom row is the
$q$-deformed parallel-transport equation
$J\hbar_q\,\iota_{\xi_q}D_A\psi = \hat{H}_q\psi$
(Remark~\ref{rem:schrodinger-jet-coaction}), linearised at a chosen section.
The left column is the jet evaluation map
$j^1_p \sigma \mapsto \sigma(p)$ restricted to $E_{\mathrm{lin}}$.
Commutativity amounts to the statement that solving the variational
equation for $L_\theta$ and then substituting $q(p)$ into the transport
equation gives the same solution set $\psi$ as first evaluating the jet
and then transporting along $A$.  This follows from the chain rule:
$\mathbf{E}[S_q]$ determines $L_\theta$ (and hence $q$), which in turn
determines $\hat{H}_q$ and $\xi_q$ appearing in the transport equation.
A full proof requires the variational bicomplex machinery of
Chapter~\ref{chap:experimental} and is given there.
\end{proof}

\section{Preview: Lifting and Descent}
\label{sec:preview-lifting}

The two central structural results of this paper are:
\begin{itemize}
  \item \textbf{Jet Prolongation (Lifting).}  An archimedean quantum
        observable universe $(M, G, P, E)$ lifts canonically to a new
        quantum observable universe on the total space of the first jet
        bundle $J^1 E$ (Proposition~\ref{prop:jet-prolongation},
        Chapter~\ref{chap:torsion}).
  \item \textbf{Brane Tower Descent.}  Iterating fixed-point reductions
        under the $\mathrm{SU}(2)$ action produces a tower of nested
        quantum observable universes $(M_k)_{k \geq 0}$, terminating at a
        discrete set of algebraic invariants
        (Section~\ref{ssec:descent}, Chapter~\ref{chap:torsion}).
\end{itemize}
The categorical and analytic elaborations needed for these theorems ---
Tannaka--Krein reconstruction, Hopf--Frobenius diagrams, and the
$q$-deformed operator algebra --- are developed in
Chapter~\ref{chap:elaboration}.  The reader wishing to proceed directly
from the definitions above to geometric applications may skip to
Chapter~\ref{chap:brings-surface}.
