% ── Preliminary: rigid monoidal categories ───────────────────────────────────

\begin{definition}[Rigid monoidal category]
\label{def:rigid-monoidal-category}
A \emph{monoidal category} $(\mathcal{C}, \otimes, \mathbf{1})$ is a
category equipped with a bifunctor
$\otimes \colon \mathcal{C} \times \mathcal{C} \to \mathcal{C}$ (the
\emph{tensor product}), a unit object~$\mathbf{1}$, and natural
isomorphisms expressing associativity and unitality:
\[
\begin{tikzcd}[column sep=3em]
  (X \otimes Y) \otimes Z
    \ar[r, "\alpha_{X,Y,Z}", "\sim"']
  & X \otimes (Y \otimes Z),
\end{tikzcd}
\qquad
\begin{tikzcd}[column sep=2.5em]
  \mathbf{1} \otimes X
    \ar[r, "\ell_X", "\sim"']
  & X
  & X \otimes \mathbf{1}.
    \ar[l, "\rho_X"', "\sim"]
\end{tikzcd}
\]

The category is \emph{rigid} if every object $V$ has a \emph{dual}
$V^{\vee}$ together with morphisms
$\mathrm{ev}_V \colon V \otimes V^{\vee} \to \mathbf{1}$
and
$\mathrm{coev}_V \colon \mathbf{1} \to V^{\vee} \otimes V$
such that the following \emph{snake identity} diagrams commute:
\begin{equation}\label{eq:snake-identities}
\begin{tikzcd}[row sep=2em, column sep=2.5em]
  V
    \ar[r, "\mathrm{id} \otimes \mathrm{coev}"]
    \ar[dr, "\mathrm{id}_V"']
  & V \otimes V^{\vee} \otimes V
    \ar[d, "\mathrm{ev} \otimes \mathrm{id}"]
  \\
  & V
\end{tikzcd}
\qquad\qquad
\begin{tikzcd}[row sep=2em, column sep=2.5em]
  V^{\vee}
    \ar[r, "\mathrm{coev} \otimes \mathrm{id}"]
    \ar[dr, "\mathrm{id}_{V^{\vee}}"']
  & V^{\vee} \otimes V \otimes V^{\vee}
    \ar[d, "\mathrm{id} \otimes \mathrm{ev}"]
  \\
  & V^{\vee}
\end{tikzcd}
\end{equation}
We say $\mathcal{C}$ is \emph{$R$-linear} (for a commutative ring~$R$)
if every hom-set is an $R$-module and composition is $R$-bilinear.
\end{definition}

\begin{example}[{$\mathbb{C}[z]$ and $\mathcal{O}(\mathrm{SU}(2))$}]
\label{ex:rigid-monoidal-vect}
The polynomial ring $\mathbb{C}[z]$ is the algebra of functions on the
complex line~$\mathbb{C}$.  The category
$\mathrm{Vect}_{\mathbb{C}}$ of finite-dimensional complex vector spaces
is the simplest $\mathbb{C}$-linear rigid monoidal category: the tensor
product is $V \otimes W$, the unit is $\mathbf{1} = \mathbb{C}$, and the
dual is $V^{\vee} = \mathrm{Hom}(V, \mathbb{C})$ with
$\mathrm{ev}(v \otimes f) = f(v)$.

Similarly, the \emph{coordinate ring} $\mathcal{O}(\mathrm{SU}(2))$ is
the algebra of polynomial functions on the group $\mathrm{SU}(2)$,
generated by the matrix entries $\alpha, \beta, \bar\alpha, \bar\beta$
with $\alpha\bar\alpha + \beta\bar\beta = 1$.  The category
$\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$ of finite-dimensional
representations is $\mathbb{C}$-linear rigid monoidal.  Its simple
objects are the spin-$j$ representations $V_j$
($j = 0, \tfrac12, 1, \ldots$, $\dim V_j = 2j+1$).  The
\emph{fundamental object} is $V = V_{1/2} \cong \mathbb{C}^2$.
\end{example}

\begin{definition}[Quantum Universe]
\label{def:quantum-universe}
\lean{QOU.QuantumUniverse}\leanok
Let $R$ be a commutative ring.  A \emph{quantum universe over~$R$} is a
triple $(\mathcal{C}, A, \omega)$ where:
\begin{enumerate}
  \item $(\mathcal{C}, \otimes, \mathbf{1})$ is an $R$-linear rigid
        monoidal category
        (Definition~\ref{def:rigid-monoidal-category});
  \item $A$ is a \emph{Frobenius--Hopf object} in~$\mathcal{C}$: an
        object equipped with morphisms
        \[
        \begin{tikzcd}[row sep=1.5em, column sep=2.5em]
          A \otimes A \ar[r, "\mu"] & A \ar[r, "\lambda"] & \mathbf{1}
          \\
          \mathbf{1} \ar[r, "\eta"] & A \ar[r, "\varepsilon"] & \mathbf{1}
          \\
          A \ar[r, "\Delta"] & A \otimes A
          & A \ar[l, "S"']
        \end{tikzcd}
        \]
        --- multiplication~$\mu$, unit~$\eta$, comultiplication~$\Delta$,
        counit~$\varepsilon$, antipode~$S$, and Frobenius
        form~$\lambda$ --- satisfying:
        \begin{enumerate}
          \item[\textup{(H)}] $(A, \mu, \eta, \Delta, \varepsilon, S)$ is
                a Hopf algebra object in~$\mathcal{C}$;
          \item[\textup{(F)}] the \emph{Frobenius pairing} is
                non-degenerate:
                \begin{equation}\label{eq:frobenius-pairing}
                \begin{tikzcd}[column sep=2.5em]
                  A \otimes A \ar[r, "\mu"]
                  & A \ar[r, "\lambda"]
                  & \mathbf{1}
                \end{tikzcd}
                \end{equation}
                induces an isomorphism $A \xrightarrow{\;\sim\;} A^{\vee}$;
        \end{enumerate}
  \item $\omega \colon \mathcal{C} \to R\text{-}\mathrm{Mod}$ is a
        \emph{fibre functor}: an exact, faithful, $R$-linear monoidal
        functor;
  \item $A$ admits a \emph{differential calculus}: the module of
        K\"ahler differentials $\Omega^1(A)$ exists in~$\mathcal{C}$.
\end{enumerate}
\end{definition}

\begin{example}[$\mathrm{SU}(2)$ as a quantum universe]
\label{ex:quantum-universe-su2}
Continuing Example~\ref{ex:rigid-monoidal-vect}, we verify each axiom
of Definition~\ref{def:quantum-universe} for $G = \mathrm{SU}(2)$.

\medskip\noindent\textbf{(i) Category.}\;
$\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$, the category of
finite-dimensional complex representations.  Objects are vector spaces
$V$ with a continuous $G$-action $\rho_V \colon G \to \mathrm{GL}(V)$;
morphisms are $G$-equivariant linear maps.  This is an
$\mathbb{C}$-linear rigid monoidal category
(Example~\ref{ex:rigid-monoidal-vect}).  The $G$-action on tensor
products is diagonal: $g \cdot (v \otimes w) = (g \cdot v) \otimes
(g \cdot w)$; on duals it is the contragredient:
$(g \cdot f)(v) = f(g^{-1} \cdot v)$.

\medskip\noindent\textbf{(ii) Frobenius--Hopf object.}\;
$A = \mathcal{O}(\mathrm{SU}(2))$, the coordinate ring of polynomial
functions on the group.  Its Hopf structure encodes the group operations:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l@{\;\;$=$\;\;}l@{\qquad}l}
  \emph{Multiplication} $\mu(f \otimes g)$ & $f \cdot g$
    & (pointwise product of functions) \\
  \emph{Unit} $\eta(1)$ & $1_A$
    & (constant function $1$) \\
  \emph{Comultiplication} $\Delta(f)(x, y)$ & $f(xy)$
    & (pullback of group multiplication) \\
  \emph{Counit} $\varepsilon(f)$ & $f(e)$
    & (evaluation at the identity) \\
  \emph{Antipode} $S(f)(x)$ & $f(x^{-1})$
    & (pullback of inversion) \\
  \emph{Frobenius form} $\lambda(f)$ & $\int_G f\, d\mu$
    & (Haar integration on $G$) \\
\end{tabular}
\end{center}
The Frobenius pairing
$\langle f, g \rangle = \lambda(\mu(f \otimes g)) = \int_G f g\, d\mu$
is positive definite (faithfulness of the Haar integral).

\medskip\noindent\textbf{(iii) Fibre functor.}\;
$\omega \colon \mathrm{Rep}(\mathrm{SU}(2)) \to
\mathrm{Vect}_{\mathbb{C}}$ is the forgetful functor: $\omega(V, \rho)
= V$ (forget the $G$-action, keep the vector space).  It is exact
(short exact sequences of representations split), faithful (a non-zero
equivariant map is non-zero as a linear map), and monoidal
($\omega(V \otimes W) = \omega(V) \otimes \omega(W)$).
By Tannaka--Krein duality, $G$ can be recovered from $\omega$:
$G \cong \mathrm{Aut}^{\otimes}(\omega)$, the group of monoidal natural
automorphisms of the fibre functor.

\medskip\noindent\textbf{(iv) Differential calculus.}\;
The left-invariant calculus on $\mathrm{SU}(2)$ provides
$\Omega^1(A) = \mathrm{span}\{\omega_1, \omega_2, \omega_3\}$,
the left-invariant $1$-forms dual to the Lie algebra basis
$\{e_1, e_2, e_3\}$ of $\mathfrak{su}(2)$ (with
$[e_a, e_b] = \epsilon_{abc}\, e_c$;
Definition~\ref{def:su2-basis}).
The universal derivation $d \colon A \to \Omega^1(A)$
acts by $df = \sum_i (\partial_i f)\, \omega_i$, recovering the
Lie derivative along each generator.
\end{example}

\begin{definition}[Higher differential forms]
\label{def:higher-forms}
\lean{QOU.HigherForms}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum universe with differential
calculus $\Omega^1(A)$ (Definition~\ref{def:quantum-universe}).  The
\emph{higher differential forms} are the graded $A$-bimodule
\[
  \Omega^k(A) \;=\; \bigwedge\nolimits_A^k \Omega^1(A),
  \qquad k = 0, 1, 2, \ldots
\]
(exterior powers of the $A$-bimodule~$\Omega^1(A)$ in~$\mathcal{C}$),
with $\Omega^0(A) = A$.  For an object $V$ in~$\mathcal{C}$,
$\Lambda^k V$ denotes the $k$-th exterior power of~$V$ in the sense
of rigid monoidal categories.  The differential
$d \colon \Omega^k(A) \to \Omega^{k+1}(A)$ extends the universal
derivation by the graded Leibniz rule, and satisfies $d^2 = 0$.
\end{definition}

\begin{example}[Higher forms on $\mathcal{O}(\mathrm{SU}(2))$]
\label{ex:higher-forms-su2}
Continuing Example~\ref{ex:quantum-universe-su2}, the left-invariant
calculus on $A = \mathcal{O}(\mathrm{SU}(2))$ has
$\Omega^1(A) = \mathrm{span}\{\omega_1, \omega_2, \omega_3\}$
(three left-invariant $1$-forms, dual to the Lie algebra
$\mathfrak{su}(2) = \mathrm{span}\{e_1, e_2, e_3\}$).  The higher
forms are:
\begin{itemize}
  \item $\Omega^0(A) = A = \mathcal{O}(\mathrm{SU}(2))$ (functions on the group).
  \item $\Omega^1(A) = A \otimes \mathrm{span}\{\omega_1, \omega_2, \omega_3\}$
        (left-invariant $1$-forms, dimension~$3$ over~$A$).
  \item $\Omega^2(A) = A \otimes \mathrm{span}\{\omega_1 \wedge \omega_2,\,
        \omega_1 \wedge \omega_3,\, \omega_2 \wedge \omega_3\}$
        (dimension~$3$ over~$A$).
  \item $\Omega^3(A) = A \otimes \mathrm{span}\{\omega_1 \wedge \omega_2 \wedge
        \omega_3\}$ (the volume form, dimension~$1$ over~$A$).
  \item $\Omega^k(A) = 0$ for $k \geq 4$.
\end{itemize}
The differential $d$ is determined by $d\omega_i = -\tfrac{1}{2}
\epsilon_{ijk}\,\omega_j \wedge \omega_k$ (the Maurer--Cartan equation
of~$\mathfrak{su}(2)$).
\end{example}

\begin{definition}[Quantum Observable Universe]
\label{def:quantum-observable-universe}
\lean{QOU.QuantumObservableUniverse}\leanok
A quantum universe $(\mathcal{C}, A, \omega)$ over~$R$
(Definition~\ref{def:quantum-universe}) is \emph{observable} if the
following three conditions hold:
\begin{enumerate}
  \item \emph{(non-degeneracy)} the Frobenius pairing
        \eqref{eq:frobenius-pairing} induces an isomorphism
        $A \xrightarrow{\;\sim\;} A^{\vee}$;
  \item \emph{(observability)} $\mathrm{Hom}_{\mathcal{C}}(\mathbf{1},\,
        \mathrm{Ad}(A)) = 0$, where $\mathrm{Ad}(A)$ is the adjoint object
        ($A$ acted on by conjugation); that is, every morphism
        $\mathbf{1} \to \mathrm{Ad}(A)$ is zero:
        \[
        \begin{tikzcd}[column sep=2.5em]
          \mathbf{1} \ar[r, "0"] & \mathrm{Ad}(A)
        \end{tikzcd}
        \]
  \item \emph{(categorical volume constraint)}
        \label{def:categorical-volume-constraint}
        let $V$ denote the fundamental object of~$\mathcal{C}$ (the simple
        object of categorical dimension~$2$); there exists a
        non-degenerate epimorphism $\epsilon$ preserved by the rigid
        dualities, making the following diagram commute for every
        $g \in \mathrm{Aut}^{\otimes}(\omega)$:
        \begin{equation}\label{eq:categorical-volume-constraint}
        \begin{tikzcd}[row sep=2.5em, column sep=3em]
          \Lambda^2 V \ar[r, "\Lambda^2 g_V"]
                      \ar[d, "\epsilon"', two heads]
          & \Lambda^2 V \ar[d, "\epsilon", two heads]
          \\
          \mathbf{1} \ar[r, "g_{\mathbf{1}} = \mathrm{id}"']
          & \mathbf{1}
        \end{tikzcd}
        \end{equation}
\end{enumerate}
\end{definition}

\begin{example}[$\mathrm{SU}(2)$ is observable]
\label{ex:qou-su2}
Continuing Examples~\ref{ex:rigid-monoidal-vect}
and~\ref{ex:quantum-universe-su2}, write
$V = V_{1/2} \cong \mathbb{C}^2$:
\begin{enumerate}
  \item \emph{Non-degeneracy.}  The Haar integral $\lambda = \int_G$ on
        $\mathcal{O}(\mathrm{SU}(2))$ is faithful: if
        $\lambda(f^* f) = 0$ then $f = 0$.
  \item \emph{Observability.}  The adjoint representation
        $\mathrm{Ad} \cong V_1$ (spin-$1$, dimension~$3$)
        is irreducible, so it contains no copy of the
        trivial representation: $\mathrm{Hom}(\mathbf{1}, V_1) = 0$.
  \item \emph{Volume constraint.}  For $v, w \in V = \mathbb{C}^2$,
        the determinant
        $\epsilon(v \wedge w) = v_1 w_2 - v_2 w_1$
        defines a non-degenerate map
        $\epsilon \colon \Lambda^2 V \to \mathbb{C}$.
\end{enumerate}
\end{example}

\begin{remark}[Base-ring independence]
\label{rem:base-field-independence}
Definitions~\ref{def:quantum-universe} and~\ref{def:quantum-observable-universe}
are formulated over an arbitrary commutative ring~$R$.  The Frobenius--Hopf axioms
replace the analytic vanishing conditions of the archimedean realisation
(Example~\ref{def:c-quantum-observable-universe}) with a single algebraic
non-degeneracy condition on the Frobenius form.  Concrete realisations
include:
\begin{itemize}
  \item $R = K$ an archimedean field (a field with an archimedean absolute
        value, e.g.\ $\mathbb{Q}$, $\mathbb{R}$, or $\mathbb{C}$): the
        setting of CR geometry and contact manifolds
        (Example~\ref{def:c-quantum-universe});
  \item $R = \mathbb{Q}_\pp$ (the $\pp$-adic field): the gauge datum becomes
        an $F$-isocrystal with Frobenius lift
        (Example~\ref{ex:p-adic-realisation},
        Chapter~\ref{chap:p-adic});
  \item $R = \mathbb{F}_\pp$ (a finite field): representations of a quantum
        group over~$\mathbb{F}_\pp$
        (Example~\ref{ex:finite-field-realisation});
  \item $R = \mathbb{Z}[q, q^{-1}]$: the universal (Lusztig) integral form.
\end{itemize}
The fibre functor $\omega$ ensures that $G = \mathrm{Aut}^{\otimes}(\omega)$
is well-defined over any~$R$.
\end{remark}

\begin{proposition}[Frobenius relation]
\label{prop:frobenius-relation}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe
(Definition~\ref{def:quantum-universe}).  Denote by $\Delta_\mu$ the
comultiplication on~$A$ dual to the multiplication~$\mu$ under the
Frobenius pairing~\eqref{eq:frobenius-pairing}.  Then the following
diagram commutes (the \emph{Frobenius relation}):
\begin{equation}\label{eq:frobenius-relation}
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  A \otimes A
    \ar[r, "\mathrm{id} \otimes \Delta_\mu"]
    \ar[d, "\mu"']
  & A \otimes A \otimes A
    \ar[d, "\mu \otimes \mathrm{id}"]
  \\
  A \ar[r, "\Delta_\mu"']
  & A \otimes A
\end{tikzcd}
\end{equation}
That is, sliding a multiplication past a comultiplication (or vice versa)
yields the same result as composing them:
$(\mu \otimes \mathrm{id}) \circ (\mathrm{id} \otimes \Delta_\mu)
= \Delta_\mu \circ \mu$.
\end{proposition}

\begin{proof}
Non-degeneracy of the pairing
$\langle a, b \rangle = \lambda(\mu(a \otimes b))$ identifies
$A \cong A^{\vee}$.  Under this identification, $\Delta_\mu$ is the
transpose of~$\mu$.  The Frobenius relation then follows from the
associativity of~$\mu$ by dualising one leg of the multiplication.
\end{proof}

\begin{example}[Frobenius relation for $\mathrm{SU}(2)$]
\label{ex:frobenius-su2}
Continuing Example~\ref{ex:qou-su2}, set
$A = \mathcal{O}(\mathrm{SU}(2))$ with multiplication
$\mu(f \otimes g) = fg$ (pointwise product) and Frobenius form
$\lambda = \int_G$ (Haar integral).  The dual comultiplication is
$\Delta_\mu(f)(g,h) = \int_G f(xg)\,\overline{h(x)}\,dx$, and the
Frobenius relation expresses the associativity of convolution:
$\int_G f(xy)\,g(y)\,dy$ can be evaluated by convolving first with $f$
or first with~$g$.
\end{example}

\begin{definition}[Involutive structure ($\ast$-structure)]
\label{def:involutive-structure}
\lean{QOU.InvolutiveStructure}\leanok
Let $(\mathcal{C}, \otimes, \mathbf{1})$ be an $R$-linear rigid monoidal
category.  An \emph{involutive structure} (also called a
\emph{$\ast$-structure} or \emph{dagger structure}) on~$\mathcal{C}$ is
a contravariant endofunctor
$\dagger \colon \mathcal{C}^{\mathrm{op}} \to \mathcal{C}$ satisfying:
\begin{enumerate}
  \item \textbf{Identity on objects:} $X^{\dagger} = X$ for every
        object~$X$.
  \item \textbf{Involutivity:} $f^{\dagger\dagger} = f$ for every
        morphism~$f$, depicted:
        \begin{equation}\label{eq:involutivity-cd}
        \begin{tikzcd}[column sep=3em]
          X \ar[r, "f"] \ar[rr, bend left=30, "f^{\dagger\dagger} = f"]
          & Y \ar[r, "f^{\dagger}"{description}]
          & X
        \end{tikzcd}
        \end{equation}
  \item \textbf{Monoidal compatibility:}
        $(\mathrm{id}_X \otimes f)^{\dagger}
        = \mathrm{id}_X \otimes f^{\dagger}$, i.e.\
        the following square commutes:
        \begin{equation}\label{eq:monoidal-dagger-cd}
        \begin{tikzcd}[row sep=2.5em, column sep=3em]
          X \otimes Y \ar[r, "\mathrm{id} \otimes f"]
            \ar[d, "(\mathrm{id} \otimes f)^{\dagger}"']
          & X \otimes Z \ar[d, "\mathrm{id} \otimes f^{\dagger}"] \\
          X \otimes Z \ar[r, equal]
          & X \otimes Z
        \end{tikzcd}
        \end{equation}
  \item \textbf{Contravariance:}
        $(g \circ f)^{\dagger} = f^{\dagger} \circ g^{\dagger}$.
\end{enumerate}
For the Hopf algebra $A$ in a quantum universe, the antipode $S$ provides
a natural involutive structure on the representation category via
$f^{\dagger} = \lambda \circ \mu(S(f) \otimes -)$.  In the
$\mathrm{SU}(2)$ example, this is the Hilbert-space adjoint
(Example~\ref{ex:dagger-su2}).
\end{definition}

\begin{definition}[Positive-definite structure]
\label{def:positive-definite-dagger}
\lean{QOU.PositiveDefiniteStructure}\leanok
Let $(\mathcal{C}, \dagger)$ be a rigid monoidal category with involutive
structure (Definition~\ref{def:involutive-structure}).  The involution is
\emph{positive-definite} if for every non-zero
morphism $f \colon \mathbf{1} \to X$ the composite
\begin{equation}\label{eq:positive-definite-cd}
\begin{tikzcd}[column sep=2.5em]
  \mathbf{1} \ar[r, "f"]
  & X \ar[r, "f^{\dagger}"]
  & \mathbf{1}
\end{tikzcd}
\end{equation}
is a positive element of $\mathrm{End}(\mathbf{1}) \cong R$.
Equivalently, the sesquilinear form
\[
  \langle g, h \rangle
  \;=\; g^{\dagger} \circ h
  \;\in\; \mathrm{End}(\mathbf{1})
\]
is positive-definite on $\mathrm{Hom}(\mathbf{1}, X)$ for every
object~$X$.  This replaces the analytic notion of ``Hilbert-space inner
product'' with a purely categorical condition.
\end{definition}

\begin{example}[Dagger structure on $\mathrm{Rep}(\mathrm{SU}(2))$]
\label{ex:dagger-su2}
Continuing Example~\ref{ex:frobenius-su2}, the category
$\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$ carries a canonical dagger
structure: for a morphism $f \colon V_j \to V_k$ between irreducible
representations, $f^{\dagger}$ is the Hilbert-space adjoint with respect
to the $\mathrm{SU}(2)$-invariant inner product on each $V_j$.
Positive-definiteness follows: for $f \colon \mathbb{C} \to V_j$
non-zero, $f^{\dagger} \circ f = \lVert f(1) \rVert^2 > 0$.
\end{example}

\begin{lemma}[Volume form from rigidity and symplectic self-duality]
\label{lem:volume-from-rigidity}
Let $(\mathcal{C}, \otimes, \mathbf{1})$ be an $R$-linear rigid monoidal
category, and let $V$ be an object of categorical dimension~$2$ that is
\emph{symplectically self-dual}: there exists an isomorphism
$\phi \colon V \xrightarrow{\;\sim\;} V^{\vee}$ satisfying
$\phi^{\vee} = -\phi$ (alternating).  Then the composite
$\mathrm{ev}_V \circ (\mathrm{id} \otimes \phi)$ factors through
$\Lambda^2 V$ (the \emph{volume form factorisation}):
\begin{equation}\label{eq:volume-form-factorisation}
\begin{tikzcd}[row sep=2.5em, column sep=3em]
  V \otimes V
    \ar[r, "\mathrm{id} \otimes \phi"]
    \ar[d, "\pi"', two heads]
  & V \otimes V^{\vee}
    \ar[d, "\mathrm{ev}_V"]
  \\
  \Lambda^2 V
    \ar[r, "\epsilon"', dashed]
  & \mathbf{1}
\end{tikzcd}
\end{equation}
and $\epsilon$ is non-degenerate; that is,
condition~\textup{(iii)} of
Definition~\textup{\ref{def:quantum-observable-universe}} holds.
\end{lemma}

\begin{proof}
Because $\phi$ is alternating, the pairing
$B = \mathrm{ev}_V \circ (\mathrm{id} \otimes \phi)$ satisfies
$B \circ \tau_{V,V} = -B$, where $\tau_{V,V}$ is the symmetry (or
braiding) isomorphism.  Hence $B$ vanishes on symmetric tensors and
factors as $B = \epsilon \circ \pi$, where
$\pi \colon V \otimes V \twoheadrightarrow \Lambda^2 V$ is the
canonical projection.  Non-degeneracy of~$\epsilon$ follows from
non-degeneracy of~$\mathrm{ev}_V$ (guaranteed by rigidity) and the
isomorphism~$\phi$.  The duality axioms
$(\mathrm{ev}_V \otimes \mathrm{id}) \circ
(\mathrm{id} \otimes \mathrm{coev}_V) = \mathrm{id}_V$ ensure that
$\epsilon$ is preserved by the rigid dualities of~$\mathcal{C}$.
\end{proof}

\begin{example}[$\mathrm{SU}(2)$ volume form from the lemma]
\label{ex:su2-volume-from-rigidity}
Continuing Example~\ref{ex:qou-su2}, take $V = V_{1/2} \cong \mathbb{C}^2$.
The symplectic form is
$\phi = \begin{psmallmatrix} 0 & 1 \\ -1 & 0 \end{psmallmatrix}$,
identifying $V \cong V^{\vee}$ with $\phi^{\vee} = -\phi$ (alternating).
Lemma~\ref{lem:volume-from-rigidity} then gives:
\begin{itemize}
  \item The evaluation pairing becomes
        $B(v, w) = v_1 w_2 - v_2 w_1$ (the determinant).
  \item $B$ is alternating, so it factors through $\Lambda^2 V$:
        the map $\epsilon \colon \Lambda^2 V \to \mathbb{C}$ is
        $\epsilon(v \wedge w) = \det(v, w)$.
  \item Non-degeneracy: $\epsilon$ is an isomorphism
        $\Lambda^2 \mathbb{C}^2 \cong \mathbb{C}$.
  \item For $g \in G$, $\Lambda^2 g = \det(g) \cdot \mathrm{id}$, so
        $\epsilon$-preservation forces $\det(g) = 1$: this is why the
        gauge group is $\mathrm{SU}(2)$ rather than
        $\mathrm{GL}(2, \mathbb{C})$ or $\mathrm{U}(2)$.
\end{itemize}
\end{example}

\begin{remark}[Tannaka--Krein duality: what the fibre functor gives us]
\label{rem:tannaka-krein-overview}
The fibre functor $\omega$ of a quantum universe encodes an extraordinary
amount of structure.  Tannaka--Krein duality \cite{deligne1990} says:
\begin{enumerate}
  \item \textbf{Reconstruction.}\; The group $G$ can be recovered from
        $\omega$ alone:
        $G \cong \mathrm{Aut}^{\otimes}(\omega)$,
        the group of monoidal natural automorphisms of~$\omega$.
        No \emph{a priori} group is needed; it \emph{emerges} from the
        category and the functor.
  \item \textbf{Equivalence.}\; The functor $\omega$ establishes a
        monoidal equivalence
        $\mathcal{C} \simeq \mathrm{Rep}(G)$:
        every object in~$\mathcal{C}$ is a representation of the
        reconstructed group.
  \item \textbf{Rigidity.}\; Any two fibre functors
        $\omega, \omega' \colon \mathcal{C} \to R\text{-}\mathrm{Mod}$
        are isomorphic (as monoidal functors), and the group they
        reconstruct is the same up to inner automorphism.
  \item \textbf{Constraint propagation.}\; Algebraic conditions on
        $\omega$ (such as the volume constraint
        $\epsilon \colon \Lambda^2 V \twoheadrightarrow \mathbf{1}$)
        propagate to $G$: they cut down the automorphism group from
        $\mathrm{GL}$ to a smaller subgroup (in our case, $\det = 1$
        forces $\mathrm{SL}$; positive-definiteness then forces $\mathrm{SU}$).
\end{enumerate}
In the classical setting, Tannaka--Krein recovers a compact group from
its representations \cite{tannaka1939}.  The version we use is
Deligne's algebraic generalisation \cite{deligne1990}, which works
over any field and reconstructs an affine group scheme.  For quantum
groups ($q \neq 1$), the appropriate extension is due to
Woronowicz~\cite{woronowicz1988}.
\end{remark}

\begin{theorem}[Emergence of the gauge group]
\label{thm:su2-emergence}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}).  Then
$G = \mathrm{Aut}^{\otimes}(\omega) \cong \mathrm{SU}(2)$
(or $\mathrm{SL}(2, R)$ depending on the signature of the Frobenius
pairing).
\end{theorem}

\begin{proof}
The gauge group $G$ is defined via Tannaka--Krein duality as
the group scheme of tensor-preserving natural automorphisms of the fibre
functor~$\omega$.  An element $g \in G$ assigns an invertible linear map
$g_X \colon \omega(X) \to \omega(X)$ to every object $X \in \mathcal{C}$.

Because $g$ is a natural transformation, it commutes with all morphisms
in~$\mathcal{C}$, including the volume constraint
$\epsilon \colon \Lambda^2 V \to \mathbf{1}$
\eqref{eq:categorical-volume-constraint}.  Applying~$\omega$ yields the
commutative diagram of $R$-modules
\begin{equation}\label{eq:su2-emergence-diagram}
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  \Lambda^2 \omega(V)
    \ar[r, "\Lambda^2 g_V"]
    \ar[d, "\omega(\epsilon)"']
  & \Lambda^2 \omega(V)
    \ar[d, "\omega(\epsilon)"]
  \\
  \omega(\mathbf{1})
    \ar[r, "g_{\mathbf{1}} = \mathrm{id}"']
  & \omega(\mathbf{1})
\end{tikzcd}
\end{equation}
The fundamental object $V$ has categorical dimension~$2$, so
$\omega(V)$ is a rank-$2$ free $R$-module and $\Lambda^2 \omega(V)$ has
rank~$1$.  The induced operator $\Lambda^2 g_V$ acts on this rank-$1$
module as scalar multiplication by $\det(g_V)$.  Following
diagram~\eqref{eq:su2-emergence-diagram} from the top-left to the
bottom-right:
\[
  \omega(\epsilon) \circ \Lambda^2 g_V
  \;=\;
  g_{\mathbf{1}} \circ \omega(\epsilon)
  \;=\;
  \omega(\epsilon).
\]
Since $\omega(\epsilon)$ is non-degenerate (hence non-zero), this forces
\begin{equation}\label{eq:det-one}
  \det(g_V) \;=\; 1,
\end{equation}
restricting the natural automorphisms from $\mathrm{GL}(2)$ to the
special subgroup $\mathrm{SL}(2)$.

The QOU non-degeneracy axiom ensures the Frobenius form
$\lambda \colon A \to \mathbf{1}$ is non-degenerate, inducing the
Frobenius self-duality $A \cong A^{\vee}$
(Proposition~\ref{prop:fiber-adjoint-involution}).  Define a dagger on
morphisms by $f^{\dagger} = \lambda \circ \mu(S(f) \otimes -)$, where
$S$ is the antipode.  Non-degeneracy of~$\lambda$ and faithfulness
of~$\omega$ imply $f^{\dagger} \circ f = 0 \Rightarrow f = 0$, so this
is a positive-definite dagger structure
(Definition~\ref{def:positive-definite-dagger}).  Every tensor
automorphism $g \in G$ preserves the dagger (it commutes with $S$ and
$\lambda$), forcing $g_V^{-1} = g_V^{\dagger}$ and locking the group
into its unitary real form $\mathrm{SU}(2)$.
\end{proof}

\begin{example}[Emergence of $\mathrm{SU}(2)$: the full chain]
\label{ex:su2-emergence}
Continuing Examples~\ref{ex:frobenius-su2}--\ref{ex:dagger-su2}, let
$\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$,
$A = \mathcal{O}(\mathrm{SU}(2))$, $\omega$ the forgetful functor.
\begin{enumerate}
  \item \emph{Volume constraint} $\Rightarrow$ $\det(g_V) = 1$.\;
        The alternating form $\epsilon(v \wedge w) = v_1 w_2 - v_2 w_1$
        on $V = \mathbb{C}^2$ is $\mathrm{SU}(2)$-invariant.  Any tensor
        automorphism $g$ preserves~$\epsilon$, forcing
        $g_V \in \mathrm{SL}(2, \mathbb{C})$.
  \item \emph{Dagger structure} $\Rightarrow$ unitarity.\;
        The dagger on $\mathrm{Rep}(\mathrm{SU}(2))$
        (Example~\ref{ex:dagger-su2}) forces
        $g_V^{-1} = g_V^{\dagger}$, cutting from
        $\mathrm{SL}(2, \mathbb{C})$ to its compact real form
        $\mathrm{SU}(2)$.
  \item \emph{Conclusion.}\;
        $G = \mathrm{Aut}^{\otimes}(\omega) \cong \mathrm{SU}(2)$,
        recovering the input gauge group from the categorical axioms
        alone.
\end{enumerate}
\end{example}

\begin{definition}[Spin labels and the fundamental object]
\label{def:spin-labels}
\lean{QOU.SpinLabels}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}).  Because
$G = \mathrm{Aut}^{\otimes}(\omega) \cong \mathrm{SU}(2)$
(Theorem~\ref{thm:su2-emergence}), the simple objects of~$\mathcal{C}$
are indexed by a \emph{spin} label $j \in \{0, \tfrac12, 1,
\tfrac32, \ldots\}$, with $V_j$ denoting the simple object of
categorical dimension $2j + 1$.
The \emph{fundamental object} is $V = V_{1/2}$, the unique simple object
of categorical dimension~$2$.  The \emph{adjoint object} is
$V_1 \cong \mathrm{Ad}(A)$, of dimension~$3$.

The symplectic form $\epsilon \colon \Lambda^2 V \to \mathbf{1}$
(Definition~\ref{def:quantum-observable-universe}, axiom~(iii)) is the
unique (up to scalar) $G$-invariant alternating pairing on~$V$.
\end{definition}

\begin{example}[Spin decomposition of $\mathrm{Rep}(\mathrm{SU}(2))$]
\label{ex:spin-labels-su2}
Continuing Example~\ref{ex:qou-su2}, the simple objects of
$\mathcal{C} = \mathrm{Rep}(\mathrm{SU}(2))$ are the irreducible
representations $V_j \cong \mathbb{C}^{2j+1}$:
\begin{equation}\label{eq:spin-decomposition}
\begin{tikzcd}[row sep=1.5em, column sep=1.5em]
  V_0 \ar[d, phantom, "\cong"] & V_{1/2} \ar[d, phantom, "\cong"]
    & V_1 \ar[d, phantom, "\cong"] & V_{3/2} \ar[d, phantom, "\cong"]
    & \cdots \\
  \mathbb{C} & \mathbb{C}^2 & \mathbb{C}^3 & \mathbb{C}^4 & \cdots
\end{tikzcd}
\end{equation}
In concrete terms:
\begin{itemize}
  \item $V_0 = \mathbb{C}$ (the trivial representation: $g \cdot z = z$).
  \item $V_{1/2} = \mathbb{C}^2$ (the defining representation:
        $g \cdot v = gv$ by matrix multiplication).  This is the
        fundamental object~$V$.
  \item $V_1 = \mathfrak{su}(2)_{\mathbb{C}} \cong \mathbb{C}^3$
        (the adjoint representation: $g \cdot X = gXg^{-1}$).  This is
        $\mathrm{Ad}(A)$.
  \item $V_{3/2} = \mathrm{Sym}^3(\mathbb{C}^2) \cong \mathbb{C}^4$
        (homogeneous cubics in two variables).
\end{itemize}
Every finite-dimensional representation decomposes as
$W \cong \bigoplus_j V_j^{\oplus m_j}$, with multiplicities
$m_j \in \mathbb{N}$.  The polynomial ring
$\mathbb{C}[z] = \bigoplus_{n \geq 0} \mathrm{Sym}^n V$ is the
coordinate ring of $\mathbb{P}^1(\mathbb{C}) = \mathbb{P}(V)$
(Definition~\ref{def:projective-line-object}), and each homogeneous
component $\mathrm{Sym}^n V \cong V_{n/2}$ is an irreducible
representation of dimension~$n + 1$.
\end{example}

\begin{definition}[Maximal torus and weight subcategory]
\label{def:maximal-torus}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) with fundamental
object~$V$ (Definition~\ref{def:spin-labels}).  The \emph{weight
subcategory} $\langle V \rangle \subset \mathcal{C}$ is the full
subcategory whose objects are direct sums of weight spaces of~$V$
(i.e.\ simultaneous eigenspaces of the Cartan subalgebra action).
The \emph{maximal torus} of the gauge group is
\[
  T \;=\; \mathrm{Aut}^{\otimes}(\omega|_{\langle V \rangle}),
\]
the sub-group scheme of
$G = \mathrm{Aut}^{\otimes}(\omega)$
(Theorem~\ref{thm:su2-emergence}) consisting of those natural
automorphisms that act diagonally on each weight space of~$V$.
Equivalently, $T$ is the largest subgroup of~$G$ for which every
object in $\langle V \rangle$ decomposes into one-dimensional
$T$-eigenspaces.
\end{definition}

\begin{example}[Maximal torus for $\mathrm{SU}(2)$]
\label{ex:maximal-torus-su2}
Continuing Example~\ref{ex:spin-labels-su2}, the fundamental object is
$V = \mathbb{C}^2$ with the standard $\mathrm{SU}(2)$-action.  The
weight decomposition is $V = \mathbb{C} e_+ \oplus \mathbb{C} e_-$
(the $\pm\tfrac{1}{2}$-eigenspaces of
$e_3 = \frac{1}{2}\begin{psmallmatrix} 1 & 0 \\ 0 & -1 \end{psmallmatrix}$).
The weight subcategory $\langle V \rangle$ consists of representations
that decompose into weight spaces: $V$~itself, $\mathbf{1}$, and their
direct sums.  The maximal torus is
\[
  T \;=\; \mathrm{U}(1)
  \;=\; \left\{
    \begin{pmatrix} e^{i\theta} & 0 \\ 0 & e^{-i\theta} \end{pmatrix}
    : \theta \in \mathbb{R}
  \right\},
\]
the diagonal matrices in $\mathrm{SU}(2)$.  Each element of $T$ acts on
$e_{\pm}$ by $e^{\pm i\theta}$: one-dimensional eigenspaces, as required.
\end{example}

\begin{definition}[Weight decomposition]
\label{def:weight-decomposition}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) with maximal torus
$T$ (Definition~\ref{def:maximal-torus}) and higher differential forms
$\Omega^k(A)$ (Definition~\ref{def:higher-forms}).  The $T$-action on
$\Omega^k(A)$ induces a $\mathbb{Z}$-grading by weights.  Denoting by
$\pi_H, \pi_V$ the projections onto the weight-$0$ and non-zero-weight
summands, every $\alpha \in \Omega^k(A)$ decomposes uniquely:
\begin{equation}\label{eq:weight-decomposition-cd}
\begin{tikzcd}[row sep=2em, column sep=3em]
  & \Omega^k(A) \ar[dl, "\pi_H"'] \ar[dr, "\pi_V"] & \\
  \Omega^k_H(A) \ar[rr, hook, "\oplus"']
  & & \Omega^k_V(A)
\end{tikzcd}
\qquad \alpha = \alpha_H + \alpha_V.
\end{equation}
Here $\alpha_H = \pi_H(\alpha)$ is the \emph{horizontal component}
(weight-$0$ under $T$) and $\alpha_V = \pi_V(\alpha)$ is the
\emph{vertical component} (non-zero weight).  A form is \emph{purely
horizontal} if and only if $\alpha_V = 0$.
\end{definition}

\begin{remark}[Physical interpretation of the weight decomposition]
\label{rem:reeb-physical}
The decomposition $\alpha = \alpha_H + \alpha_V$ is the geometric origin of
spin--statistics: fermions (non-trivial $\alpha_V$) carry half-integer spin;
bosons (purely horizontal) carry integer spin.  See
Chapter~\ref{chap:experimental} for experimental confirmation.
\end{remark}

\begin{definition}[Categorical Reeb derivation]
\label{def:categorical-reeb}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
maximal torus $T$ (Definition~\ref{def:maximal-torus}) and weight
decomposition $\pi_H, \pi_V$
(Definition~\ref{def:weight-decomposition}).  The \emph{categorical
Reeb derivation} is the endomorphism $\xi \colon A \to A$ that
generates the $T$-action: $\xi$ is the infinitesimal generator of the
one-parameter subgroup $T \cong \mathrm{U}(1) \subset G$, acting on
the Frobenius--Hopf object~$A$ via the adjoint coaction.  It satisfies:
\begin{enumerate}
  \item \textbf{Vertical generator:}\;
        $\pi_V \circ d = \xi \lrcorner\, d\theta_A$, where
        $\theta_A \in \Omega^1(A)$ is the canonical $1$-form dual to~$\xi$.
  \item \textbf{Normalisation:}\;
        $\theta_A(\xi) = \mathrm{id}_{\mathbf{1}}$.
  \item \textbf{Kernel:}\;
        $\Omega^k_H(A) = \ker(\xi \lrcorner)$ (horizontal forms are
        precisely those annihilated by contraction with~$\xi$).
\end{enumerate}
\end{definition}

\begin{example}[Reeb derivation for $\mathrm{SU}(2)$]
\label{ex:reeb-su2}
Continuing Example~\ref{ex:maximal-torus-su2}, the maximal torus is
$T = \mathrm{U}(1)$ with generator
$\xi = e_3 = \frac{1}{2}\begin{psmallmatrix} 1 & 0 \\ 0 & -1
\end{psmallmatrix} \in \mathfrak{su}(2)$.
On $A = \mathcal{O}(\mathrm{SU}(2))$, $\xi$ acts by
$(\xi \cdot f)(g) = \frac{d}{dt}\big|_{t=0} f(e^{-t\xi} g)$
(left-invariant vector field).  The canonical $1$-form is
$\theta_A = \omega_3$ (the left-invariant $1$-form dual to~$e_3$), and:
\begin{itemize}
  \item $\omega_3(\xi) = 1$ (normalisation).
  \item $\Omega^k_H(A) = \{\alpha : \iota_\xi\,\alpha = 0\}$
        (forms with no $\omega_3$ component).
  \item In the archimedean realisation on a CR manifold $(M, \theta)$,
        $\xi$ becomes the Reeb vector field with $\iota_\xi\,d\theta = 0$
        and $\theta(\xi) = 1$
        (Example~\ref{def:reeb-decomposition}).
\end{itemize}
\end{example}

\begin{definition}[Projective line object]
\label{def:projective-line-object}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
maximal torus $T \subset G$ (Definition~\ref{def:maximal-torus}).
The \emph{projective line object} is the quotient
\[
  \mathbb{P}(V) \;=\; G \,/\, T,
\]
regarded as an object of $\mathrm{Ind}(\mathcal{C})$ via its function
algebra $\mathcal{O}(\mathbb{P}(V)) = A^{\mathrm{co}\,T}$
(the $T$-coinvariant subalgebra of~$A$).
\end{definition}

\begin{example}[Projective line for $\mathrm{SU}(2)$]
\label{ex:projective-line-su2}
Continuing Example~\ref{ex:maximal-torus-su2},
$G = \mathrm{SU}(2)$ and $T = \mathrm{U}(1)$.
The quotient is
$\mathbb{P}(V) = \mathrm{SU}(2)/\mathrm{U}(1) \cong \mathbb{P}^1(\mathbb{C})$.
A point $[\ell] \in \mathbb{P}^1(\mathbb{C})$ is a line $\ell \subset V = \mathbb{C}^2$;
equivalently, it is the kernel of a linear form on~$V$, well-defined up
to scalar.  The function algebra is the polynomial ring
$\mathcal{O}(\mathbb{P}^1(\mathbb{C})) = \mathbb{C}[z]$ in the affine coordinate
$z = v_1 / v_2$.
\end{example}

\begin{proposition}[Fiber--adjoint-fiber involution]
\label{prop:fiber-adjoint-involution}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}).  The Frobenius form
$\lambda \colon A \to \mathbf{1}$ induces an isomorphism
$A \xrightarrow{\;\sim\;} A^{\vee}$ (the Frobenius self-duality), under
which:
\begin{enumerate}
  \item the projective line object $\mathbb{P}(V)$
        (Definition~\ref{def:projective-line-object}) embeds in the adjoint
        object $\mathrm{Ad}(A)$ as the coadjoint orbit of a highest-weight
        vector;
  \item the assignment $\sigma \colon \mathbb{P}(V) \hookrightarrow
        \mathrm{Ad}(A)$ is an involution in the sense that $\sigma$
        determines $\mathbb{P}(V)$ from $\mathrm{Ad}(A)$ and vice versa;
  \item the non-degeneracy of the Frobenius pairing forces the
        observability condition:
        \[
          \mathrm{Hom}(\mathbf{1},\, \mathrm{Ad}(A)) \;=\; 0.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
The Frobenius isomorphism $A \cong A^{\vee}$ sends $a \mapsto \lambda \circ
\mu(a \otimes -)$, which is non-degenerate by the Frobenius--Hopf axiom.
Under this identification, the quotient $\mathbb{P}(V) = G/T$ embeds
in $\mathrm{Ad}(A)$ via the orbit map $gT \mapsto \mathrm{Ad}(g)(h)$ for a
fixed highest-weight element $h \in \mathrm{Ad}(A)$.  The orbit is closed
(by rigidity of~$\mathcal{C}$) and determines a sub-object of
$\mathrm{Ad}(A)$ isomorphic to $\mathbb{P}(V)$.  Conversely, given the
sub-object, the quotient $G/T$ is recovered as its stabiliser quotient.  Item
(3) follows because $\mathrm{Hom}(\mathbf{1}, \mathrm{Ad}(A))$ would provide
a $G$-fixed point in $\mathrm{Ad}(A)$, but the coadjoint orbit
$\mathbb{P}(V)$ is a homogeneous space with no fixed points (the $T$-action
on $\mathbb{P}(V)$ is free away from the poles, and non-degeneracy of
$\lambda$ excludes the degenerate case).
\end{proof}

\begin{example}[Fiber--adjoint-fiber involution for $\mathrm{SU}(2)$]
\label{ex:fiber-adjoint-su2}
Continuing Example~\ref{ex:projective-line-su2}, take
$G = \mathrm{SU}(2)$, $T = \mathrm{U}(1)$, and
$V = \mathbb{C}^2$.  The projective line object is
$\mathbb{P}(V) = \mathbb{P}^1(\mathbb{C})$, and the adjoint object is
$\mathrm{Ad}(A) \cong \mathfrak{su}(2) \cong \mathbb{R}^3$
(trace-free skew-Hermitian $2 \times 2$ matrices under the Killing
form).  The embedding $\sigma$ sends a line
$\ell = \mathrm{span}(v) \subset \mathbb{C}^2$ to the rank-$1$
projector
\[
  \sigma([\ell]) \;=\;
  \frac{v v^{\dagger}}{\lVert v \rVert^2}
  \;-\; \tfrac{1}{2}\,\mathrm{id}_V
  \;\in\; \mathfrak{su}(2).
\]
This identifies $\mathbb{P}^1(\mathbb{C})$ with the unit $2$-sphere
$S^2 \subset \mathfrak{su}(2)$ (the coadjoint orbit of
$\tfrac{1}{2}\,\mathrm{diag}(i, -i)$).  Observability follows:
a $G$-fixed point in $\mathfrak{su}(2)$ would be a matrix commuting
with all of $\mathrm{SU}(2)$, hence a scalar --- but trace-free matrices
have no non-zero scalars, so
$\mathrm{Hom}(\mathbf{1},\, \mathrm{Ad}(A)) = 0$.
\end{example}

\begin{definition}[Categorical connection and curvature]
\label{def:categorical-connection}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over~$R$ with
differential forms $\Omega^k(A)$ (Definition~\ref{def:higher-forms}).

A \emph{connection} on~$A$ is a morphism
$\nabla \colon A \to A \otimes \Omega^1(A)$ in~$\mathcal{C}$ satisfying
the Leibniz rule: $\nabla(\mu(a \otimes b)) =
\nabla(a) \cdot b + a \cdot \nabla(b)$, expressed by the commutativity
of the \emph{Leibniz triangle}:
\begin{equation}\label{eq:leibniz-triangle}
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  A \otimes A
    \ar[r, "\mu"]
    \ar[dr, "{\nabla \otimes 1 \,+\, 1 \otimes \nabla}"']
  & A \ar[d, "\nabla"]
  \\
  & A \otimes \Omega^1(A)
\end{tikzcd}
\end{equation}
The \emph{curvature} of~$\nabla$ is the composition
\begin{equation}\label{eq:curvature-composition}
\begin{tikzcd}[column sep=3em]
  A \ar[r, "\nabla"]
  & A \otimes \Omega^1(A) \ar[r, "\nabla"]
  & A \otimes \Omega^2(A)
\end{tikzcd}
\end{equation}
denoted $F_\nabla = \nabla^2$, which is $A$-linear (a standard
consequence of the Leibniz rule).
\end{definition}

\begin{example}[Connection and curvature on $\mathcal{O}(\mathrm{SU}(2))$]
\label{ex:connection-su2}
Continuing Example~\ref{ex:fiber-adjoint-su2}, take
$A = \mathcal{O}(\mathrm{SU}(2))$ with its left-invariant calculus
$\Omega^1(A) = \mathrm{span}\{\omega_1, \omega_2, \omega_3\}$
(Example~\ref{ex:higher-forms-su2}).  A
connection is a covariant derivative
$\nabla f = df + \mathcal{A} \cdot f$ for $f \in A$, where
$\mathcal{A} = A_1\,\omega_1 + A_2\,\omega_2 + A_3\,\omega_3$ is the
connection $1$-form.  The Leibniz triangle says
$\nabla(fg) = (\nabla f)\,g + f\,(dg)$.  The curvature is the $2$-form
$F_\nabla = d\mathcal{A} + \mathcal{A} \wedge \mathcal{A} \in \Omega^2(A)$,
which is $A$-linear ($F_\nabla(f \cdot s) = f \cdot F_\nabla(s)$).
(When the substrate parameter~$q$ is introduced in
Definition~\ref{def:substrate-parameter}, this calculus deforms to the
Woronowicz $3D$-calculus on $\mathcal{O}_q(\mathrm{SU}(2))$.)
\end{example}

\begin{definition}[Categorical jet object]
\label{def:categorical-jet}
\lean{QOU.CategoricalJet}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum universe
(Definition~\ref{def:quantum-universe}) with connection~$\nabla$
(Definition~\ref{def:categorical-connection}).  The \emph{jet
object} is
\[
  J^1(A) \;:=\; A \otimes_{\mathcal{C}} \Omega^1(A),
\]
where $\Omega^1(A) = \ker(\mu \colon A \otimes A \to A)$ is the
bimodule of K\"ahler differentials.  The jet object fits into the
universal extension
\begin{equation}\label{eq:jet-ses}
\begin{tikzcd}[column sep=2.5em]
  0 \ar[r]
  & A \otimes \Omega^1(A) \ar[r]
  & J^1(A) \ar[r, "j^1"]
  & A \ar[r]
  & 0
\end{tikzcd}
\end{equation}
where $j^1 \colon J^1(A) \to A$ is the jet evaluation (the counit of
the tensor--hom adjunction).
\end{definition}

\begin{proposition}[Jet prolongation]
\label{prop:categorical-jet-prolongation}
\lean{QOU.categorical_jet_prolongation}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum universe with
connection~$\nabla$.
Then $J^1(A)$ (Definition~\ref{def:categorical-jet}) inherits a
Frobenius--Hopf structure from~$A$, and the quadruple
$(\mathcal{C},\, J^1(A),\, \omega',\, q)$ --- where $\omega'$ is
the induced fibre functor --- is again a quantum universe.
The following diagram commutes:
\begin{equation}\label{eq:jet-prolongation-cd}
\begin{tikzcd}[row sep=2.5em, column sep=3em]
  J^1(A) \ar[r, "\nabla'"] \ar[d, "j^1"']
  & J^1(A) \otimes \Omega^1(J^1(A)) \ar[d, "j^1 \otimes j^1"] \\
  A \ar[r, "\nabla"']
  & A \otimes \Omega^1(A)
\end{tikzcd}
\end{equation}
where $\nabla'$ is the prolonged connection and $\nabla$ the original.
\end{proposition}

\begin{definition}[Categorical Hodge star]
\label{def:categorical-hodge-star}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over~$R$
(Definition~\ref{def:quantum-universe}) with higher differential forms
$\Omega^k(A)$ (Definition~\ref{def:higher-forms}).  The
\emph{categorical Hodge star} is the involution
$\star \colon \Omega^k(A) \to \Omega^{n-k}(A)$ defined by
\[
  \alpha \wedge \star \beta
  \;=\; \lambda\bigl(\alpha^* \cdot \beta\bigr)\;\mathrm{vol},
\]
where $\lambda \colon A \to \mathbf{1}$ is the Frobenius form,
$\alpha^* = S(\alpha)$ is the antipode-conjugate, and
$\mathrm{vol} \in \Omega^n(A)$ is the volume form
(the unique $G$-invariant top form normalised by
$\lambda(\mathrm{vol}) = 1$).
That $\star^2 = \mathrm{id}$ (i.e.\ $\star$ is an involution)
follows from the non-degeneracy of the Frobenius form~$\lambda$ and the
involutivity of the antipode ($S^2 = \mathrm{id}$ on the centre
of~$A$).
\end{definition}

\begin{remark}[Deformation parameter and quantum gauge group]
\label{rem:quantum-universe-explication}
The deformation parameter~$q$ is not part of the categorical
definition of a quantum universe or observable universe; it enters via
the substrate parameter
(Definition~\ref{def:substrate-parameter} below) when the Hopf structure
on~$A$ is specialised.  Once a substrate parameter is fixed, the
categorical dimension of the fundamental object~$V$ deforms from $2$
to~$[2]_q = q + q^{-1}$, and the gauge group
$G = \mathrm{Aut}^{\otimes}(\omega)$
(Theorem~\ref{thm:su2-emergence}) deforms from $\mathrm{SU}(2)$
to~$\mathrm{SU}_q(2)$.
\end{remark}

\begin{definition}[Substrate parameter]
\label{def:substrate-parameter}
\lean{QOU.SubstrateParameter}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum universe over a ring~$R$
(Definition~\ref{def:quantum-universe}).  Let
$\mathcal{U}^{\mathbb{Z}}_q(\mathfrak{su}(2))$ denote the
\emph{Lusztig integral form}~\cite{lusztig1993}: the
$\mathbb{Z}[q,q^{-1}]$-subalgebra of $\mathcal{U}_q(\mathfrak{su}(2))$
generated by $K^{\pm 1}$ and the divided powers
$E^{(n)} = E^n/[n]_q!$, $F^{(n)} = F^n/[n]_q!$.
The \emph{substrate parameter} is a ring homomorphism
\[
  \iota \;\colon\; \mathbb{Z}[q, q^{-1}] \;\longrightarrow\; k,
  \qquad q \;\longmapsto\; q_0 \in k^{\times},
\]
and the base-changed algebra
$\mathcal{U}^{\mathbb{Z}}_q(\mathfrak{su}(2))
  \otimes_{\mathbb{Z}[q,q^{-1}]} k$
is the quantised enveloping algebra at the specialised value~$q_0$.
When the quantum universe is fibred over a space~$M$ (as in the
archimedean realisation, Example~\ref{def:c-quantum-universe}), $q_0$
may vary from point to point, yielding a \emph{substrate field}
$q \colon M \to k^{\times}$.
\end{definition}

\begin{definition}[Quantum Gauge Group]
\label{def:quantum-gauge-group}
\lean{QOU.QuantumGaugeGroup}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) with substrate
parameter $q_0$ (Definition~\ref{def:substrate-parameter}).  The
\emph{quantum gauge group} is
$G_{q_0} = \mathrm{SU}_{q_0}(2)$, the quantum group obtained by
deforming the Tannakian gauge group
$G = \mathrm{Aut}^{\otimes}(\omega) \cong \mathrm{SU}(2)$
(Theorem~\ref{thm:su2-emergence}) at the parameter~$q_0$.  When the
substrate parameter varies over a space~$M$ (e.g.\ the archimedean
realisation), this yields a field of quantum groups
$\{G_{q(p)}\}_{p \in M}$.
\end{definition}

\begin{definition}[Quantum Planck Constant]
\label{def:quantum-planck-constant}
\lean{QOU.quantumPlanckConstant}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
substrate parameter $q_0$
(Definition~\ref{def:substrate-parameter}).  The
\emph{quantum Planck constant} is the element
$\hbar_q = 1 - q_0^{-1} \in R$.  When the substrate parameter varies
over a space~$M$, this defines a field
$\hbar_q \colon M \to R$ by $\hbar_q(p) = 1 - q(p)^{-1}$.
\end{definition}

\begin{remark}[Limiting regimes of $\hbar_q$]
\label{rem:planck-limits}
Like $q$ itself, $\hbar_q$ is not a universal constant but a field on $M$.
As $q(p) \to 1$ (classicality), $\hbar_q(p) \to 0$ and quantum effects
vanish; as $q(p) \to \infty$ (deep-quantum regime), $\hbar_q(p) \to 1$.
\end{remark}

\begin{definition}[$q$-Hodge star]
\label{def:q-hodge-star}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
categorical Hodge star $\star$
(Definition~\ref{def:categorical-hodge-star}) and substrate
parameter~$q$ (Definition~\ref{def:substrate-parameter}).  The
\emph{$q$-Hodge star} is the $q$-deformed involution
$\star_q \colon \Omega^k(A) \to \Omega^{n-k}(A)$ defined by
\[
  \alpha \wedge_q \star_q \beta
  \;=\; q \cdot \lambda\bigl(\alpha^* \cdot \beta\bigr)\;\mathrm{vol}.
\]
At $q = 1$ this recovers the categorical Hodge star:
$\star_1 = \star$.  The factor of~$q$ rescales the inner product on
forms, coupling the Hodge duality to the local deformation strength.
\end{definition}

\begin{remark}[Notation for the categorical action]
\label{rem:categorical-action-notation}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) equipped with
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}),
$q$-Hodge star $\star_q$ (Definition~\ref{def:q-hodge-star}),
a connection $\nabla$ (Definition~\ref{def:categorical-connection}),
and a section $\psi \colon \mathbf{1} \to A$ (a ``state'').
The following notation is used in
Definition~\ref{def:categorical-action}:
\begin{itemize}
  \item $\mathrm{tr} \colon \mathrm{End}_{\mathcal{C}}(A) \to R$ is the
        \emph{categorical trace} (the composition
        $\mathrm{coev} \to \mathrm{id} \to \mathrm{ev}$ via the rigid
        structure).
  \item $R_\lambda \in \mathrm{End}_{\mathcal{C}}(A)$ is the
        \emph{Frobenius--Ricci endomorphism}: the curvature of the
        Frobenius form~$\lambda$ itself, defined by
        $R_\lambda = (\mathrm{id}_A \otimes \lambda) \circ F_{\nabla_\lambda}$,
        where $\nabla_\lambda$ is the canonical connection on~$A$ induced
        by~$\lambda$ via $\nabla_\lambda(a) = \lambda(a_{(1)}) \otimes
        a_{(2)} - a \otimes 1$.
  \item $\mathrm{tr}_A \colon
        \mathrm{End}_{\mathcal{C}}(A \otimes \Omega^{\bullet}(A)) \to
        \mathrm{End}_{\mathcal{C}}(\Omega^{\bullet}(A))$ is the
        \emph{partial trace} over the~$A$ factor.
  \item $\psi^* = \lambda \circ S(\psi) \colon A \to \mathbf{1}$ is the
        \emph{adjoint section} (antipode composed with the Frobenius form).
  \item $D_\nabla = q^{-1}\,\nabla + (1 - q^{-1})\,\iota_\xi\,\nabla$ is
        the \emph{$q$-deformed covariant derivative}, where $\xi$ is the
        Reeb derivation of~$A$
        (Definition~\ref{def:reeb-decomposition}).
\end{itemize}
\end{remark}

\begin{example}[Categorical-action notation for $\mathrm{SU}(2)$]
\label{ex:categorical-action-notation-su2}
Continuing Example~\ref{ex:connection-su2}, take
$A = \mathcal{O}(\mathrm{SU}(2))$ with Haar integral $\lambda = \int_G$,
connection $\nabla f = df + \mathcal{A} \cdot f$, and a state
$\psi \in A$ (a function on the group).
\begin{itemize}
  \item \emph{Categorical trace.}\;  For $T \in \mathrm{End}(V_j)$,
        $\mathrm{tr}(T) = \sum_m \langle j,m \mid T \mid j,m \rangle$
        is the usual matrix trace.
  \item \emph{Frobenius--Ricci.}\;  $R_\lambda$ measures curvature of
        the Haar integral itself: on $\mathrm{SU}(2)$ (flat Haar measure),
        $R_\lambda = 0$ in the classical limit $q = 1$.
  \item \emph{Adjoint section.}\;  For
        $\psi \colon \mathbb{C} \to A$, the adjoint is
        $\psi^*(f) = \int_G \overline{\psi(g)}\,f(g)\,dg$, i.e.\ the
        inner product with~$\psi$ under the Haar integral.
  \item \emph{$q$-deformed derivative.}\;  At $q = 1$,
        $D_\nabla = \nabla$ (ordinary covariant derivative).  For
        $q \neq 1$, the Reeb component is re-weighted, coupling the
        connection to the contact geometry.
\end{itemize}
\end{example}

\begin{definition}[Categorical action]
\label{def:categorical-action}
\uses{def:quantum-universe, def:quantum-observable-universe, def:categorical-connection, def:substrate-parameter, def:q-hodge-star}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) equipped with
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}),
$q$-Hodge star $\star_q$ (Definition~\ref{def:q-hodge-star}),
a connection $\nabla$ (Definition~\ref{def:categorical-connection}),
and a section $\psi \colon \mathbf{1} \to A$ (a ``state'').
The \emph{categorical action} is the $R$-valued functional
\begin{equation}\label{eq:categorical-action}
  \mathcal{S}_q[\nabla,\, \lambda,\, \psi]
  \;=\;
  \underbrace{%
    \mathrm{tr}\bigl(q \cdot R_\lambda\bigr)
  }_{\text{gravity}}
  \;+\;
  \underbrace{%
    \lambda \circ \mathrm{tr}_A\bigl(
      F_\nabla \wedge_q \star_q F_\nabla
    \bigr)
  }_{\text{gauge}}
  \;+\;
  \underbrace{%
    \lambda\bigl(\psi^* \cdot D_\nabla\,\psi\bigr)
  }_{\text{matter}},
\end{equation}
with notation as in Remark~\ref{rem:categorical-action-notation}.
\end{definition}

\begin{proposition}[Variational equations of the categorical action]
\label{prop:variational-equations}
Let $\mathcal{S}_q$ be the categorical action
(Definition~\ref{def:categorical-action}).  The three variational
equations are:
\begin{enumerate}
  \item \emph{Gravity (vary~$\lambda$)}:
    $\delta\mathcal{S}_q / \delta\lambda = 0$ yields the
    \emph{categorical Einstein equation}:
    \[
      R_\lambda - \tfrac{1}{2}\,\mathrm{tr}(R_\lambda)\,\mathrm{id}_A
      \;=\; T_q[\nabla, \psi],
    \]
    where $T_q$ is the stress-energy endomorphism constructed from the
    gauge and matter terms.
  \item \emph{Gauge (vary~$\nabla$)}:
    $\delta\mathcal{S}_q / \delta\nabla = 0$ yields the
    \emph{categorical Yang--Mills equation}:
    \[
      \nabla \star_q F_\nabla + q \cdot [F_\nabla,\, \star_q F_\nabla]
      \;=\; J_\psi,
    \]
    where $J_\psi = \psi^* \otimes D_\nabla\psi - D_\nabla\psi^* \otimes \psi$
    is the matter current.
  \item \emph{Matter (vary~$\psi$)}:
    $\delta\mathcal{S}_q / \delta\psi = 0$ yields the
    \emph{categorical transport equation}:
    \[
      D_\nabla\,\psi \;=\; 0.
    \]
\end{enumerate}
\end{proposition}

\begin{proof}
Each equation is obtained by computing
$\delta\mathcal{S}_q = 0$ with respect to the indicated variable, using
the non-degeneracy of~$\lambda$.

\emph{Item~(1): Vary $\lambda$.}\;
The gravity term $\mathrm{tr}(q \cdot R_\lambda)$ depends on~$\lambda$
through the Frobenius--Ricci endomorphism~$R_\lambda$.  Varying
$\lambda \mapsto \lambda + \delta\lambda$ and requiring
$\delta\mathcal{S}_q = 0$ for all~$\delta\lambda$ gives, by
non-degeneracy of the categorical trace,
$R_\lambda - \tfrac{1}{2}\,\mathrm{tr}(R_\lambda)\,\mathrm{id}_A
= T_q[\nabla, \psi]$,
where the stress-energy $T_q$ collects the $\lambda$-dependence of the
gauge and matter terms.  The trace subtraction ensures
$\mathrm{tr}(T_q) = \tfrac{1}{2}\,\mathrm{tr}(R_\lambda)$ (the
contracted Bianchi identity).

\emph{Item~(2): Vary $\nabla$.}\;
The gauge term
$\lambda \circ \mathrm{tr}_A(F_\nabla \wedge_q \star_q F_\nabla)$
depends on~$\nabla$ through $F_\nabla = \nabla^2$.  A standard
computation using $\delta F_\nabla = \nabla(\delta\nabla)$ and
integration by parts (via non-degeneracy of~$\lambda$) yields
$\nabla \star_q F_\nabla + q \cdot [F_\nabla,\, \star_q F_\nabla]
= J_\psi$.
The commutator term $q \cdot [F_\nabla, \star_q F_\nabla]$ arises from
the non-commutativity of the $q$-deformed wedge product; it vanishes
when $q = 1$ (the abelian/classical limit).

\emph{Item~(3): Vary $\psi$.}\;
The matter term $\lambda(\psi^* \cdot D_\nabla\,\psi)$ is linear
in~$\psi$.  Varying $\psi \mapsto \psi + \delta\psi$ and using
$(\delta\psi)^* \cdot D_\nabla\,\psi = 0$ for
all~$\delta\psi$ gives $D_\nabla\,\psi = 0$ by non-degeneracy
of~$\lambda$ and the dagger structure
(Definition~\ref{def:positive-definite-dagger}).
\end{proof}

\begin{definition}[Categorical $q$-deformed exterior derivative]
\label{def:categorical-q-ext-deriv}
\lean{QOU.CategoricalQExtDeriv}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
connection $\nabla$ (Definition~\ref{def:categorical-connection}),
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}),
and weight decomposition $\pi_H, \pi_V$
(Definition~\ref{def:weight-decomposition}).  The \emph{$q$-deformed
exterior derivative} $d_q \colon \Omega^k(A) \to \Omega^{k+1}(A)$ is
\[
  d_q \;=\; q^{-1}\,d + (1 - q^{-1})\,\pi_V \circ d,
\]
where $d \colon \Omega^k(A) \to \Omega^{k+1}(A)$ is the differential
of the non-commutative de~Rham complex.  Equivalently, $d_q$ acts as
$q^{-1}\,d$ on horizontal forms and as $d$ on vertical forms:
\begin{equation}\label{eq:categorical-dq-cd}
\begin{tikzcd}[row sep=2.5em, column sep=2.5em]
  \Omega^k_H(A) \ar[r, "d"] \ar[d, hook]
  & \Omega^{k+1}(A) \ar[r, "q^{-1}"]
  & \Omega^{k+1}(A) \\
  \Omega^k(A) \ar[rru, "d_q"']
\end{tikzcd}
\end{equation}
The \emph{$q$-adjoint} $\delta_q \colon \Omega^{k+1}(A) \to \Omega^k(A)$
is the adjoint of $d_q$ with respect to the dagger structure
(Definition~\ref{def:positive-definite-dagger}).
\end{definition}

\begin{definition}[Particle]
\label{def:q-harmonic-form}
\lean{QOU.QHarmonicForm}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
$q$-deformed exterior derivative $d_q$
(Definition~\ref{def:categorical-q-ext-deriv}).  The \emph{$q$-Laplacian}
on $\Omega^k(A)$ is $\Delta_q = d_q\,\delta_q + \delta_q\,d_q$.
A \emph{particle} is an element $\alpha \in \Omega^k(A)$ that is
$q$-harmonic:
\[
  \Delta_q\,\alpha \;=\; 0.
\]
\end{definition}

\begin{definition}[Fermion--Boson decomposition]
\label{def:fermion}
\label{def:boson}
\lean{QOU.Fermion}\leanok
Let $\mathcal{H}_q^k = \ker\Delta_q \subset \Omega^k(A)$ denote the
space of particles (Definition~\ref{def:q-harmonic-form}).  The weight
decomposition (Definition~\ref{def:weight-decomposition}) restricts to
a short exact sequence
\begin{equation}\label{eq:fermion-boson-ses}
\begin{tikzcd}[column sep=2.5em]
  0 \ar[r]
  & \mathcal{H}_{q,H}^k \ar[r, hook]
  & \mathcal{H}_q^k \ar[r, "\pi_V"]
  & \mathcal{H}_{q,V}^k \ar[r]
  & 0
\end{tikzcd}
\end{equation}
where $\mathcal{H}_{q,H}^k = \mathcal{H}_q^k \cap \Omega^k_H(A)$
and $\mathcal{H}_{q,V}^k = \pi_V(\mathcal{H}_q^k)$.  A \emph{boson}
is a particle in the kernel $\mathcal{H}_{q,H}^k$ (purely horizontal:
$\alpha_V = 0$).  A \emph{fermion} is a particle with
$\alpha_V \neq 0$, i.e.\ one whose image under $\pi_V$ is non-zero.

The sequence splits via $\pi_H$:
\begin{equation}\label{eq:fermion-boson-split}
\begin{tikzcd}[column sep=2.5em]
  \mathcal{H}_q^k
    \ar[r, "\pi_H \oplus \pi_V", "\sim"']
  & \mathcal{H}_{q,H}^k \;\oplus\; \mathcal{H}_{q,V}^k
\end{tikzcd}
\qquad
\text{(bosons $\oplus$ fermion vertical parts).}
\end{equation}
\end{definition}

\begin{definition}[Categorical point]
\label{def:categorical-point}
Let $(\mathcal{C}, A, \omega)$ be a quantum universe
(Definition~\ref{def:quantum-universe}).  A \emph{categorical point}
of~$A$ is a morphism
\[
  p \;\colon\; \mathbf{1} \;\longrightarrow\; A
  \qquad\text{in } \mathcal{C}.
\]
In the archimedean realisation $A = \mathcal{O}(G)$, a point
$p \colon \mathbf{1} \to A$ is a character $\mathrm{ev}_g \colon f \mapsto f(g)$
for some $g \in G$; equivalently, a point of the group~$G$.  For a
manifold $M$ with $A = C^{\infty}(M)$, a point $p$ corresponds to
evaluation at $p \in M$:
\begin{equation}\label{eq:point-diagram}
\begin{tikzcd}[column sep=2.5em]
  \{*\} \ar[r, "p"]
  & M
\end{tikzcd}
\qquad\longleftrightarrow\qquad
\begin{tikzcd}[column sep=2.5em]
  \mathbf{1} \ar[r, "p"]
  & A
\end{tikzcd}
\end{equation}
The \emph{augmentation ideal} at~$p$ is
$\mathfrak{m}_p = \ker(p^* \colon A \to \mathbf{1})$, where
$p^* = \lambda \circ \mu(p \otimes -)$.  The \emph{germ} of~$A$
at~$p$ is the quotient $A / \mathfrak{m}_p^2$, which captures the
first-order behaviour of~$A$ near~$p$.
\end{definition}

\begin{definition}[Observation]
\label{def:observation}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe over a
ring~$R$ (Definition~\ref{def:quantum-observable-universe}), let $q$
denote the substrate parameter
(Definition~\ref{def:substrate-parameter}), and let
$p \colon \mathbf{1} \to A$ be a categorical point
(Definition~\ref{def:categorical-point}).  An
\emph{observation at~$p$} is a morphism
\[
  \mathcal{O}_p \;\colon\; A / \mathfrak{m}_p^2
  \;\longrightarrow\; A / \mathfrak{m}_p^2
  \qquad\text{in } \mathcal{C},
\]
acting on germs at~$p$, satisfying:
\begin{enumerate}
  \item \textbf{Frobenius compatibility.}\; The observation preserves
    the Frobenius form at~$p$:
    \begin{equation}\label{eq:observation-frobenius}
    \begin{tikzcd}[row sep=2.5em, column sep=3em]
      A / \mathfrak{m}_p^2 \ar[r, "\mathcal{O}_p"] \ar[dr, "\lambda_p"']
      & A / \mathfrak{m}_p^2 \ar[d, "\lambda_p"] \\
      & \mathbf{1}
    \end{tikzcd}
    \end{equation}
    where $\lambda_p$ is the germ of the Frobenius form at~$p$.
  \item \textbf{Substrate perturbation.}\; The observation shifts the
        deformation parameter $q \mapsto q' = q + \delta q$ via the
        Hopf coaction:
    \begin{equation}\label{eq:observation-substrate}
    \begin{tikzcd}[row sep=2.5em, column sep=3em]
      A \ar[r, "\Delta"] \ar[d, "\mathcal{O}_p"']
      & A \otimes A \ar[r, "\mathcal{O}_p \otimes S"]
      & A \otimes A \ar[d, "\mu"] \\
      A \ar[rr, "\lambda"']
      & & \mathbf{1}
    \end{tikzcd}
    \end{equation}
        with $\delta q(p) = \lambda(\mathcal{O}_p(a_{(1)})\,S(a_{(2)}))$
        evaluated at the point~$p$.
\end{enumerate}
\end{definition}

\begin{remark}[Two-observer parallelogram]
\label{rem:two-observer-parallelogram}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
substrate parameter~$q$, and let
$p \colon \mathbf{1} \to A$ be a categorical point
(Definition~\ref{def:categorical-point}).  Let
$\alpha$ be a fermion (Definition~\ref{def:fermion}) with germ
$\alpha_p \in A / \mathfrak{m}_p^2$.
Consider two independent observations at~$p$,
$\mathcal{O}_{1,p},\, \mathcal{O}_{2,p} \colon
A / \mathfrak{m}_p^2 \to A / \mathfrak{m}_p^2$
(Definition~\ref{def:observation}).  These data
assemble into a \emph{parallelogram of observations at~$p$}:
\begin{equation}\label{eq:observation-parallelogram}
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  & \mathcal{O}_{1,p}(\alpha_p) \arrow[dr, "\text{compare}"] & \\
  \alpha_p \arrow[ur, "\mathcal{O}_{1,p}"]
           \arrow[dr, "\mathcal{O}_{2,p}"']
  & & \delta\alpha_p \\
  & \mathcal{O}_{2,p}(\alpha_p) \arrow[ur, "\text{compare}"'] &
\end{tikzcd}
\end{equation}
The discrepancy is the \emph{commutator of observations at~$p$}:
\[
  [\mathcal{O}_{1,p},\, \mathcal{O}_{2,p}](\alpha_p)
  \;=\;
  \mathcal{O}_{1,p}(\mathcal{O}_{2,p}(\alpha_p))
  - \mathcal{O}_{2,p}(\mathcal{O}_{1,p}(\alpha_p)).
\]
Each observation perturbs the substrate parameter at~$p$ by
$\delta q_i(p)$ (Definition~\ref{def:observation}, condition~(2)),
inducing shifted parameters $q_1'(p) = q(p) + \delta q_1(p)$
and $q_2'(p) = q(p) + \delta q_2(p)$.  The commutator
$[q_1'(p),\, q_2'(p)]$ measures the obstruction:
\[
  [\mathcal{O}_{1,p},\, \mathcal{O}_{2,p}](\alpha_p)
  \;=\;
  [q_1'(p),\, q_2'(p)] \cdot (\alpha_V)_p.
\]
This vanishes if and only if $[q_1'(p), q_2'(p)] = 0$ (the shifted
parameters commute at~$p$) or $(\alpha_V)_p = 0$ (the particle is a
boson at~$p$).  Generically, the non-commutativity of the Hopf
structure ensures $[q_1'(p), q_2'(p)] \neq 0$: independent
observations of a fermion at the same point necessarily disagree.

\medskip\noindent\textbf{Why germs, not global morphisms.}\;
The categorical algebra~$A$ carries no intrinsic topology (it is an
object in~$\mathcal{C}$, not a topological ring).  Working with germs
$A / \mathfrak{m}_p^2$ at a point~$p$ plays the role of localisation:
it captures the first-order jet of the observation without requiring a
topology on~$A$.  In the archimedean realisation
(Example~\ref{def:c-quantum-observable-universe}), where
$A = C^{\infty}(M)$, germs at~$p$ recover the usual smooth germs
$C^{\infty}_p(M) / \mathfrak{m}_p^2$.
\end{remark}

\begin{definition}[Categorical instanton]
\label{def:categorical-instanton}
\lean{QOU.CategoricalInstanton}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
(Definition~\ref{def:quantum-observable-universe}) with connection
$\nabla$ (Definition~\ref{def:categorical-connection}) and
$q$-Hodge star $\star_q$ (Definition~\ref{def:q-hodge-star}).
The connection $\nabla$ is an \emph{instanton} if its curvature
$F_\nabla \in \Omega^2(A)$ is $\star_q$-self-dual:
\[
  \star_q\, F_\nabla \;=\; F_\nabla.
\]
Equivalently, the curvature factors through the self-dual projection:
\begin{equation}\label{eq:instanton-cd}
\begin{tikzcd}[row sep=2.5em, column sep=3em]
  A \otimes A \ar[r, "\nabla \circ \nabla"]
    \ar[dr, "0"']
  & \Omega^2(A) \ar[d, "\pi^-"] \\
  & \Omega^-(A)
\end{tikzcd}
\end{equation}
where $\pi^- = \tfrac{1}{2}(\mathrm{id} - \star_q)$ is the
anti-self-dual projector and $\Omega^-(A) = \ker(\mathrm{id} + \star_q)$.
\end{definition}

\begin{lemma}[Irreducibility from observability]
\label{lem:categorical-irreducible}
\lean{QOU.categorical_irreducible}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe.
Any categorical instanton $\nabla$
(Definition~\ref{def:categorical-instanton}) is irreducible:
$\operatorname{Hom}_{\mathcal{C}}(\mathbf{1},\,\mathrm{Ad}(A)) = 0$.
\end{lemma}

\begin{proof}
Suppose $s \colon \mathbf{1} \to \mathrm{Ad}(A)$ is a non-zero
invariant morphism.  The following diagram commutes:
\[
\begin{tikzcd}[row sep=2.5em, column sep=3em]
  \mathbf{1} \ar[r, "s"] \ar[dr, "0"']
  & \mathrm{Ad}(A) \ar[d, "\lambda|_{\mathrm{Ad}}"] \\
  & \mathbf{1}
\end{tikzcd}
\]
Since $\mathrm{Ad}(A)$ is the trace-free part of $A \otimes A^*$,
the Frobenius form $\lambda$ vanishes on it:
$\lambda \circ s = 0$.  But $\lambda$ is non-degenerate by the
observability axiom, so $s = 0$.
\end{proof}

\begin{proposition}[Categorical self-duality]
\label{prop:categorical-self-duality}
\lean{QOU.categorical_self_duality}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe
with volume form $\mathrm{vol} \in \Omega^n(A)$
(Definition~\ref{def:q-hodge-star}).  Every connection $\nabla$
that is a critical point of the categorical Yang--Mills functional
\[
  S[\nabla] \;=\; \lambda\!\bigl(\star_q(F_\nabla) \cdot F_\nabla\bigr)
\]
is an instanton (Definition~\ref{def:categorical-instanton}).  That is,
the following diagram commutes:
\begin{equation}\label{eq:self-duality-cd}
\begin{tikzcd}[row sep=2.5em, column sep=3.5em]
  \Omega^2(A) \ar[r, "\star_q"] \ar[d, "\lambda(\,\cdot\, {*}\, \cdot\,)"']
  & \Omega^2(A) \ar[d, "\lambda(\,\cdot\, {*}\, \cdot\,)"] \\
  \mathbf{1} \ar[r, "\mathrm{id}"']
  & \mathbf{1}
\end{tikzcd}
\qquad\Longrightarrow\qquad
F_\nabla \in \Omega^+(A).
\end{equation}
\end{proposition}

\begin{proof}
By the Bogomolny decomposition in the dagger category,
\[
  \lambda(F_\nabla^* \cdot F_\nabla)
  \;=\; 2\,\lambda({F_\nabla^-}^* \cdot F_\nabla^-)
        + \lambda(F_\nabla \wedge_q F_\nabla),
\]
where the second term is topological (determined by the class
$[F_\nabla] \in \Omega^2(A)/\mathrm{exact}$).  Minimising
$S[\nabla]$ forces $F_\nabla^- = 0$, which is exactly the
self-duality condition.
\end{proof}

\begin{proposition}[Non-vanishing of $d_q^2$]
\label{prop:categorical-dq-squared}
\lean{QOU.categorical_dq_squared}\leanok
Let $d_q$ be the categorical $q$-deformed exterior derivative
(Definition~\ref{def:categorical-q-ext-deriv}).  The composition
$d_q^2 = d_q \circ d_q \colon \Omega^k(A) \to \Omega^{k+2}(A)$ is
given by
\[
  d_q^2\,\alpha
  \;=\; q^{-1}(1 - q^{-1})\;\omega_V \wedge_q [\xi,\, \alpha],
\]
where $\omega_V \in \Omega^2(A)$ is the curvature of the weight
decomposition (the image of the unit $1_A$ under
$\pi_V \circ d \circ d$) and $[\xi, \alpha]$ is the Lie derivative
along the generator $\xi$ of the $T$-action
(Definition~\ref{def:weight-decomposition}).  The following diagram
displays the obstruction to nilpotence:
\begin{equation}\label{eq:dq-squared-cd}
\begin{tikzcd}[row sep=2.5em, column sep=2.5em]
  \Omega^k(A) \ar[r, "d_q"] \ar[dr, "d_q^2"']
  & \Omega^{k+1}(A) \ar[d, "d_q"] \\
  & \Omega^{k+2}(A)
\end{tikzcd}
\qquad
d_q^2 = 0 \;\Longleftrightarrow\;
q = 1 \;\text{ or }\; [\xi, \alpha] = 0.
\end{equation}
In particular, $d_q^2 \neq 0$ for generic $\alpha$ whenever
$q \neq 1$: the $q$-deformed de~Rham complex is \emph{not} a cochain
complex.  On $\xi$-invariant forms (the horizontal subcomplex
$\Omega^k_H(A)$), $d_q$ does reduce to $q^{-1} d$ and $d_q^2 = 0$.
\end{proposition}

\begin{proof}
Write $a = q^{-1}$ and $b = 1 - q^{-1}$, so $d_q = a\,d + b\,\pi_V d$.
Then
\begin{align*}
  d_q^2
  &= (a\,d + b\,\pi_V d)(a\,d + b\,\pi_V d) \\
  &= a^2\,\underbrace{d^2}_{= 0}
     + ab\,(d \circ \pi_V d + \pi_V d \circ d)
     + b^2\,\underbrace{\pi_V d \circ \pi_V d}_{= 0}.
\end{align*}
The first term vanishes because $d^2 = 0$ in the non-commutative
de~Rham complex.  The last term vanishes because $\pi_V \circ \pi_V$
acts as zero on the image of $d$ restricted to forms already in the
vertical summand (by degree considerations in the weight grading).
The middle term simplifies:
$d(\pi_V d\,\alpha) + \pi_V d(d\,\alpha) = \omega_V \wedge_q [\xi, \alpha]$
where $\omega_V = \pi_V(d^2_{\text{total}})$ represents the curvature
of the weight projection, and the commutator with $\xi$ arises
from $[d, \pi_V] = [\xi, {-}]$ on the graded pieces.
\end{proof}

\begin{definition}[Categorical transport equation]
\label{def:categorical-transport}
\lean{QOU.CategoricalTransport}\leanok
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
instanton $\nabla$ (Definition~\ref{def:categorical-instanton}) and
$q$-deformed exterior derivative $d_q$
(Definition~\ref{def:categorical-q-ext-deriv}).  The \emph{categorical
transport equation} for a section $\psi \in \omega(J^1(A))$
(Definition~\ref{def:categorical-jet}) is
\begin{equation}\label{eq:categorical-transport}
  \hbar_q\;\pi_V(\nabla\,\psi) \;=\; H_q(\psi),
\end{equation}
where $H_q \colon \omega(J^1(A)) \to \omega(J^1(A))$ is the
\emph{$q$-Hamiltonian} endomorphism
$H_q = -\tfrac{1}{2}\hbar_q\,\Delta_q + V_q$
(with $V_q$ the potential arising from the curvature $F_\nabla$).
The following diagram encodes the transport as the bottom row of the
main commutative diagram:
\begin{equation}\label{eq:transport-cd}
\begin{tikzcd}[row sep=2.5em, column sep=3em]
  \omega(J^1(A)) \ar[r, "\nabla"] \ar[d, "j^1"']
  & \omega(J^1(A)) \otimes \Omega^1 \ar[d, "\pi_V"] \\
  \omega(A) \ar[r, "H_q"']
  & \omega(J^1(A))
\end{tikzcd}
\end{equation}
\end{definition}

\begin{definition}[Categorical $q$-Lie bracket]
\label{def:categorical-q-lie-bracket}
Let $(\mathcal{C}, A, \omega)$ be a quantum observable universe with
Reeb derivation $\xi$ (Definition~\ref{def:categorical-reeb}) and
substrate parameter~$q$ (Definition~\ref{def:substrate-parameter}).
The \emph{$q$-deformed Lie bracket} on derivations of~$A$ is
\[
  [X, Y]_q \;=\; q^{-1}\,[X, Y]
  + (1 - q^{-1})\,\bigl(\theta_A(X)\,[\xi, Y]
                        - \theta_A(Y)\,[\xi, X]\bigr),
\]
where $\theta_A$ is the canonical $1$-form dual to~$\xi$ and
$[X, Y]$ is the commutator of derivations in~$\mathcal{C}$.
\end{definition}

\begin{definition}[Categorical $q$-deformed connection]
\label{def:categorical-q-connection}
Let $\nabla$ be a connection on~$A$
(Definition~\ref{def:categorical-connection}) and $\xi$ the Reeb
derivation (Definition~\ref{def:categorical-reeb}).  The
\emph{$q$-deformed connection} is
\[
  \nabla_q \;=\; q^{-1}\,\nabla
  + (1 - q^{-1})\,(\xi \lrcorner\, \nabla)\,\theta_A.
\]
At $q = 1$, $\nabla_q = \nabla$.
\end{definition}

\begin{definition}[Categorical $q$-Reeb flow]
\label{def:categorical-q-reeb-flow}
Let $\xi$ be the Reeb derivation of~$A$
(Definition~\ref{def:categorical-reeb}) and $[\,\cdot\,,\cdot\,]_q$
the $q$-Lie bracket (Definition~\ref{def:categorical-q-lie-bracket}).
The \emph{$q$-deformed Reeb derivation} $\xi_q$ is the derivation of~$A$
defined by
\[
  \xi_q(a) \;=\; q^{-1}\,\xi(a)
  + (1 - q^{-1})\,[\xi,\, X_a]_q,
\]
where $X_a$ is the horizontal derivation associated to~$a$ (i.e.\
$\pi_H(da) = X_a \lrcorner\, \theta_A$).  The \emph{$q$-Reeb flow}
is the one-parameter family of automorphisms
$\phi_t^q \in \mathrm{Aut}(A)$ generated by $\xi_q$.
\end{definition}

\begin{definition}[Categorical $q$-torsion leaf]
\label{def:categorical-q-torsion-leaf}
A \emph{$q$-torsion leaf} is a maximal sub-object
$L \hookrightarrow A$ such that:
\begin{enumerate}
  \item \textbf{Horizontality:}\; $L \subset \ker(\pi_V)$
        (all forms on~$L$ are horizontal).
  \item \textbf{Flow invariance:}\; $\phi_t^q(L) = L$ for all
        $t$ (the $q$-Reeb flow preserves~$L$).
\end{enumerate}
\end{definition}

\section{Archimedean quantum universe}
\label{sec:archimedean-qou}

We now specialise the categorical framework to the archimedean setting,
where the base ring is an archimedean field $K$ (e.g.\
$\mathbb{Q}$, $\mathbb{R}$, or $\mathbb{C}$) and the quantum universe
is realised by CR geometry on a contact manifold.

\begin{example}[Archimedean realisation: quantum universe]
\label{def:c-quantum-universe}
\lean{QOU.CRManifoldData}\leanok
Let $K$ be an archimedean field.  A quantum universe over~$K$
(Definition~\ref{def:quantum-universe}) is realised by a $5$-tuple
$(M, \theta, G, P, E)$ where:
\begin{enumerate}
  \item $M$ is a CR manifold of hypersurface type
        (real dimension $2n+1$, $n \geq 1$) with non-degenerate Levi form
        of signature $(r,s)$, $r + s = n$;
  \item $\theta$ is a pseudo-Hermitian structure (contact form) on $M$,
        determining the Levi distribution $H = \ker\theta$, the Reeb vector
        field $\xi$, the Webster metric $g_\theta$ (of signature $(2r,2s)$
        on $H$), and the Levi form $L_\theta$;
  \item $G = \mathrm{SU}(2)$;
  \item $P \to M$ is a principal $G$-bundle whose holonomy reduces to
        $\mathrm{SO}(3) \subset \mathrm{SU}(2)/\{\pm I\}$;
  \item $E = P \times_G \mathbb{P}^1(\mathbb{C}) \to M$ is the associated
        \emph{state bundle}, where $\mathrm{SU}(2)$ acts on
        $\mathbb{P}^1(\mathbb{C})$ by M\"obius transformations.
\end{enumerate}
The monoidal category is $\mathcal{C} = \mathrm{Rep}(\mathrm{SU}_{q(p)}(2))$
(the representation category of the quantum group at the substrate
parameter~$q(p)$, Definition~\ref{def:substrate-parameter}), the
Frobenius--Hopf object is
$A = \mathcal{O}_q(\mathrm{SU}(2))$ with Frobenius form given by the
Haar state, and the fibre functor $\omega$ is the forgetful functor
to $K\text{-}\mathrm{Mod}$.
\end{example}

\begin{remark}[Notation and special cases]
\label{rem:notation-special-cases}
When $\theta$ is understood from context, we abbreviate the $5$-tuple to
$(M, G, P, E)$.  The \emph{strictly pseudoconvex} case corresponds to
signature $(n,0)$, where $L_\theta$ is positive definite.
\end{remark}

\begin{example}[Archimedean realisation: quantum observable universe]
\label{def:c-quantum-observable-universe}
\lean{QOU.CQuantumObservableUniverse}\leanok
An \emph{archimedean quantum observable universe} is an archimedean quantum
universe $(M, \theta, G, P, E)$
(Example~\ref{def:c-quantum-universe}) that satisfies
the vanishing conditions
\[
  H^0(M, \mathrm{Ad}(P)) = 0
  \qquad\text{and}\qquad
  \Gamma_c(M, E) = 0.
\]
These analytic conditions are the archimedean realisation of the
categorical observability conditions of
Definition~\ref{def:quantum-observable-universe}: the Frobenius
non-degeneracy of the Haar state on $\mathcal{O}_q(\mathrm{SU}(2))$,

\end{example}

\begin{remark}[Archimedean realisations of categorical structures]

\begin{remark}[Archimedean substrate via the Levi form]
\label{rem:c-substrate-via-levi}
\label{rem:substrate-contraction}
\label{rem:substrate-exponential}
Let $A$ be an instanton connection on~$P$
(Definition~\ref{def:instanton}), $F_A^+$ its self-dual curvature
(Proposition~\ref{prop:self-dual-curvature}), and
$L_\theta(V, W) = -i\,d\theta(V, \bar{W})$ the Levi form.  Since
$\Lambda^{1,1} H^* \subset \Lambda^+ H^*$ on the $4$-dimensional
distribution~$H$, the Webster metric and Killing form yield a natural
contraction $\langle L_\theta, F_A^+\rangle_p \in \mathbb{R}$.  The
substrate parameter
(Definition~\ref{def:substrate-parameter}) specialises to the smooth
field
\[
  q(p) \;=\; 1 + \bigl\lvert\langle L_\theta,\, F_A^+ \rangle\bigr\rvert(p),
  \qquad p \in M,
\]
with $q = 1$ at classicality ($F_A^+ = 0$) and $q \to \infty$ in the
deep-quantum regime.  All downstream formulas ($\hbar_q$, commutators,
Yang--Mills) are algebraic in~$q$, so the formula $1 + x$ may equally be
replaced by $\exp(x)$ without affecting the theory; the algebraic form is
preferred because it is meaningful over every base ring~$R$.
Non-archimedean analogues appear in Chapter~\ref{chap:p-adic}.
\end{remark}

\begin{remark}[Heisenberg commutator and the Reeb vector]
\label{rem:heisenberg-commutator}
\label{rem:commutator-lie-analogy}
\label{rem:commutator-reeb-vector}
The Heisenberg uncertainty principle takes the form of a $q$-deformed
commutator.  Let $\hat{X}, \hat{P}$ be position and momentum observables
on the fibre $\mathbb{P}^1(\mathbb{C})$, and let $J \colon H \to H$ be the complex
structure on the Levi distribution ($J^2 = -\mathrm{Id}$; throughout this
paper $i$ is identified with the action of~$J$ on horizontal planes).
The commutator at $p \in M$ is
\[
  [\hat{X},\, \hat{P}]_q \;:=\; J\,\hbar_q(p),
  \qquad \hbar_q(p) = 1 - q(p)^{-1}.
\]
As $q \to 1$, $[\hat{X}, \hat{P}]_q \to 0$ (classicality); as
$q \to \infty$, $[\hat{X}, \hat{P}]_q \to J$ (maximal uncertainty).

Geometrically, $[\hat{X}, \hat{P}]_q$ equals $\theta(\xi_q)$, the
$q$-deformed Reeb field (Definition~\ref{def:q-reeb-vector-field})
evaluated on the contact form: $\xi_q$ is the unique direction along
which $d\theta$ degenerates, so it encodes the obstruction to
commutativity.  Algebraically, the commutator is the quantum-mechanical
counterpart of the Lie bracket $[e_a, e_b] = \epsilon_{abc}\,e_c$ in
$\mathfrak{su}(2)$ (Definition~\ref{def:su2-basis}); in the braid-group
setting it appears as $\sigma_1\sigma_2\sigma_1 = \sigma_2\sigma_1\sigma_2$
(Definition~\ref{def:braid-group}).
\end{remark}

\begin{remark}[$q$-Torsion leaves: existence and Frobenius obstruction]
\label{rem:q-torsion-existence}
The horizontal distribution $H = \ker\theta$ on a contact manifold is
maximally non-integrable: $d\theta|_{H \times H}$ is non-degenerate.
By the Frobenius theorem, maximal integral submanifolds of $H$ have
dimension at most~$n$ (half the rank of~$H$); these are the
\emph{Legendrian submanifolds}.  Definition~\ref{def:q-torsion-leaf}
therefore requires $L$ to be a Legendrian submanifold invariant under the
$q$-deformed Reeb flow.

Existence is not automatic: it depends on the dynamics of~$\xi_q$.
When $q \to 1$ and $\xi_q \to \xi$, invariant Legendrian submanifolds
exist whenever the Reeb flow is periodic or integrable (e.g.\ on
$S^3/\Gamma$).  For $q > 1$, the deformation generically breaks
periodicity, and one must appeal to KAM-type persistence results.

In the category of Lie algebroids over~$M$, the Frobenius integrability
condition is equivalent to $H$ carrying a Frobenius structure
(Definition~\ref{def:quantum-universe}).  Since the Levi form
$d\theta|_{H \times H}$ is non-degenerate, this Frobenius structure
\emph{fails} --- and it is precisely this failure that forces the
dimension bound on integral submanifolds and gives $q$-torsion leaves
their rigidity.
\end{remark}

\begin{definition}[$q$-Weighted Yang--Mills--Levi functional]
\label{def:q-yang-mills-levi}
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with instanton connection~$A$ and substrate parameter~$q$
(Definition~\ref{def:substrate-parameter}).  The
\emph{$q$-weighted Yang--Mills--Levi functional} is
\[
  S_q[A, \theta]
  \;=\; \int_M q(p)\,\lVert F_A^+|_H \rVert^2\,
        \theta \wedge (d\theta)^n,
\]
where $q(p) = 1 + \lvert\langle L_\theta, F_A^+\rangle\rvert(p)$
(Remark~\ref{rem:c-substrate-via-levi}).
\end{definition}

\begin{remark}[General Relativity as the ``Substrate Geometry'']
\label{rem:gr-substrate-geometry}
In this model the metric is effectively replaced by the Levi form
$L_\theta$ and its coupling to the gauge curvature $F_A$.  Gravity is
expressed through the Variational Bicomplex as the back-reaction of the
gauge fields on the CR manifold $M$.
Variation of $S_q$ (Definition~\ref{def:q-yang-mills-levi}) with respect
to the Levi form $L_\theta$ yields field equations of the schematic form
\[
  \frac{\delta S_q}{\delta L_\theta}
  \;=\; 0
  \qquad\Longrightarrow\qquad
  \mathrm{Ric}_{g_\theta} \;=\; T^{(q)},
\]
where $\mathrm{Ric}_{g_\theta}$ is the Webster Ricci curvature and
$T^{(q)}$ is a $q$-weighted stress-energy $2$-tensor on $H$ (both
coordinate-free objects).  The geometry of $M$ is forced to curve to accommodate the tension of the
$\mathrm{SU}(2)$ instantons.
\end{remark}

\begin{proposition}[Main Commutative Diagram]
\label{prop:main-commutative-diagram}
\lean{QOU.main_commutative_diagram}\leanok
Let $(M, \theta, G, P, E)$ be an archimedean quantum observable universe
with instanton connection $A$ (Definition~\ref{def:instanton}) and substrate
parameter $q$ (Definition~\ref{def:substrate-parameter}).  Let
$\mathbf{E}$ denote the Euler--Lagrange operator of the
$q$-weighted Yang--Mills--Levi functional $S_q$
(Definition~\ref{def:q-yang-mills-levi}), and let $\nabla^A$ denote
parallel transport along $A$.  The following diagram commutes:
\[
\begin{tikzcd}[row sep=2.5em, column sep=2.5em]
  J^1E
    \ar[r, "{\mathbf{E}[S_q]}"]
    \ar[d, "\text{jet evaluation}"']
  & \bigl\{ L_\theta : \mathrm{Ric}_{g_\theta} = T^{(q)} \bigr\}
    \ar[d, "{q = 1 + \lvert\langle L_\theta, F_A^+\rangle\rvert}"]
  \\
  \Gamma(E_{\mathrm{lin}})
    \ar[r, "\nabla^A"']
  & \bigl\{ \psi : J\hbar_q\,\iota_{\xi_q}D_A\psi = \hat{H}_q\psi \bigr\}
\end{tikzcd}
\]
where the top row sends a jet (a first-order perturbation of the gauge field)
to the Einstein-type field equation via the Euler--Lagrange operator; the
left column evaluates a jet at a fibre point to produce a linearised section;
the right column is the coupling through the substrate field $q$; and the
bottom row is the parallel-transport equation whose linearisation is the
Schr\"odinger equation.
\end{proposition}

\begin{proof}[Proof sketch]
The top row is the variational equation $\delta S_q / \delta L_\theta = 0$,
which involves $q(p)$ through the functional $S_q$.  The bottom row is the
$q$-deformed parallel-transport equation
$J\hbar_q\,\iota_{\xi_q}D_A\psi = \hat{H}_q\psi$
(Remark~\ref{rem:schrodinger-jet-coaction}), linearised at a chosen section.
The left column is the jet evaluation map
$j^1_p \sigma \mapsto \sigma(p)$ restricted to $E_{\mathrm{lin}}$.
Commutativity amounts to the statement that solving the variational
equation for $L_\theta$ and then substituting $q(p)$ into the transport
equation gives the same solution set $\psi$ as first evaluating the jet
and then transporting along $A$.  This follows from the chain rule:
$\mathbf{E}[S_q]$ determines $L_\theta$ (and hence $q$), which in turn
determines $\hat{H}_q$ and $\xi_q$ appearing in the transport equation.
A full proof requires the variational bicomplex machinery of
Chapter~\ref{chap:experimental} and is given there.
\end{proof}

\section{Preview: Lifting and Descent}
\label{sec:preview-lifting}

The two central structural results of this paper are:
\begin{itemize}
  \item \textbf{Jet Prolongation (Lifting).}  An archimedean quantum
        observable universe $(M, G, P, E)$ lifts canonically to a new
        quantum observable universe on the total space of the first jet
        bundle $J^1 E$ (Proposition~\ref{prop:jet-prolongation},
        Chapter~\ref{chap:torsion}).
  \item \textbf{Brane Tower Descent.}  Iterating fixed-point reductions
        under the $\mathrm{SU}(2)$ action produces a tower of nested
        quantum observable universes $(M_k)_{k \geq 0}$, terminating at a
        discrete set of algebraic invariants
        (Section~\ref{ssec:descent}, Chapter~\ref{chap:torsion}).
\end{itemize}
The categorical and analytic elaborations needed for these theorems ---
Tannaka--Krein reconstruction, Hopf--Frobenius diagrams, and the
$q$-deformed operator algebra --- are developed in
Chapter~\ref{chap:elaboration}.  The reader wishing to proceed directly
from the definitions above to geometric applications may skip to
Chapter~\ref{chap:brings-surface}.
